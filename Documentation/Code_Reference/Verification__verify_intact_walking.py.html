   1 """Recompute 60 seconds of intact walking and compare all saved state arrays.
   2 
   3 Run: python Verification/verify_intact_walking.py
   4 Output: a new local Simulation_Results/Verification_Runs folder, or --output.
   5 This checks numerical reproducibility from the supplied fixed input and model
   6 parameters. It neither fits parameters nor claims biological validation.
   7 """
   8 from pathlib import Path
   9 from datetime import datetime
  10 import argparse
  11 import hashlib
  12 import json
  13 import os
  14 import platform
  15 import sys
  16 
  17 ROOT = Path(__file__).resolve().parents[1]
  18 os.environ.setdefault('OPENBLAS_NUM_THREADS', '1')
  19 os.environ.setdefault('OMP_NUM_THREADS', '1')
  20 os.environ['CAT_PARAMETER_FOLDER'] = str(ROOT / 'Data/Model_Parameters')
  21 sys.path.insert(0, str(ROOT / 'Code'))
  22 import numpy as np
  23 from neuromechanics.load_model_configuration import load_model_configuration
  24 from neuromechanics.simulate_forward_dynamics import simulate_forward_dynamics
  25 
  26 
  27 def main():
  28     """Run the complete intact trial; exit with an error if comparison fails."""
  29     parser = argparse.ArgumentParser(description=__doc__)
  30     parser.add_argument('--output', type=Path, help='New folder for the recomputed arrays and comparison')
  31     args = parser.parse_args()
  32     output = args.output or ROOT / 'Simulation_Results/Verification_Runs' / datetime.now().strftime('%Y%m%d_%H%M%S_%f')
  33     output.mkdir(parents=True, exist_ok=False)
  34     cfg = load_model_configuration(ROOT / 'Data/Model_Parameters')
  35     cfg['seconds'] = 60.0
  36     print('Computing 60 seconds of intact walking...', flush=True)
  37     result, state = simulate_forward_dynamics(cfg)
  38     source = ROOT / 'Simulation_Results/Intact_Walking/forward_simulation_timeseries.npz'
  39     relative_tolerance, absolute_tolerance = 1e-7, 1e-8
  40     comparisons = {}
  41     with np.load(source, allow_pickle=False) as reference:
  42         for key in ['time', 'qpos', 'qvel', 'joint_q', 'grf', 'excitation', 'activation',
  43                     'force', 'fiber_length', 'fiber_velocity', 'mtu_length', 'tendon_length', 'pad_depth']:
  44             actual, expected = np.asarray(result[key]), reference[key]
  45             same_shape = actual.shape == expected.shape
  46             comparisons[key] = {
  47                 'shape_matches': same_shape,
  48                 'maximum_absolute_difference': float(np.max(np.abs(actual - expected))) if same_shape else None,
  49                 'within_tolerance': bool(same_shape and np.allclose(actual, expected,
  50                     rtol=relative_tolerance, atol=absolute_tolerance, equal_nan=True)),
  51             }
  52         outcome_matches = str(result['failure']) == str(reference['failure'])
  53     passed = outcome_matches and all(c['within_tolerance'] for c in comparisons.values())
  54     report = {'passed': passed, 'requested_duration_s': 60.0,
  55         'final_saved_time_s': float(result['time'][-1]), 'failure': str(result['failure']),
  56         'relative_tolerance': relative_tolerance, 'absolute_tolerance_in_native_units': absolute_tolerance,
  57         'comparisons': comparisons, 'python': platform.python_version(), 'numpy': np.__version__,
  58         'platform': platform.platform(), 'reference_sha256': hashlib.sha256(source.read_bytes()).hexdigest()}
  59     np.savez_compressed(output / 'recomputed_forward_timeseries.npz', **result)
  60     (output / 'numerical_reproduction_comparison.json').write_text(json.dumps(report, indent=2), encoding='utf-8')
  61     print(('PASS' if passed else 'FAIL') + ': ' + str(output), flush=True)
  62     return 0 if passed else 1
  63 
  64 
  65 if __name__ == '__main__':
  66     raise SystemExit(main())
