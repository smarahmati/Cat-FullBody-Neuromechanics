   3 from model_configuration.formulation_constants import FALL_PELVIS_HEIGHT_M, FALL_ROOT_PITCH_RAD

   6 def detect_body_fall(model, data):
   7     """A low pelvis, excessive pitch, or head/trunk contact terminates walking.
   8 
   9     These are explicit simulation failure criteria, not clinical thresholds.
  10     They do not generate support forces or modify the simulated state.
  11     """
  12     if data.qpos[1] < FALL_PELVIS_HEIGHT_M or abs(data.qpos[2]) > FALL_ROOT_PITCH_RAD:
  13         return True
  14     for contact in data.contact:
  15         names = [model.geom(int(i)).name for i in [contact.geom1, contact.geom2]]
  16         if "ground_contact" in names and any("head" in n or "torso" in n for n in names):
  17             return True
  18     return False
