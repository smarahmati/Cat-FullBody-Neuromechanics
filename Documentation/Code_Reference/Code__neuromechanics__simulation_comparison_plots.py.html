  17 class ReplayPlotPanel:
  18     """Keep axes and artists alive; redraw static labels only after selection/resize.
  19 
  20     A four-second window follows playback, with actual simulation-time labels.
  21     Its bounds stay between zero and the last saved sample. Seeking backward
  22     moves the window backward too, including after playback has finished.
  23     """
  24 
  25     def __init__(self, parent, result, target, pairs, peaks, period_s):
  26         self.result, self.target, self.pairs = result, target, pairs
  27         self.peaks, self.period = peaks, period_s
  28         self.names = list(result["names"])
  29         self.times = result["time"]
  30         # Number of seconds visible at once; short trials use their whole duration.
  31         self.window_seconds = 4.0
  32         # About four labeled time intervals fit comfortably within each small plot.
  33         self.time_tick_locator = MaxNLocator(nbins=4)
  34         self.figure = Figure(figsize=(9, 6), dpi=90)
  35         self.axes = self.figure.subplots(2, 2)
  36         self.figure.subplots_adjust(
  37             left=0.09,
  38             right=0.98,
  39             top=0.93,
  40             bottom=0.09,
  41             hspace=0.48,
  42             wspace=0.28,
  43         )
  44         self.canvas = FigureCanvasTkAgg(self.figure, master=parent)
  45         self.canvas.get_tk_widget().pack(fill="both", expand=True)
  46         self.background = None
  47         self.key = None
  48         self.artists = []
  49         self.curves = []
  50         self.markers = []
  51         self.frame_times = deque(maxlen=120)
  52         self.emg_cache = {}
  53         self.last_time = None
  54         self.needs_redraw = True
  55         self.canvas.mpl_connect("draw_event", self._capture_background)
  56 
  57     def _capture_background(self, event):
  58         self.background = self.canvas.copy_from_bbox(self.figure.bbox)
  59         self.needs_redraw = True

  61     def _setup(self, side, name, joint_index, normalized, show_emg):
  62         # Discard the previous selection's lines and moving indicators.
  63         self.curves = []
  64         # Store one moving dot for every simulated or reference trace.
  65         self.markers = []
  66         # Redraw only these changing objects during normal playback.
  67         self.artists = []
  68         # Store a current-time vertical line for each of the four plots.
  69         self.cursors = []
  70         # Cache movable time labels and vertical grid lines instead of rebuilding axes.
  71         self.time_ticks = []
  72         for ax in self.axes.ravel():
  73             ax.clear()
  74             ax.grid(alpha=0.2)
  75             # Start with the first four seconds, or the whole trial if it is shorter.
  76             ax.set_xlim(0.0, max(min(self.window_seconds, float(self.times[-1])), 0.001))
  77             # Standard x ticks would require a costly full canvas redraw while scrolling.
  78             ax.set_xticks([])
  79             # Reserve enough labels for the small set returned by the time tick locator.
  80             for tick_number in range(7):
  81                 # Move this vertical grid line to a labeled absolute time during update().
  82                 grid_line = ax.axvline(0.0, color="gray", alpha=0.2, lw=0.8, animated=True)
  83                 # Draw the corresponding time just below the horizontal axis.
  84                 tick_label = ax.text(0.0, -0.045, "", transform=ax.get_xaxis_transform(),
  85                                      ha="center", va="top", fontsize=10, animated=True)
  86                 # Keep the grid line and its label together with their assigned tick number.
  87                 self.time_ticks.append((tick_number, grid_line, tick_label))
  88                 # These lightweight objects can move without redrawing titles and legends.
  89                 self.artists.extend([grid_line, tick_label])
  90             # Create the moving current-time line; its initial position is zero.
  91             cursor = ax.axvline(0.0, color="#aa7b27", lw=1.0, animated=True)
  92             # Remember the line so update() can move it without rebuilding the axes.
  93             self.cursors.append(cursor)
  94             # Include the cursor among the objects redrawn at each playback frame.
  95             self.artists.append(cursor)
  96             # Label the horizontal coordinate as actual elapsed simulation time.
  97             ax.set_xlabel("Simulation time (s)")
  98             # Leave room for the movable numeric labels above the axis caption.
  99             ax.xaxis.set_label_coords(0.5, -0.15)
 100         channel = self.names.index(name)
 101         scale = self.peaks[channel] if normalized else 1.0
 102         traces = [
 103             [
 104                 (
 105                     self.result["excitation"][:, channel] / scale,
 106                     "#087f9c",
 107                     "-",
 108                     "Excitation",
 109                 ),
 110                 (
 111                     self.result["activation"][:, channel] / scale,
 112                     "#aa3484",
 113                     "-",
 114                     "Activation",
 115                 ),
 116             ]
 117         ]
 118         if show_emg and name in self.pairs:
 119             if name not in self.emg_cache:
 120                 ref = self.pairs[name]["reference"]
 121                 phase = np.arange(len(ref)) / len(ref)
 122                 self.emg_cache[name] = np.interp(
 123                     (self.times / self.period) % 1,
 124                     np.r_[phase, 1.0],
 125                     np.r_[ref, ref[0]],
 126                 )
 127             traces[0].append((self.emg_cache[name], "black", "--", "EMG (peak normalized)"))
 128         sign = ([-1, 1, -1, -1] if side < 2 else [1, -1, -1, -1])[joint_index]
 129         offset = ([0, np.pi, np.pi, np.pi] if side < 2 else [np.pi] * 4)[joint_index]
 130         traces.append(
 131             [
 132                 (
 133                     np.rad2deg(
 134                         offset
 135                         + sign * self.target["joint_q"][:, side, joint_index]
 136                     ),
 137                     "black",
 138                     "--",
 139                     "Reference",
 140                 ),
 141                 (
 142                     np.rad2deg(
 143                         offset
 144                         + sign * self.result["joint_q"][:, side, joint_index]
 145                     ),
 146                     "#087f9c",
 147                     "-",
 148                     "Forward",
 149                 ),
 150             ]
 151         )
 152         for component in [2, 0]:
 153             traces.append(
 154                 [
 155                     (self.target["grf"][:, side, component], "black", "--", "Reference"),
 156                     (
 157                         self.result["grf"][:, side, component],
 158                         "#087f9c",
 159                         "-",
 160                         "Raw paw force",
 161                     ),
 162                 ]
 163             )
 164         limb = ["LH", "RH", "LF", "RF"][side]
 165         joint = (
 166             ["Hip", "Knee", "Ankle", "MTP"]
 167             if side < 2
 168             else ["Shoulder", "Elbow", "Wrist", "MCP"]
 169         )[joint_index]
 170         titles = [
 171             name + (" / fixed peak" if normalized else " / absolute input"),
 172             limb + " " + joint,
 173             limb + " vertical GRF",
 174             limb + " anterior–posterior (AP) GRF",
 175         ]
 176         for panel, (ax, items) in enumerate(zip(self.axes.ravel(), traces)):
 177             ax.set_title(titles[panel], fontsize=11)
 178             ax.set_ylabel(
 179                 ["Input / activation", "Anatomical angle (deg)", "Force (N)", "Force (N)"][
 180                     panel
 181                 ]
 182             )
 183             lower = min(float(v[0].min()) for v in items)
 184             upper = max(float(v[0].max()) for v in items)
 185             if panel != 1:
 186                 lower = min(0.0, lower)
 187             margin = max(0.05, (upper - lower) * 0.08)
 188             ax.set_ylim(lower - margin, upper + margin)
 189             for values, color, style, label in items:
 190                 (line,) = ax.plot(
 191                     # update() supplies the saved samples inside the current time window.
 192                     [],
 193                     # Data remain unchanged; only the visible interval is selected.
 194                     [],
 195                     color=color,
 196                     linestyle=style,
 197                     label=label,
 198                     lw=1.3,
 199                     # Redraw the visible curve when the four-second window moves.
 200                     animated=True,
 201                 )
 202                 (marker,) = ax.plot([], [], "o", color=color, ms=4, animated=True)
 203                 self.curves.append((line, values))
 204                 self.markers.append((marker, values))
 205                 # Reuse both the curve and its current-time dot during scrolling.
 206                 self.artists.extend([line, marker])
 207             ax.legend(fontsize=7, loc="upper right")
 208         self.canvas.draw()

 210     def update(self, time_s, side, name, joint_index, normalized=True, show_emg=True):
 211         # Keep the displayed cursor inside the recorded interval, including at Reset.
 212         time_s = float(np.clip(time_s, 0.0, self.times[-1]))
 213         key = (side, name, joint_index, normalized, show_emg)
 214         if key != self.key:
 215             self.key = key
 216             self._setup(*key)
 217         elif time_s == self.last_time and not self.needs_redraw:
 218             return
 219         # Keep a four-second view centered on the current time where possible.
 220         trial_end = max(float(self.times[-1]), 0.001)
 221         # Short recordings need a shorter window so no time outside the trial appears.
 222         window_width = min(self.window_seconds, trial_end)
 223         # Clamp the moving window to the first and last available times.
 224         window_start = min(max(time_s - window_width / 2.0, 0.0), trial_end - window_width)
 225         # All four plots use exactly the same interval.
 226         window_end = window_start + window_width
 227         # Change the time transform without rebuilding static titles, legends, and borders.
 228         if self.axes[0, 0].get_xlim() != (window_start, window_end):
 229             # Move every plot together so excitation, angles, and forces stay aligned.
 230             for axis in self.axes.ravel():
 231                 # These are actual simulation seconds, never time relative to a cursor.
 232                 axis.set_xlim(window_start, window_end)
 233         # Choose readable absolute times for grid lines inside the visible window.
 234         tick_times = self.time_tick_locator.tick_values(window_start, window_end)
 235         # Keep labels within the actual recorded time range.
 236         tick_times = tick_times[(tick_times >= window_start) & (tick_times <= window_end)]
 237         # Move cached labels and grid lines using the same absolute-time transform.
 238         for tick_number, grid_line, tick_label in self.time_ticks:
 239             # Unused label slots must disappear rather than showing old time values.
 240             visible = tick_number < len(tick_times)
 241             # Show only the grid lines needed in this window.
 242             grid_line.set_visible(visible)
 243             # Give the matching label the same visibility.
 244             tick_label.set_visible(visible)
 245             # Update coordinates only for the label slots in use.
 246             if visible:
 247                 # Read this grid line's actual simulation time.
 248                 tick_time = float(tick_times[tick_number])
 249                 # Position the grid line at that time.
 250                 grid_line.set_xdata([tick_time, tick_time])
 251                 # Align its numeric label just below the same time coordinate.
 252                 tick_label.set_position((tick_time, -0.045))
 253                 # Avoid long floating-point strings in the small plot panels.
 254                 tick_label.set_text(f"{tick_time:.4g}")
 255         # Include one neighboring sample on either side for continuous boundary segments.
 256         first_sample = max(0, np.searchsorted(self.times, window_start) - 1)
 257         # Keep the next sample as well, without exceeding the saved array.
 258         last_sample = min(len(self.times), np.searchsorted(self.times, window_end, side="right") + 1)
 259         # Draw raw saved samples within the moving window; do not filter the waveforms.
 260         for line, values in self.curves:
 261             # Every visible point retains its absolute simulation time and original value.
 262             line.set_data(self.times[first_sample:last_sample], values[first_sample:last_sample])
 263         # Move all vertical lines to the current absolute simulation time.
 264         for cursor in self.cursors:
 265             # Two equal x coordinates make a vertical line at this time.
 266             cursor.set_xdata([time_s, time_s])
 267         for marker, values in self.markers:
 268             # Move each dot to this time and its interpolated signal value.
 269             marker.set_data([time_s], [np.interp(time_s, self.times, values)])
 270         if self.background is None:
 271             self.canvas.draw()
 272         self.canvas.restore_region(self.background)
 273         for artist in self.artists:
 274             artist.axes.draw_artist(artist)
 275         self.canvas.blit(self.figure.bbox)
 276         self.last_time = time_s
 277         self.needs_redraw = False
 278         self.frame_times.append(time.perf_counter())

 280     # Apply this function/property behavior.
 281     @property
 282     def measured_hz(self):
 283         if len(self.frame_times) < 2:
 284             return 0.0
 285         return (len(self.frame_times) - 1) / (self.frame_times[-1] - self.frame_times[0])
