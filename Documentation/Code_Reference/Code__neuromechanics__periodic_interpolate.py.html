  11 def periodic_interpolate(values, phase):
  12     """Linear interpolation of endpoint-inclusive cycle samples; never extrapolate."""
  13     values = np.asarray(values)
  14     x = (phase % 1.0) * (len(values) - 1)
  15     i = int(x)
  16     f = x - i
  17     return values[i] * (1 - f) + values[i + 1] * f
