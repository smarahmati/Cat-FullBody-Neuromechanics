  10 def apply_initial_parameter_file(state, path):
  11     """Apply this state once at t=0. Never reset it at a stride boundary."""
  12     p = load_parameter_file(path)
  13     m = state["model"]
  14     coordinates = [m.joint(i).name for i in range(m.njnt)]
  15     if m.nq != m.nv or m.nv != m.njnt:
  16         # Stop and explain the invalid model setting or failed calculation.
  17         raise ValueError("Named initial state currently supports scalar joints only")
  18     names = list(state["names"])
  19     pads = [m.geom(int(i)).name for i in state["pad_ids"]]
  20     for section, expected in [
  21         ("coordinates", coordinates),
  22         ("muscles", names),
  23         ("contact_memory", pads),
  24     ]:
  25         if set(p[section]) != set(expected):
  26             # Stop and explain the invalid model setting or failed calculation.
  27             raise ValueError(f"Initial {section} names do not match model")
  28     q = np.empty(m.nq)
  29     v = np.empty(m.nv)
  30     for i, name in enumerate(coordinates):
  31         q[m.jnt_qposadr[i]] = p["coordinates"][name]["position"]
  32         v[m.jnt_dofadr[i]] = p["coordinates"][name]["velocity"]
  33     apply_forward_initial_state(
  34         state,
  35         dict(
  36             qpos=q,
  37             qvel=v,
  38             activation=np.array([p["muscles"][n]["activation_0_to_1"] for n in names]),
  39             fiber_length=np.array([p["muscles"][n]["fiber_length_m"] for n in names]),
  40             bristle=np.array([p["contact_memory"][n] for n in pads]),
  41         ),
  42     )
