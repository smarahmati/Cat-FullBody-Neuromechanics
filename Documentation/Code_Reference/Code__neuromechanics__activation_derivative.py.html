  11 def activation_derivative(muscle_excitation, activation, time_constants):
  12     """MuJoCo muscle activation law; tau=[activation,deactivation] in seconds."""
  13     u = np.clip(muscle_excitation, 0, 1)
  14     a = np.clip(activation, 0, 1)
  15     factor = 0.5 + 1.5 * a
  16     tau = np.where(u > a, time_constants[:, 0] * factor, time_constants[:, 1] / factor)
  17     return (u - a) / tau
