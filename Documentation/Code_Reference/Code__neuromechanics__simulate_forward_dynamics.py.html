  19 def simulate_forward_dynamics(overrides=None, graphics=False, state=None):
  20     """One uninterrupted forward trial, saved without time warping/filtering."""
  21     state = (
  22         initialize_neuromechanical_model(overrides, graphics)
  23         if state is None
  24         else state
  25     )
  26     if state is not None:
  27         state["settings"] = dict(state["settings"], **(overrides or {}))
  28     model, data = state["model"], state["data"]
  29     settings = state["settings"]
  30     stepper = advance_partitioned_timestep
  31     if settings.get("forward_step_implementation") == "readable":
  32         from .advance_neuromechanical_timestep import advance_neuromechanical_timestep
  33 
  34         stepper = advance_neuromechanical_timestep
  35     history = {
  36         key: []
  37         for key in [
  38             "time",
  39             "qpos",
  40             "qvel",
  41             "joint_q",
  42             "grf",
  43             "activation",
  44             "excitation",
  45             "force",
  46             "fiber_length",
  47             "fiber_velocity",
  48             "mtu_length",
  49             "tendon_length",
  50             "pad_depth",
  51         ]
  52     }
  53     start = time.perf_counter()
  54     interval = max(1, round(settings["sample_s"] / model.opt.timestep))
  55     failure = None
  56     fall_time = np.nan
  57     peak_contact = np.abs(state["grf"]).copy()
  58     impulse = np.zeros((4, 3))
  59     max_fiber_residual = 0.0
  60     fiber_bound_steps = 0
  61     max_interior_residual = 0.0
  62     bound_channels = np.zeros(model.nu, dtype=int)
  63     peak_normalized_fiber_velocity = abs(
  64         state["fiber_velocity"]
  65         / (state["lopt"] * state["muscle_params"][:, 6])
  66     ).copy()
  67     for step in range(round(settings["seconds"] / model.opt.timestep) + 1):
  68         if step % interval == 0:
  69             mtu = data.actuator_length + state["mtu_length_offset"]
  70             tendon = mtu - state["fiber_length"] * state["cosine"]
  71             for key, value in dict(
  72                 time=data.time,
  73                 qpos=data.qpos,
  74                 qvel=data.qvel,
  75                 joint_q=data.qpos[state["joint_indices"]],
  76                 grf=state["grf"],
  77                 activation=data.act,
  78                 excitation=data.ctrl,
  79                 force=state["force"],
  80                 fiber_length=state["fiber_length"],
  81                 fiber_velocity=state["fiber_velocity"],
  82                 mtu_length=mtu,
  83                 tendon_length=tendon,
  84                 pad_depth=state["pad_depth"],
  85             ).items():
  86                 history[key].append(np.array(value, copy=True))
  87         if data.time >= settings["seconds"] - model.opt.timestep / 2:
  88             # Leave this loop because the convergence or stopping condition was met.
  89             break
  90         if not stepper(state):
  91             failure = "numerical instability"
  92             # Leave this loop because the convergence or stopping condition was met.
  93             break
  94         peak_contact = np.maximum(peak_contact, np.abs(state["grf"]))
  95         impulse += state["grf"] * model.opt.timestep
  96         peak_normalized_fiber_velocity = np.maximum(
  97             peak_normalized_fiber_velocity,
  98             abs(state["fiber_velocity"] / (state["lopt"] * state["muscle_params"][:, 6])),
  99         )
 100         max_fiber_residual = max(max_fiber_residual, float(np.max(abs(state["residual"]))))
 101         fiber_bound_steps += int(np.any(state["fiber_bound"]))
 102         interior = state["fiber_bound"] == 0
 103         if np.any(interior):
 104             max_interior_residual = max(
 105                 max_interior_residual,
 106                 float(np.max(abs(state["residual"][interior]))),
 107             )
 108         bound_channels += state["fiber_bound"].astype(int)
 109         # Pelvic collapse or a head/trunk collision. Never reset a fallen run.
 110         fall = detect_body_fall(model, data)
 111         if fall:
 112             if not np.isfinite(fall_time):
 113                 fall_time = data.time
 114             failure = "body fall"
 115             if settings["stop_on_fall"]:
 116                 # Leave this loop because the convergence or stopping condition was met.
 117                 break
 118     result = {k: np.array(v) for k, v in history.items()}
 119     result["names"] = state["names"]
 120     result["failure"] = np.array(failure or "")
 121     result["first_fall_time_s"] = np.array(fall_time)
 122     result["contact_peak_abs_N"] = peak_contact
 123     result["contact_impulse_Ns"] = impulse
 124     result["max_fiber_equilibrium_residual_N"] = np.array(max_fiber_residual)
 125     result["fiber_bound_steps"] = np.array(fiber_bound_steps)
 126     result["max_interior_fiber_residual_N"] = np.array(max_interior_residual)
 127     result["fiber_bound_steps_by_channel"] = bound_channels
 128     result["optimal_fiber_length_m"] = state["lopt"]
 129     result["tendon_slack_length_m"] = state["slack"]
 130     result["coordinate_names"] = np.array(
 131         [model.joint(model.dof_jntid[i]).name for i in range(model.nv)]
 132     )
 133     result["peak_abs_normalized_fiber_velocity"] = peak_normalized_fiber_velocity
 134     result["wall_seconds"] = np.array(time.perf_counter() - start)
 135     return result, state
