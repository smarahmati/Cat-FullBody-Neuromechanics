  13     PASSIVE_LENGTH_EXPONENT,

  18 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  19 @njit(cache=False)
  20 def muscle_force_length_velocity(normalized_length, normalized_velocity, muscle_parameters):
  21     if len(muscle_parameters) > 9:
  22         # Algebraic form of the original MATLAB equations with shortening Vmax<0.
  23         # c=aV/Fmax=0.00915*SO-0.00467; e=0.8/(aV+1).
  24         active_length_factor = np.exp(
  25             -abs(
  26                 (max(normalized_length, 0.0) ** ACTIVE_LENGTH_POWER - 1)
  27                 / ACTIVE_LENGTH_WIDTH
  28             )
  29             ** muscle_parameters[10]
  30         )
  31         c = muscle_parameters[11]
  32         e = muscle_parameters[12]
  33         velocity_factor = (
  34             max(0.0, c * (1 + normalized_velocity) / (c - normalized_velocity))
  35             if normalized_velocity < 0
  36             else (e + (1.8 + 0.8 * e) * normalized_velocity) / (e + normalized_velocity)
  37         )
  38         if len(muscle_parameters) > 14 and muscle_parameters[14] > 0:
  39             velocity_factor = muscle_force_velocity_degroote(normalized_velocity)[0]
  40         # Parallel elastic fiber force: Prilutsky et al. (2016), Table 10.3,
  41         # DOI 10.1007/978-1-4939-3267-2_10; Rahmati et al. (2025), Table 3,
  42         # DOI 10.1113/JP287448. Published coefficients: 0.0075 and 11.6.
  43         # The released configuration enables this term in ALL 104 channels.
  44         # Our tension-only adaptation sets force to zero at length <= l_opt;
  45         # the exponential cap is a numerical overflow guard. Neither choice
  46         # is a separately measured passive curve for each added muscle.
  47         # Archived MATLAB copies zeroed FPE; we follow the published law here.
  48         passive_length_factor = (
  49             (PASSIVE_LENGTH_SCALE / PASSIVE_LENGTH_EXPONENT)
  50             * np.expm1(
  51                 min(PASSIVE_EXPONENT_CAP, PASSIVE_LENGTH_EXPONENT * (normalized_length - 1))
  52             )
  53             if len(muscle_parameters) > 13
  54             and muscle_parameters[13] > 0
  55             and normalized_length > 1
  56             else 0.0
  57         )
  58         return active_length_factor, velocity_factor, passive_length_factor
  59     low, high = muscle_parameters[4], muscle_parameters[5]
  60     a = (low + 1) / 2
  61     b = (1 + high) / 2
  62     if normalized_length < low or normalized_length > high:
  63         active_length_factor = 0.0
  64     elif normalized_length <= a:
  65         active_length_factor = 0.5 * ((normalized_length - low) / (a - low)) ** 2
  66     elif normalized_length <= 1:
  67         active_length_factor = 1 - 0.5 * ((1 - normalized_length) / (1 - a)) ** 2
  68     elif normalized_length <= b:
  69         active_length_factor = 1 - 0.5 * ((normalized_length - 1) / (b - 1)) ** 2
  70     else:
  71         active_length_factor = 0.5 * ((high - normalized_length) / (high - b)) ** 2
  72     if normalized_velocity <= -1:
  73         velocity_factor = 0.0
  74     elif normalized_velocity <= 0:
  75         velocity_factor = (normalized_velocity + 1) ** 2
  76     elif normalized_velocity <= muscle_parameters[8] - 1:
  77         velocity_factor = muscle_parameters[8] - (
  78             muscle_parameters[8]
  79             - 1
  80             - normalized_velocity
  81         ) ** 2 / (muscle_parameters[8] - 1)
  82     else:
  83         velocity_factor = muscle_parameters[8]
  84     if normalized_length <= 1:
  85         passive_length_factor = 0.0
  86     elif normalized_length <= b:
  87         passive_length_factor = (
  88             0.5
  89             * muscle_parameters[7]
  90             * ((normalized_length - 1) / (b - 1)) ** 2
  91         )
  92     else:
  93         passive_length_factor = muscle_parameters[7] * (
  94             0.5
  95             + (normalized_length - b) / (b - 1)
  96         )
  97     return active_length_factor, velocity_factor, passive_length_factor
