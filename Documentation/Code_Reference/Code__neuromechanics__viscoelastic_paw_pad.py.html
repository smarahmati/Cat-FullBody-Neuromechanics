   7 from model_configuration.formulation_constants import SHEAR_DAMPING_DEPTH_M

  10 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  11 @njit(cache=False)
  12 def viscoelastic_paw_pad(depth, velocity, bristle, dt, parameters):
  13     """Kelvin--Voigt shear element in series with a Coulomb slider.
  14 
  15     The dashpot responds to elastic shear rate, not to the total sliding rate.
  16     This distinction makes the backward-Euler return map dissipative even
  17     during rapid slip. The same Hunt--Crossley normal law is retained.
  18     This is an explicit constitutive modeling choice, not measured cat tissue.
  19     """
  20     k, c, kt, bt, mu = parameters[:5]
  21     if depth <= 0:
  22         return np.zeros(3), np.zeros(2), 0.0
  23     elastic = k * depth**1.5
  24     normal_energy = k * depth**2.5 / 2.5
  25     # Optional finite-thickness-inspired progressive stiffness. This smooth,
  26     # conservative elastic term becomes important near the assumed depth.
  27     # Its exact coefficient is a modeling choice, not a cat tissue measurement.
  28     if len(parameters) > 5 and parameters[5] > 0.0:
  29         scale = parameters[5]
  30         elastic *= 1.0 + (depth / scale) ** 4
  31         normal_energy += k * depth**6.5 / (6.5 * scale**4)
  32     normal = elastic * max(0.0, 1.0 - c * velocity[2])
  33     if normal <= 0:
  34         return np.zeros(3), np.zeros(2), normal_energy
  35     damping = bt * min(1.0, depth / SHEAR_DAMPING_DEPTH_M)
  36     displacement = bristle + dt * velocity[:2]
  37     trial_stress = kt * displacement + damping * velocity[:2]
  38     magnitude = np.linalg.norm(trial_stress)
  39     # Optional smooth Stribeck-like sliding envelope. This is an engineering
  40     # friction assumption, NOT an identified cat-paw coefficient or an exact
  41     # reproduction of the Dupont et al. (2002) elastoplastic state equation.
  42     # The nonnegative return-map slip rate still dissipates stored shear energy.
  43     if len(parameters) > 7:
  44         dynamic_mu = parameters[6]
  45         vscale = parameters[7]
  46         speed2 = np.dot(velocity[:2], velocity[:2])
  47         mu = dynamic_mu + (mu - dynamic_mu) * np.exp(-speed2 / (vscale * vscale))
  48     limit = mu * normal
  49     stress = trial_stress.copy()
  50     if magnitude > limit:
  51         direction = trial_stress / magnitude
  52         stress = limit * direction
  53         if dt > 0:
  54             # Plastic slip rate gamma is nonnegative. The implicit spring and
  55             # dashpot share the elastic rate (v - gamma*direction).
  56             gamma = (magnitude - limit) / (kt * dt + damping)
  57             displacement -= dt * gamma * direction
  58         # With dt=0 this is a force evaluation only; stored strain is unchanged.
  59     force = np.empty(3)
  60     force[:2] = -stress
  61     force[2] = normal
  62     return (
  63         force,
  64         displacement,
  65         normal_energy + 0.5 * kt * np.dot(displacement, displacement),
  66     )
