   1 """Draw the saved four-paw force vectors in the passive MuJoCo viewer."""

   3 import mujoco as mj

   5 import numpy as np

   8 def draw_saved_grf_vectors(viewer, state, forces, enabled=True, metres_per_newton=0.006):
   9     """Use actual saved Ground_Reaction_Forces; arrow tails are ground projections of paw sites.
  10 
  11     The archive has no centre-of-pressure trace. A projected paw site is an
  12     explicit visualization anchor, not a measured or reconstructed COP.
  13     Forces below 0.05 N are hidden only as arrows; time series remain unchanged.
  14     Call with the viewer lock held. This viewer reserves user_scn for arrows.
  15     """
  16     scene = viewer.user_scn
  17     scene.ngeom = 0
  18     if not enabled:
  19         return []
  20     colors = [
  21         [0.05, 0.82, 1.0, 1.0],
  22         [1.0, 0.15, 0.58, 1.0],
  23         [1.0, 0.72, 0.05, 1.0],
  24         [0.68, 0.35, 1.0, 1.0],
  25     ]
  26     labels = ["LH", "RH", "LF", "RF"]
  27     records = []
  28     m, d = state["model"], state["data"]
  29     planes = np.flatnonzero(m.geom_type == int(mj.mjtGeom.mjGEOM_PLANE))
  30     ground_z = float(m.geom_pos[planes[0], 2]) if len(planes) else 0.0
  31     for side, force in enumerate(forces):
  32         magnitude = float(np.linalg.norm(force))
  33         if magnitude < 0.05:
  34             # Skip the remaining work for this item.
  35             continue
  36         origin = d.site_xpos[state["paw_ids"][side]].copy()
  37         origin[2] = ground_z + 0.002
  38         tip = origin + metres_per_newton * force
  39         geom = scene.geoms[scene.ngeom]
  40         mj.mjv_initGeom(
  41             geom,
  42             mj.mjtGeom.mjGEOM_ARROW,
  43             np.zeros(3),
  44             origin,
  45             np.eye(3).ravel(),
  46             np.asarray(colors[side], dtype=np.float32),
  47         )
  48         mj.mjv_connector(geom, mj.mjtGeom.mjGEOM_ARROW, 0.0035, origin, tip)
  49         geom.label = f"{labels[side]} {magnitude:.1f} N"
  50         scene.ngeom += 1
  51         records.append(
  52             dict(
  53                 limb=labels[side],
  54                 force_N=force.tolist(),
  55                 origin_m=origin.tolist(),
  56                 tip_m=tip.tolist(),
  57             )
  58         )
  59     return records
