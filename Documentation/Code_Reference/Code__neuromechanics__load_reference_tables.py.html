  14 def load_reference_tables(folder=None):
  15     """Return source cycles, LH assembly correction and the separate FHL EMG.
  16 
  17     Optional text tables are exactly the numeric source arrays, not polynomial
  18     fits. Each file has its own original sampling grid. The caller applies the
  19     documented fixed limb phase offsets. No reference is used as a force.
  20     """
  21     if folder is None:
  22         return (
  23             dict(np.load(ROOT / "Data/Experimental_Reference_Data/hindlimb_forelimb_reference_cycles.npz")),
  24             np.load(ROOT / "Data/Experimental_Reference_Data/hindlimb_pose_initialization.npz")[
  25                 "left_joint_correction_rad"
  26             ],
  27             np.load(ROOT / "Data/Experimental_Reference_Data/hindlimb_flexor_hallucis_longus_emg.npz")["FHL_peak_normalized"],
  28         )
  29     folder = ROOT / Path(folder)
  30     manifest = json.loads((folder / "TABLE_MANIFEST.json").read_text())
  31     out = {}
  32     for name, item in manifest["tables"].items():
  33         phase, channels, values = read_time_series(folder / item["file"], item["columns"])
  34         if not np.allclose(phase, np.linspace(0, 1, len(phase)), atol=1e-12, rtol=0):
  35             # Stop and explain the invalid model setting or failed calculation.
  36             raise ValueError(f"Reference {name} needs a uniform phase grid")
  37         out[name] = values.reshape((len(phase),) + tuple(item["shape_after_time"]))
  38     correction = out.pop("left_joint_correction_rad")
  39     fhl = out.pop("FHL_peak_normalized")
  40     return out, correction, fhl
