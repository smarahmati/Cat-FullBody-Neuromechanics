   6 def interpolate_simulation_for_display(result, time_s):
   7     """Return one display sample between the two surrounding archived samples.
   8 
   9     The present sagittal model has scalar translations/hinges (nq == nv),
  10     so coordinate interpolation does not involve quaternions. Recorded arrays
  11     and force peaks are never edited. No smoothing or new dynamics is applied.
  12     """
  13     times = result["time"]
  14     t = float(np.clip(time_s, times[0], times[-1]))
  15     right = min(len(times) - 1, int(np.searchsorted(times, t, side="right")))
  16     left = max(0, right - 1)
  17     weight = 0.0 if right == left else (t - times[left]) / (times[right] - times[left])
  18     fields = ["qpos", "qvel", "activation", "excitation", "force", "grf", "joint_q"]
  19     sample = {
  20         key: (1.0 - weight) * result[key][left] + weight * result[key][right]
  21         for key in fields
  22     }
  23     sample.update(time=t, left_index=left, right_index=right, fraction=float(weight))
  24     return sample
