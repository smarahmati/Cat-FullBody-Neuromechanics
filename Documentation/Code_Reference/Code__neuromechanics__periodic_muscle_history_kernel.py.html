  10 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  11 @njit(cache=False)
  12 def periodic_muscle_history_kernel(
  13     mtu,
  14     table,
  15     lopt,
  16     slack,
  17     params,
  18     e0,
  19     beta,
  20     time_constants,
  21     period,
  22     cycles,
  23 ):
  24     n, nm = mtu.shape
  25     dt = period / n
  26     a = np.zeros(nm)
  27     lf = np.maximum(0.21 * lopt, (mtu[0] - slack) / params[:, 9])
  28     before_a = a.copy()
  29     before_lf = lf.copy()
  30     for k in range(cycles * n):
  31         frame = k % n
  32         pos = frame / n * (len(table) - 1)
  33         left = int(pos)
  34         w = pos - left
  35         u = table[left] * (1 - w) + table[left + 1] * w
  36         factor = 0.5 + 1.5 * a
  37         tau = np.where(u > a, time_constants[:, 0] * factor, time_constants[:, 1] / factor)
  38         a = np.clip(a + dt * (u - a) / tau, 0.0, 1.0)
  39         lf, v, f, r, b = solve_elastic_muscle_equilibrium(
  40             mtu[(frame + 1) % n],
  41             lf,
  42             a,
  43             lopt,
  44             slack,
  45             params,
  46             e0,
  47             beta,
  48             dt,
  49         )
  50         if k == (cycles - 1) * n - 1:
  51             before_a = a.copy()
  52             before_lf = lf.copy()
  53     return a, lf, v, f, r, b, a - before_a, lf - before_lf
