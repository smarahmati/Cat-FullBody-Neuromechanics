  12     ACTIVE_LENGTH_POWER,
  13     ACTIVE_LENGTH_WIDTH,
  14     PASSIVE_LENGTH_SCALE,
  15     PASSIVE_LENGTH_EXPONENT,
  16     PASSIVE_EXPONENT_CAP,

  20 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  21 @njit(cache=False)
  22 def muscle_force_curve_derivatives(l, v, p):
  23     fl, fv, fp = muscle_force_length_velocity(l, v, p)
  24     if len(p) > 9:
  25         x = (max(l, 0.0) ** ACTIVE_LENGTH_POWER - 1) / ACTIVE_LENGTH_WIDTH
  26         dfl = (
  27             -fl
  28             * p[10]
  29             * abs(x) ** (p[10] - 1)
  30             * np.sign(x)
  31             * ACTIVE_LENGTH_POWER
  32             * max(l, 0.0) ** (ACTIVE_LENGTH_POWER - 1)
  33             / ACTIVE_LENGTH_WIDTH
  34         )
  35         c, e = p[11], p[12]
  36         if v < 0:
  37             dfv = c * (c + 1) / (c - v) ** 2 if fv > 0 else 0.0
  38         else:
  39             dfv = e * (0.8 + 0.8 * e) / (e + v) ** 2
  40         if len(p) > 14 and p[14] > 0:
  41             dfv = muscle_force_velocity_degroote(v)[1]
  42         dfp = (
  43             PASSIVE_LENGTH_SCALE * np.exp(PASSIVE_LENGTH_EXPONENT * (l - 1))
  44             if len(p) > 13
  45             and p[13] > 0
  46             and 1 < l < 1 + PASSIVE_EXPONENT_CAP / PASSIVE_LENGTH_EXPONENT
  47             else 0.0
  48         )
  49     else:
  50         low, high = p[4], p[5]
  51         a = (low + 1) / 2
  52         b = (1 + high) / 2
  53         if l < low or l > high:
  54             dfl = 0.0
  55         elif l <= a:
  56             dfl = (l - low) / (a - low) ** 2
  57         elif l <= 1:
  58             dfl = (1 - l) / (1 - a) ** 2
  59         elif l <= b:
  60             dfl = -(l - 1) / (b - 1) ** 2
  61         else:
  62             dfl = -(high - l) / (high - b) ** 2
  63         if v <= -1:
  64             dfv = 0.0
  65         elif v <= 0:
  66             dfv = 2 * (v + 1)
  67         elif v <= p[8] - 1:
  68             dfv = 2 * (p[8] - 1 - v) / (p[8] - 1)
  69         else:
  70             dfv = 0.0
  71         if l <= 1:
  72             dfp = 0.0
  73         elif l <= b:
  74             dfp = p[7] * (l - 1) / (b - 1) ** 2
  75         else:
  76             dfp = p[7] / (b - 1)
  77     return fl, fv, fp, dfl, dfv, dfp
