  12 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  13 @njit(cache=False)
  14 def sum_paw_contact_forces_and_moments(
  15     position,
  16     radius,
  17     side,
  18     body,
  19     body_com,
  20     paw_position,
  21     paw_velocity,
  22     bristle,
  23     dt,
  24     parameters,
  25     viscoelastic_slider=False,
  26 ):
  27     force_sum = np.zeros((4, 3))
  28     cop = np.zeros((4, 3))
  29     weight = np.zeros(4)
  30     applied = np.zeros((len(body_com), 6))
  31     depth_max = 0.0
  32     energy = 0.0
  33     power = 0.0
  34     for n in range(len(position)):
  35         s = side[n]
  36         b = body[n]
  37         depth = max(0.0, radius[n] - position[n, 2])
  38         point = position[n].copy()
  39         point[2] = 0.0
  40         velocity = paw_velocity[s, 3:] + np.cross(
  41             paw_velocity[s, :3],
  42             point - paw_position[s],
  43         )
  44         p = parameters[n] if parameters.ndim == 2 else parameters
  45         if viscoelastic_slider:
  46             f, displacement, stored = viscoelastic_paw_pad(
  47                 depth,
  48                 velocity,
  49                 bristle[n],
  50                 dt,
  51                 p,
  52             )
  53         else:
  54             f, displacement, stored = pad_force_law(depth, velocity, bristle[n], dt, p)
  55         bristle[n] = displacement
  56         applied[b, :3] += f
  57         applied[b, 3:] += np.cross(point - body_com[b], f)
  58         force_sum[s] += f
  59         cop[s] += f[2] * point
  60         weight[s] += f[2]
  61         depth_max = max(depth_max, depth)
  62         energy += stored
  63         power += np.dot(f, velocity)
  64     for s in range(4):
  65         cop[s] = cop[s] / weight[s] if weight[s] > 1e-12 else paw_position[s]
  66     return force_sum, cop, applied, depth_max, energy, power
