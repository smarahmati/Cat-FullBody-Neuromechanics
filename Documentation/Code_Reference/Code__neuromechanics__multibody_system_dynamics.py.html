   6 def multibody_system_dynamics(state, muscle_excitation):
   7     """All four limbs share one mass matrix and one root; no imposed motion.
   8 
   9     Native muscle actuators use the elastic tendon forces computed on the
  10     preceding step. Paw wrenches are physical contact forces. This partitioned
  11     update is why time-step convergence must be checked.
  12     """
  13     model, data = state["model"], state["data"]
  14     data.ctrl[:] = muscle_excitation
  15     data.qfrc_applied[:] = 0.0
  16     data.xfrc_applied[:] = state["contact_wrench"]
  17     # Integrate the common skeleton and native activation by one mechanical timestep.
  18     mj.mj_step(model, data)
