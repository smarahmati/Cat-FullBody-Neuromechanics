  12 def load_model_configuration(folder):
  13     """Read all named physical parameter files once, before t=0.
  14 
  15     Excitation is read afresh on every run. An edited table therefore changes
  16     actual motor input, not just its plotted label. References remain scoring
  17     data and do not prescribe the motion.
  18     """
  19     folder = (ROOT / Path(folder)).resolve()
  20     cfg = load_parameter_file(folder / "simulation_settings.py")
  21     # Physical overrides belong in the named files. Accepting a legacy
  22     # multiplier here would be misleading because absolute values win later.
  23     allowed = {
  24         "model_variant",
  25         "period_s",
  26         "dt_s",
  27         "sample_s",
  28         "seconds",
  29         "stop_on_fall",
  30         "activation_parameter_mode",
  31         "initialize_from_assembled_reference",
  32         "forward_step_implementation",
  33         "reference_table_dir",
  34         "fit_angle_scale_deg",
  35         "fit_ap_scale_N",
  36         "fit_vertical_scale_N",
  37         "emg_phase_threshold",
  38         "emg_phase_iou_min",
  39         "emg_phase_edge_tolerance_s",
  40     }
  41     if set(cfg) - allowed:
  42         # Stop and explain the invalid model setting or failed calculation.
  43         raise ValueError("Unknown run settings: " + str(sorted(set(cfg) - allowed)))
  44     if any(cfg[key] <= 0 for key in ["dt_s", "sample_s", "period_s", "seconds"]):
  45         # Stop and explain the invalid model setting or failed calculation.
  46         raise ValueError("Times and period must be positive")
  47     cfg["parameter_bundle_dir"] = str(folder)
  48     cfg["initial_parameter_file"] = str(folder / "initial_state.py")
  49     cfg["excitation_table_file"] = str((folder / "excitation_cycle.tsv") if (folder / "excitation_cycle.tsv").exists() else (ROOT / "Data/Muscle_Excitation_Input/excitation_cycle.tsv"))
  50     # Keep the exact phase samples; interpolation is performed at runtime.
  51     # Validation against muscle names is done after loading the model.
  52     read_time_series(cfg["excitation_table_file"])
  53     return cfg
