   6 def muscle_tendon_dynamics(state):
   7     """Solve fiber/tendon equilibrium with backward-Euler fiber velocity.
   8
   9     Whole MT path length comes from current bones and attachment sites.
  10     Fiber length differs from MT length because the tendon can stretch.
  11     Forces returned here, in N, act on the skeleton on the next time step.
  12     """
  13     model, data = state["model"], state["data"]
  14     # Solve every muscle's fiber length, fiber speed, tension, residual and bound flag.
  15     length, velocity, force, residual, bound = solve_elastic_muscle_equilibrium(
  16         # Whole muscle-tendon length: routed path plus its fixed length offset (m).
  17         data.actuator_length + state["mtu_length_offset"],
  18         # Fiber length at the preceding time step (m).
  19         state["fiber_length"],
  20         # Current muscle activation, between zero and one.
  21         data.act,
  22         # Optimal fiber lengths used to normalize fiber extension (m).
  23         state["lopt"],
  24         # Unloaded series-tendon lengths (m).
  25         state["slack"],
  26         # Muscle strengths, pennation and force-curve parameters from configuration.
  27         state["muscle_params"],
  28         # Tendon strain when tension equals maximum isometric muscle force along the tendon direction.
  29         state["e0"],
  30         # Dimensionless damping of fiber motion.
  31         state["fiber_damping_coefficients"],
  32         # Elapsed mechanical step, in seconds, for backward-Euler fiber velocity.
  33         model.opt.timestep,
  34     )
  35     # Save the newly solved muscle state for output and the next mechanical step.
  36     state.update(
  37         fiber_length=length,
  38         fiber_velocity=velocity,
  39         force=force,
  40         # Remaining tendon-minus-muscle force imbalance (N).
  41         residual=residual,
  42         # Flag any fiber that reached a numerical/geometric length bound.
  43         fiber_bound=bound,
  44     )
  45     # Native length increases away from the origin; tensile force is negative
  46     # in this actuator convention. Maximum force is not multiplied twice.
  47     model.actuator_biasprm[:, 0] = -force
