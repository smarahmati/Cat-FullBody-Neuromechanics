   1 """Read back the actual physical values used by the engine and elastic model."""
   2 
   3 import numpy as np
   4 
   5 import mujoco as mj
   6 
   7 from model_configuration.parameter_schema import MUSCLE_FIELDS, BODY_FIELDS, JOINT_FIELDS, CONTACT_FIELDS
   8 
   9 
  10 def read_applied_model_parameters(state):
  11     """Return ordinary dictionaries for config export and per-run auditing."""
  12     s = state
  13     m = s["model"]
  14     d = s["data"]
  15     value = lambda x: np.asarray(x).tolist()
  16     muscles = {}
  17     for i, name in enumerate(s["names"]):
  18         muscles[str(name)] = {
  19             field: value(
  20                 s[array][i]
  21                 if col is None
  22                 else s[array][i, col] if col < s[array].shape[1] else 0.0
  23             )
  24             for field, (array, col, unit, text) in MUSCLE_FIELDS.items()
  25         }
  26     activation = {
  27         str(n): dict(
  28             activation_time_constant_s=float(m.actuator_dynprm[i, 0]),
  29             deactivation_time_constant_s=float(m.actuator_dynprm[i, 1]),
  30             switch_smoothing_width=float(m.actuator_dynprm[i, 2]),
  31         )
  32         for i, n in enumerate(s["names"])
  33     }
  34     bodies = {
  35         m.body(i).name: {f: value(getattr(m, a)[i]) for f, (a, u, t) in BODY_FIELDS.items()}
  36         for i in range(1, m.nbody)
  37     }
  38     joints = {}
  39     for i in range(m.njnt):
  40         joints[m.joint(i).name] = {
  41             f: value(
  42                 getattr(m, a)[
  43                     (
  44                         int(m.jnt_dofadr[i])
  45                         if a.startswith("dof_")
  46                         else int(m.jnt_qposadr[i]) if a == "qpos_spring" else i
  47                     )
  48                 ]
  49             )
  50             for f, (a, u, t) in JOINT_FIELDS.items()
  51         }
  52     contact = {}
  53     for row, i in enumerate(s["pad_ids"]):
  54         contact[m.geom(int(i)).name] = dict(
  55             radius_m=float(m.geom_size[i, 0]),
  56             position_in_body_m=value(m.geom_pos[i]),
  57             **{
  58                 f: float(s["contact_parameters"][row, c])
  59                 for f, (c, u, t) in CONTACT_FIELDS.items()
  60             }
  61         )
  62     dynamic_sites = {int(row[5]) for row in s["routing_meta"]}
  63     attachments = {
  64         m.site(i).name: dict(
  65             position_in_body_m=value(m.site_pos[i]),
  66             orientation_in_body_wxyz=value(m.site_quat[i]),
  67         )
  68         for i in range(m.nsite)
  69         if i not in dynamic_sites
  70     }
  71     route_fields = [
  72         "origin_radius_m",
  73         "insertion_radius_m",
  74         "origin_angle_rad",
  75         "insertion_angle_rad",
  76         "proximal_wrap_radius_m",
  77         "proximal_wrap_sign",
  78         "proximal_joint_sign",
  79         "distal_wrap_radius_m",
  80         "distal_wrap_sign",
  81         "distal_joint_sign",
  82         "joint_spacing_m",
  83     ]
  84     routing = {
  85         m.site(int(meta[5])).name: dict(zip(route_fields, value(s["routing_geom"][row])))
  86         for row, meta in enumerate(s["routing_meta"])
  87     }
  88     wrap_ids = sorted(
  89         {
  90             int(m.wrap_objid[k])
  91             for k in range(m.nwrap)
  92             if int(m.wrap_type[k])
  93             in (int(mj.mjtWrap.mjWRAP_CYLINDER), int(mj.mjtWrap.mjWRAP_SPHERE))
  94         }
  95     )
  96     wrapping = {
  97         m.geom(i).name: dict(
  98             position_in_body_m=value(m.geom_pos[i]),
  99             orientation_in_body_wxyz=value(m.geom_quat[i]),
 100             radius_m=float(m.geom_size[i, 0]),
 101         )
 102         for i in wrap_ids
 103     }
 104     initial = dict(
 105         coordinates={
 106             m.joint(i).name: dict(
 107                 position=float(d.qpos[m.jnt_qposadr[i]]),
 108                 velocity=float(d.qvel[m.jnt_dofadr[i]]),
 109             )
 110             for i in range(m.njnt)
 111         },
 112         muscles={
 113             str(n): dict(
 114                 activation_0_to_1=float(d.act[i]),
 115                 fiber_length_m=float(s["fiber_length"][i]),
 116             )
 117             for i, n in enumerate(s["names"])
 118         },
 119         contact_memory={
 120             m.geom(int(i)).name: value(s["bristle"][row])
 121             for row, i in enumerate(s["pad_ids"])
 122         },
 123     )
 124     environment = dict(
 125         gravity_m_s2=value(m.opt.gravity),
 126         solver_iterations=int(m.opt.iterations),
 127         solver_tolerance=float(m.opt.tolerance),
 128         # Read back the actual integration scheme used by the engine.
 129         integrator=int(m.opt.integrator),
 130     )
 131     return dict(
 132         muscle_tendon=muscles,
 133         activation=activation,
 134         bodies=bodies,
 135         joints=joints,
 136         contact=contact,
 137         attachments=attachments,
 138         routing=routing,
 139         wrapping=wrapping,
 140         environment=environment,
 141         initial_state=initial,
 142     )
