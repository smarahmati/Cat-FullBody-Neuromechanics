  20 def plot_emg_comparison(result, settings, folder):
  21     t = result["time"]
  22     T = settings["period_s"]
  23     target = experimental_comparison_targets(t, T, settings.get("reference_table_dir"))
  24     full = experimental_comparison_targets(
  25         np.linspace(0, T, 801),
  26         T,
  27         settings.get("reference_table_dir"),
  28     )["excitation"]
  29     provenance = json.loads((ROOT / "Data/Experimental_Reference_Data/INPUT_PROVENANCE.json").read_text())[
  30         "channels"
  31     ]
  32     names = list(result["names"])
  33     u = result["excitation"]
  34     table = configured_excitation_samples(settings, u)
  35     interval = "sequence" if settings.get("excitation_sequence_file") else "cycle"
  36     fhl = load_reference_tables(settings.get("reference_table_dir"))[2]
  37     files = []
  38     for side, prefix in enumerate(["L", "R", "FL", "FR"]):
  39         curves = []
  40         for index, item in enumerate(provenance):
  41             if "EMG" not in item["kind"] or item["kind"].startswith("pooled_fore") != (
  42                 side
  43                 >= 2
  44             ):
  45                 # Skip the remaining work for this item.
  46                 continue
  47             name = prefix + "_" + item["channel"]
  48             if name not in names:
  49                 # Skip the remaining work for this item.
  50                 continue
  51             column = (
  52                 (22 if side == 2 else 31) + index - 11
  53                 if side >= 2
  54                 else (0 if side == 0 else 11) + index
  55             )
  56             curves.append(
  57                 (
  58                     name,
  59                     target["excitation"][:, column]
  60                     / max(1e-12, float(full[:, column].max())),
  61                 )
  62             )
  63         if side < 2:
  64             curves.append(
  65                 (
  66                     prefix + "_FHL",
  67                     np.array(
  68                         [periodic_interpolate(fhl, x / T + 0.5 + [0.5, 0][side]) for x in t]
  69                     ),
  70                 )
  71             )
  72         fig, axes = plt.subplots(2, 4, figsize=(13, 6), sharex=True, sharey=True)
  73         for ax, (name, ref) in zip(axes.ravel(), curves):
  74             i = names.index(name)
  75             peak = max(0.02, float(table[:, i].max()))
  76             ax.plot(t, ref, "k--", label="EMG shape reference")
  77             ax.plot(
  78                 t,
  79                 u[:, i] / peak,
  80                 color="#087f9c",
  81                 label="Excitation / " + interval + " peak",
  82             )
  83             ax.set_title(f"{name}\nexcitation peak {peak:.3f}", fontsize=10)
  84             ax.grid(alpha=0.2)
  85             ax.set_ylim(-0.05, 1.1)
  86         for ax in axes[-1]:
  87             ax.set_xlabel("Time (s)")
  88         for ax in axes[:, 0]:
  89             ax.set_ylabel("Normalized envelope")
  90         axes[0, 0].legend(fontsize=8)
  91         label = [
  92             "LH: assumed mirror of RH EMG",
  93             "RH: measured EMG envelopes",
  94             "LF: pooled forelimb EMGs",
  95             "RF: pooled forelimb EMGs",
  96         ][side]
  97         fig.suptitle(
  98             label
  99             + "; fixed input clock, no phase shift to improve the fit\nNormalization uses the complete input "
 100             + interval
 101             + "; EMG does not identify absolute muscle_excitation amplitude.",
 102             fontsize=11,
 103         )
 104         fig.tight_layout()
 105         filename = f"emg_{prefix}.png"
 106         # Save the computed values or figure for inspection and reproducibility.
 107         fig.savefig(folder / filename, dpi=140)
 108         plt.close(fig)
 109         files.append(filename)
 110     return files
