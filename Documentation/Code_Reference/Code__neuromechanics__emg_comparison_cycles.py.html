  16 def emg_comparison_cycles(names, settings, samples=2048):
  17     """Return measured/pooled/assumed channels only; no inferred EMG targets."""
  18     names = list(names)
  19     phase = np.arange(samples) / samples
  20     T = settings["period_s"]
  21     target = experimental_comparison_targets(
  22         phase * T,
  23         T,
  24         settings.get("reference_table_dir"),
  25     )["excitation"]
  26     if settings.get("excitation_table_file"):
  27         from .read_time_series import read_time_series
  28 
  29         table = read_time_series(ROOT / settings["excitation_table_file"], names)[2]
  30     elif "excitation_control_points" in settings:
  31         table = np.asarray(settings["excitation_control_points"])
  32     elif settings.get("excitation_file"):
  33         table = np.load(ROOT / settings["excitation_file"])["excitation_table"]
  34     else:
  35         folder = (
  36             ROOT / "Data/Skeletal_Geometry/Model"
  37             if settings.get("model_variant", "base") == "base"
  38             else ROOT / "Data/Skeletal_Geometry/Model" / settings["model_variant"]
  39         )
  40         table = np.load(folder / "muscle_parameters.npz")["excitation_table"]
  41     provenance = json.loads((ROOT / "Data/Experimental_Reference_Data/INPUT_PROVENANCE.json").read_text())[
  42         "channels"
  43     ]
  44     pairs = []
  45     for side, prefix in enumerate(["L", "R", "FL", "FR"]):
  46         for item in provenance:
  47             if "EMG" not in item["kind"]:
  48                 # Skip the remaining work for this item.
  49                 continue
  50             fore = item["kind"].startswith("pooled_fore")
  51             if fore != (side >= 2):
  52                 # Skip the remaining work for this item.
  53                 continue
  54             name = prefix + "_" + item["channel"]
  55             if name not in names:
  56                 # Skip the remaining work for this item.
  57                 continue
  58             column = (
  59                 (
  60                     (22 if side == 2 else 31)
  61                     + [c["channel"] for c in provenance[11:]].index(item["channel"])
  62                 )
  63                 if fore
  64                 else (
  65                     (0 if side == 0 else 11)
  66                     + [c["channel"] for c in provenance[:11]].index(item["channel"])
  67                 )
  68             )
  69             i = names.index(name)
  70             pred = np.interp(phase, np.linspace(0, 1, len(table)), table[:, i]) / max(
  71                 0.02,
  72                 float(table[:, i].max()),
  73             )
  74             ref = target[:, column] / max(1e-12, float(target[:, column].max()))
  75             pairs.append(
  76                 dict(
  77                     channel=name,
  78                     limb=prefix,
  79                     kind=item["kind"] if side else "assumed_LH_from_recorded_RH",
  80                     excitation=pred,
  81                     reference=ref,
  82                 )
  83             )
  84     fhl = load_reference_tables(settings.get("reference_table_dir"))[2]
  85     for side, prefix in enumerate(["L", "R"]):
  86         name = prefix + "_FHL"
  87         if name not in names:
  88             # Skip the remaining work for this item.
  89             continue
  90         i = names.index(name)
  91         pred = np.interp(phase, np.linspace(0, 1, len(table)), table[:, i]) / max(
  92             0.02,
  93             float(table[:, i].max()),
  94         )
  95         ref = np.array([periodic_interpolate(fhl, p + 0.5 + [0.5, 0][side]) for p in phase])
  96         ref /= max(1e-12, float(ref.max()))
  97         pairs.append(
  98             dict(
  99                 channel=name,
 100                 limb=prefix,
 101                 kind="recorded_RH_EMG" if side else "assumed_LH_from_recorded_RH",
 102                 excitation=pred,
 103                 reference=ref,
 104             )
 105         )
 106     return pairs
