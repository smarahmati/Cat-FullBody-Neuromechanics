   1 """Read signed muscle moment arms from MuJoCo's current transmission Jacobian.
   2 
   3 Input: the model and data after muscle paths and positions have been updated.
   4 Output: an (muscle, generalized-coordinate) array equal to -dL_MT/dq.
   5 Rotational columns are metres; translational columns are dimensionless.
   6 Positive tension times this array gives the applied generalized muscle force.
   7 """
   8 import numpy as np
   9 
  10 
  11 def muscle_moment_arms(model, data):
  12     """Convert the engine's sparse transmission representation to a named matrix."""
  13     jacobian = np.zeros((model.nu, model.nv))
  14     if data.actuator_moment.ndim == 2:
  15         jacobian[:] = data.actuator_moment
  16     else:
  17         for muscle in range(model.nu):
  18             first = int(data.moment_rowadr[muscle])
  19             last = first + int(data.moment_rownnz[muscle])
  20             columns = data.moment_colind[first:last]
  21             jacobian[muscle, columns] = data.actuator_moment[first:last]
  22     # A tensile actuator pulls towards decreasing muscle-tendon length.
  23     return -jacobian
