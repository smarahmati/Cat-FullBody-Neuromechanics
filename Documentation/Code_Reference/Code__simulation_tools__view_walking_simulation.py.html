   1 """Inspect an already computed forward run; never solve or alter its motion.

 109     def set_time(value):
 110         """Pause at any recorded time; this works before or after the trial ends."""
 111         # Convert text or button input into seconds.
 112         requested_time = float(value)
 113         # Reject invalid numbers before they can reach the playback worker.
 114         if not np.isfinite(requested_time):
 115             # Explain how to correct an invalid Go-to-time entry.
 116             raise ValueError("Enter a finite time in seconds.")
 117         # Keep the requested position within the saved trial.
 118         requested_time = float(np.clip(requested_time, 0.0, times[-1]))
 119         # Hold the chosen frame so the user can inspect its curves and pose.
 120         paused.set(True)
 121         # Move the full-trial timeline to the same recorded instant.
 122         position.set(requested_time)
 123         if worker is not None:
 124             # Seeking changes playback only; it does not calculate a new trial.
 125             worker.command(paused=True, seek=requested_time)
 126 
 127     def slider_seek(value):
 128         if not updating_slider and worker is not None:
 129             # Pause and inspect the frame selected anywhere on the full timeline.
 130             set_time(value)
 131 
 132     # Mark the beginning of mouse dragging on the full-trial timeline.
 133     def start_timeline_drag(event):
 134         # Update the flag shared with the periodic refresh callback.
 135         nonlocal dragging_slider
 136         # Keep automatic playback updates from moving the thumb under the mouse.
 137         dragging_slider = True
 138         # Hold motion while browsing the recording.
 139         paused.set(True)
 140 
 141     # Finish a timeline drag without losing the author's selected time.
 142     def finish_timeline_drag(event):
 143         # Update the flag shared with the periodic refresh callback.
 144         nonlocal dragging_slider
 145         # Send the final thumb position to the 3D viewer and plots.
 146         set_time(position.get())
 147         # Allow subsequent programmatic timeline updates again.
 148         dragging_slider = False

 150     def reset_playback():
 151         """Return to the archived initial state without changing model parameters."""
 152         paused.set(True)
 153         if worker is not None:
 154             worker.command(paused=True, seek=float(times[0]))
 155         position.set(float(times[0]))

 349     def close():
 350         # Use the surrounding GUI or module state rather than creating a new local variable.
 351         nonlocal closed
 352         closed = True
 353         worker.stop()
 354         viewer.close()
 355         root.destroy()
 356 
 357     def refresh():
 358         # Use the surrounding GUI or module state rather than creating a new local variable.
 359         nonlocal last_status, updating_slider
 360         if closed:
 361             return
 362         if not viewer.is_running():
 363             close()
 364             return
 365         started = time.perf_counter()
 366         snapshot = worker.snapshot()
 367         current = snapshot["time"]
 368         if snapshot["error"]:
 369             paused.set(True)
 370             status.set("Viewer error: " + snapshot["error"])
 371             return
 372         if current >= times[-1]:
 373             paused.set(True)
 374         worker.command(
 375             paused=paused.get(),
 376             speed=float(speed.get()),
 377             fps=float(fps.get()),
 378             forces=forces.get(),
 379             rich=rich.get(),
 380             follow=follow.get(),
 381         )
 382         side = ["LH", "RH", "LF", "RF"].index(limb.get())
 383         joints = (
 384             ["Hip", "Knee", "Ankle", "MTP"]
 385             if side < 2
 386             else ["Shoulder", "Elbow", "Wrist", "MCP"]
 387         )
 388         panel.update(
 389             current,
 390             side,
 391             muscle.get(),
 392             joints.index(joint.get()),
 393             norm.get(),
 394             emg.get(),
 395         )
 396         if started - last_status > 0.2:
 397             last_status = started
 398             updating_slider = True
 399             # Preserve the thumb position while the user drags to another time.
 400             if not dragging_slider:
 401                 # Keep the timeline synchronized with ordinary playback.
 402                 position.set(current)
 403             updating_slider = False
 404             status.set(
 405                 f'{"PAUSED" if paused.get() else "PLAYING"} | {current:.2f} / {times[-1]:.2f} s | {speed.get()}x | 3D updates: {snapshot["update_hz"]:.1f}/s (target {fps.get()}) | plots: {panel.measured_hz:.1f}/s'
 406             )
 407             force_text.set(
 408                 "Paw Ground_Reaction_Forces | "
 409                 + "   ".join(
 410                     f"{label}: Fx {force[0]:+.1f}, Fz {force[2]:.1f} N"
 411                     for label, force in zip(
 412                         ["LH cyan", "RH pink", "LF gold", "RF purple"],
 413                         snapshot["forces"],
 414                     )
 415                 )
 416             )
 417         # Compensate for plot-update work instead of adding a fixed delay to it.
 418         # 20 plot updates/s are sufficient for the traces and leave more CPU
 419         # time for the independent 60-Hz 3D visualization and window rendering.
 420         root.after(
 421             max(1, int(1000 * (1 / 20.0 - (time.perf_counter() - started)))),
 422             refresh,
 423         )
 424

 426     root.after(100, refresh)

 428         "GUI_READY: independent 60-Hz-target playback, paw-force arrows and blitted comparison plots open",

 430     )
