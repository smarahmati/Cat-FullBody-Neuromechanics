   1 """Check website publication rules and dual local/online navigation.
   2 
   3 These tests use only the standard library. They do not run the simulation.
   4 """
   5 from pathlib import Path
   6 import importlib.util
   7 import unittest
   8 
   9 ROOT = Path(__file__).resolve().parents[1]
  10 spec = importlib.util.spec_from_file_location('website', ROOT / 'Code/simulation_tools/build_documentation_website.py')
  11 website = importlib.util.module_from_spec(spec)
  12 spec.loader.exec_module(website)
  13 
  14 
  15 class WebsitePublicationVerification(unittest.TestCase):
  16     def test_private_files_never_enter_artifact(self):
  17         for name in ['.git/config', '.venv/pyvenv.cfg',
  18                      'Documentation/References/Local_Papers/private.pdf',
  19                      'Data/Skeletal_Geometry/Model/meshes/bone.obj']:
  20             self.assertFalse(website.belongs_in_website(name))
  21 
  22     def test_raw_results_stay_available_without_website_duplication(self):
  23         raw = 'Simulation_Results/Intact_Walking/forward_simulation_timeseries.npz'
  24         self.assertFalse(website.belongs_in_website(raw))
  25         self.assertTrue(website.belongs_in_website('Simulation_Results/Intact_Walking/simulation_report.html'))
  26         link = website.resolve_link('Simulation_Results/index.html', 'Intact_Walking/forward_simulation_timeseries.npz',
  27                                     set(), {raw}, 'owner/repo', 'commit123')
  28         self.assertEqual(link, 'https://github.com/owner/repo/blob/commit123/' + raw)
  29 
  30     def test_relative_navigation_keeps_project_prefix(self):
  31         link = website.resolve_link('Documentation/model_methods.html', '../START_HERE.html',
  32                                     {'START_HERE.html'}, {'START_HERE.html'}, 'owner/repo', 'main')
  33         self.assertEqual(link, '../START_HERE.html')
  34         with self.assertRaises(ValueError):
  35             website.local_target('Documentation/model_methods.html', '/START_HERE.html')
  36 
  37     def test_missing_or_ignored_download_fails_instead_of_publishing(self):
  38         with self.assertRaises(ValueError):
  39             website.resolve_link('Documentation/index.html', 'secret.pdf', set(), set(), 'owner/repo', 'main')
  40 
  41     def test_inline_script_and_code_listing_are_preserved(self):
  42         source = '<script>const x="<a href=\"not-a-link\">";</script><pre>&lt;a href="also-not-a-link"&gt;</pre><a href="../START_HERE.html">Home</a>'
  43         output = website.prepare_html('Documentation/index.html', source,
  44                                       {'START_HERE.html'}, {'START_HERE.html'}, 'owner/repo', 'main')
  45         self.assertEqual(source, output)
  46 
  47     def test_directory_link_is_portable(self):
  48         self.assertEqual(website.resolve_link('Documentation/index.html', '../Data/',
  49                          {'Data/index.html'}, {'Data/index.html'}, 'owner/repo', 'main'), '../Data/index.html')
  50 
  51 
  52 if __name__ == '__main__':
  53     unittest.main()
