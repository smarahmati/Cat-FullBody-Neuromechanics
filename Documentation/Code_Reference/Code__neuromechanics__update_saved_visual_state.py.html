   8 def update_saved_visual_state(state, sample):
   9     """Avoid collision/constraint/muscle solves during a visualization replay.
  10 
  11     This is not a replacement for advance_partitioned_timestep. The archived forward dynamics
  12     already computed the motion and Ground_Reaction_Forces. These calls place bones, sites and
  13     wrapped tendon paths at the saved/interpolated pose for rendering only.
  14     """
  15     m, d = state["model"], state["data"]
  16     if m.nq != m.nv:
  17         # Stop and explain the invalid model setting or failed calculation.
  18         raise ValueError("This display interpolator requires scalar joint coordinates")
  19     d.qpos[:] = sample["qpos"]
  20     d.qvel[:] = sample["qvel"]
  21     d.act[:] = sample["activation"]
  22     d.ctrl[:] = sample["excitation"]
  23     d.time = sample["time"]
  24     update_muscle_paths(state)
  25     mj.mj_kinematics(m, d)
  26     mj.mj_comPos(m, d)
  27     mj.mj_tendon(m, d)
