   1 """Read the intact muscle excitation table using simulation time.
   2 
   3 Input: time in seconds and a table with values between 0 and 1. Output: one
   4 excitation per muscle-tendon path. Joint angles and forces do not alter it.
   5 """
   6 
   7 import numpy as np
   8 
   9 from .periodic_interpolate import periodic_interpolate
  10 
  11 
  12 
  13 def muscle_excitation(state, time_s):
  14     """Calculate every muscle's prescribed excitation at this time."""
  15     # Fixed finite sequences use their specified time index.
  16     if "excitation_sequence" in state or "offline_cycle_input" in state:
  17         from .fixed_sequence_excitation import fixed_sequence_excitation
  18 
  19         original_excitation = fixed_sequence_excitation(state, time_s)
  20     else:
  21         # Find how far time has progressed through the input cycle.
  22         cycle_fraction = time_s / state["settings"]["period_s"]
  23         # Interpolate neighboring rows of the repeated excitation table.
  24         original_excitation = periodic_interpolate(
  25             state["excitation_table"],
  26             cycle_fraction,
  27         )
  28         # Keep excitation inside the defined 0-to-1 range.
  29         original_excitation = np.clip(original_excitation, 0.0, 1.0)
  30     # The prescribed input is independent of the resulting motion and forces.
  31     return original_excitation
