  22 class SavedReplayWorker:
  23     """Own MuJoCo data/viewer mutations; exchange plain values with the Tk thread."""
  24 
  25     def __init__(self, viewer, state, result, profile_path, profile_seconds=12.0):
  26         self.viewer, self.state, self.result = viewer, state, result
  27         self.profile_path, self.profile_seconds = profile_path, profile_seconds
  28         self.lock = threading.Lock()
  29         self.stop_event = threading.Event()
  30         self.controls = dict(
  31             paused=True,
  32             speed=0.5,
  33             fps=60.0,
  34             forces=True,
  35             rich=False,
  36             follow=True,
  37             seek=None,
  38         )
  39         self.current = dict(
  40             time=0.0,
  41             update_hz=0.0,
  42             # Calculate the error in this comparison.
  43             error=None,
  44             forces=np.zeros((4, 3)),
  45             arrows=[],
  46         )
  47         self.intervals = []
  48         self.work_seconds = []
  49         self.profile_saved = False
  50         self.thread = threading.Thread(
  51             target=self._run,
  52             name="cat-saved-3d-replay",
  53             daemon=True,
  54         )
  55 
  56     def start(self):
  57         self.thread.start()
  58 
  59     def command(self, **values):
  60         # Keep this file or shared state safely scoped to the following operations.
  61         with self.lock:
  62             self.controls.update(values)
  63 
  64     def snapshot(self):
  65         # Keep this file or shared state safely scoped to the following operations.
  66         with self.lock:
  67             return dict(self.current)
  68 
  69     def stop(self):
  70         self.stop_event.set()
  71         self.thread.join(timeout=3.0)
  72 
  73     def _run(self):
  74         # Attempt the operation and report a failure in the exception branch.
  75         try:
  76             self._loop()
  77         # Handle the stated error without hiding its cause.
  78         except Exception as error:
  79             import traceback
  80 
  81             traceback.print_exc()
  82             # Keep this file or shared state safely scoped to the following operations.
  83             with self.lock:
  84                 self.current["error"] = str(error)
  85 
  86     def _loop(self):
  87         frame_times = deque(maxlen=180)
  88         beginning = previous = time.perf_counter()
  89         simulation_time = 0.0
  90         pelvis_id = self.state["model"].body("pelvis").id
  91         # Repeat while not self.stop_event.is_set() and self.viewer.is_running().
  92         while not self.stop_event.is_set() and self.viewer.is_running():
  93             started = time.perf_counter()
  94             elapsed = started - previous
  95             previous = started
  96             # Keep this file or shared state safely scoped to the following operations.
  97             with self.lock:
  98                 control = dict(self.controls)
  99                 self.controls["seek"] = None
 100             if control["seek"] is not None:
 101                 simulation_time = float(
 102                     np.clip(control["seek"], 0.0, self.result["time"][-1])
 103                 )
 104             elif not control["paused"]:
 105                 simulation_time = min(
 106                     float(self.result["time"][-1]),
 107                     simulation_time + elapsed * control["speed"],
 108                 )
 109             sample = interpolate_simulation_for_display(self.result, simulation_time)
 110             # Keep this file or shared state safely scoped to the following operations.
 111             with self.viewer.lock():
 112                 update_saved_visual_state(self.state, sample)
 113                 if control["follow"]:
 114                     self.viewer.cam.lookat[0] = self.state["data"].xpos[pelvis_id, 0] + 0.12
 115                 self.viewer.user_scn.flags[mj.mjtRndFlag.mjRND_SHADOW] = control["rich"]
 116                 self.viewer.user_scn.flags[mj.mjtRndFlag.mjRND_REFLECTION] = control["rich"]
 117                 arrows = draw_saved_grf_vectors(
 118                     self.viewer,
 119                     self.state,
 120                     sample["grf"],
 121                     control["forces"],
 122                 )
 123             self.viewer.sync()
 124             finished = time.perf_counter()
 125             frame_times.append(finished)
 126             if len(frame_times) > 1:
 127                 rate = (len(frame_times) - 1) / (frame_times[-1] - frame_times[0])
 128             else:
 129                 rate = 0.0
 130             if started - beginning > 2.0 and len(frame_times) > 1:
 131                 self.intervals.append(frame_times[-1] - frame_times[-2])
 132                 self.work_seconds.append(finished - started)
 133             # Keep this file or shared state safely scoped to the following operations.
 134             with self.lock:
 135                 self.current.update(
 136                     time=simulation_time,
 137                     update_hz=rate,
 138                     forces=sample["grf"].copy(),
 139                     arrows=arrows,
 140                 )
 141                 if simulation_time >= self.result["time"][-1]:
 142                     self.controls["paused"] = True
 143             if (
 144                 not self.profile_saved
 145                 and started - beginning >= self.profile_seconds
 146                 and self.intervals
 147             ):
 148                 report = dict(
 149                     target_update_hz=control["fps"],
 150                     measured_3d_update_hz=1.0 / float(np.mean(self.intervals)),
 151                     frame_interval_ms_median=1000 * float(np.median(self.intervals)),
 152                     frame_interval_ms_p95=1000 * float(np.quantile(self.intervals, 0.95)),
 153                     frame_work_ms_median=1000 * float(np.median(self.work_seconds)),
 154                     displayed_arrow_count=len(arrows),
 155                     mesh_triangles=int(self.state["model"].nmeshface),
 156                     shadows_and_reflections=control["rich"],
 157                     rate_scope="Python state + passive-viewer synchronization, not measured monitor/GPU presentation FPS",
 158                     recording_unchanged=True,
 159                     interpolation="Linear between saved scalar-coordinate states for display only",
 160                 )
 161                 self.profile_path.parent.mkdir(parents=True, exist_ok=True)
 162                 # Save the computed values or figure for inspection and reproducibility.
 163                 self.profile_path.write_text(json.dumps(report, indent=2))
 164                 print("GUI_PERFORMANCE " + json.dumps(report), flush=True)
 165                 self.profile_saved = True
 166             # Python 3.12 sleep uses a high-resolution Windows waitable timer.
 167             # Event.wait rounds short waits to the coarser system tick on this
 168             # host, which reduced a 60-Hz request to roughly 30 actual updates.
 169             time.sleep(max(0.0, 1.0 / control["fps"] - (time.perf_counter() - started)))
