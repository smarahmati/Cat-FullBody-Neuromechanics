   1 """Run the coupled forward model from editable files.
   2 
   3 Examples, from the project directory:
   4     python run.py --output my_trial
   5     python run.py --parameters Data/Model_Parameters --seconds 5 --output five_seconds
   6     python run.py --config Simulation_Results/Intact_Walking/settings.json --seconds 3
   7     python run.py --output trial_with_viewer --gui
   8 
   9 Without --config/--parameters, Data/Model_Parameters is used. Nothing is fitted
  10 automatically: each run reads the requested physical parameters and fixed
  11 muscle excitation table, integrates forward once, and compares with source references.
  12 """
  13 
  14 import os, json, argparse, hashlib, shutil, sys
  15 
  16 from pathlib import Path
  17 
  18 sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
  19 from datetime import datetime
  20 
  21 os.environ.setdefault("OPENBLAS_NUM_THREADS", "1")
  22 os.environ.setdefault("OMP_NUM_THREADS", "1")
  23 import numpy as np
  24 
  25 from model_configuration.paths import ROOT
  26 
  27 from neuromechanics.simulate_forward_dynamics import simulate_forward_dynamics
  28 
  29 from neuromechanics.initialize_neuromechanical_model import initialize_neuromechanical_model
  30 
  31 from neuromechanics.load_model_configuration import load_model_configuration
  32 
  33 from neuromechanics.read_applied_model_parameters import read_applied_model_parameters

  36 def main(default_seconds=None, default_gui=False):
  37     """Read settings, calculate motion, and save results and comparison plots.
  38 
  39     default_seconds: duration chosen in run.py; None uses the parameter file.
  40     default_gui: whether run.py asks to open the viewer after calculation.
  41     Explicit --seconds, --gui, or --no-gui arguments take precedence.
  42     """
  43     parser = argparse.ArgumentParser(description=__doc__)
  44     group = parser.add_mutually_exclusive_group()
  45     group.add_argument("--parameters", help="Folder of commented physical parameter files")
  46     group.add_argument("--config", help="Saved trial settings JSON")
  47     parser.add_argument(
  48         "--seconds",
  49         type=float,
  50         default=default_seconds,
  51         help="Run duration in seconds; overrides run.py and simulation_settings.py",
  52     )
  53     parser.add_argument(
  54         "--output",
  55         default="User_Simulations/walking_" + datetime.now().strftime("%Y%m%d_%H%M%S"),
  56     )
  57     parser.add_argument(
  58         "--replace-results", action="store_true",
  59         help="Recalculate and overwrite generated files in the named Results folder",
  60     )
  61     gui_options = parser.add_mutually_exclusive_group()
  62     gui_options.add_argument(
  63         "--gui",
  64         # Use the editable run.py choice when neither GUI flag is supplied.
  65         default=default_gui,
  66         action="store_true",
  67         help="After simulation, open the intact walking animation and plots",
  68     )
  69     gui_options.add_argument(
  70         "--no-gui",
  71         action="store_true",
  72         help="Calculate and save results without opening windows",
  73     )
  74     args = parser.parse_args()
  75     out = (ROOT / "Simulation_Results" / args.output).resolve()
  76     if not out.is_relative_to((ROOT / "Simulation_Results").resolve()):
  77         raise ValueError("Output must be within this project's Results folder")
  78     # Replacing an existing release is explicit; ordinary runs remain separate.
  79     out.mkdir(parents=True, exist_ok=args.replace_results)
  80     # Preserve the implementations and global curve constants as well as the
  81     # values. A parameter record alone cannot identify a changed physical law.
  82     for source_dir in ["Code/neuromechanics", "Code/model_configuration"]:
  83         target = out / "Simulation_Code_Used" / source_dir
  84         target.mkdir(parents=True, exist_ok=args.replace_results)
  85         for path in (ROOT / source_dir).glob("*.py"):
  86             shutil.copy2(path, target / path.name)
  87     shutil.copy2(ROOT / "run.py", out / "Simulation_Code_Used" / "run.py")
  88     shutil.copy2(
  89         ROOT / "Code/simulation_tools/run_forward_simulation.py",
  90         out / "Simulation_Code_Used" / "run_forward_simulation.py",
  91     )
  92     if args.config:
  93         cfg = json.loads((ROOT / args.config).read_text())
  94         # Save the computed values or figure for inspection and reproducibility.
  95         (out / "original_settings.json").write_text(json.dumps(cfg, indent=2))
  96     else:
  97         original = (ROOT / (args.parameters or "Data/Model_Parameters")).resolve()
  98         copied = out / "Model_Parameters_Used"
  99         shutil.copytree(original, copied, dirs_exist_ok=args.replace_results)
 100         if not (copied / "excitation_cycle.tsv").exists():
 101             shutil.copy2(ROOT / "Data/Muscle_Excitation_Input/excitation_cycle.tsv", copied / "excitation_cycle.tsv")
 102         cfg = load_model_configuration(copied)
 103         source_refs = (ROOT / cfg["reference_table_dir"]).resolve()
 104         shutil.copytree(source_refs, out / "Experimental_References_Used", dirs_exist_ok=args.replace_results)
 105         cfg["reference_table_dir"] = str(out / "Experimental_References_Used")
 106         files = list(copied.rglob("*")) + list((out / "Experimental_References_Used").rglob("*"))
 107         hashes = {
 108             str(p.relative_to(out)): hashlib.sha256(p.read_bytes()).hexdigest()
 109             for p in files
 110             if p.is_file()
 111         }
 112         # Save the computed values or figure for inspection and reproducibility.
 113         (out / "INPUT_HASHES.json").write_text(
 114             json.dumps(
 115                 dict(original_parameter_folder=str(original), files=hashes),
 116                 indent=2,
 117             )
 118         )
 119     if args.seconds is not None:
 120         cfg["seconds"] = args.seconds
 121     if cfg.get("seconds", 0) <= 0:
 122         # Stop and explain the invalid model setting or failed calculation.
 123         raise ValueError("Run duration must be positive")
 124     state = initialize_neuromechanical_model(cfg)
 125     applied = read_applied_model_parameters(state)
 126     from neuromechanics.write_attachment_map import write_attachment_map
 127 
 128     write_attachment_map(state, out)
 129     # Save the computed values or figure for inspection and reproducibility.
 130     (out / "APPLIED_PARAMETERS.json").write_text(json.dumps(applied, indent=2))
 131     if not args.config:
 132         from neuromechanics.write_parameter_tables import write_parameter_tables
 133 
 134         write_parameter_tables(
 135             applied,
 136             json.loads((copied / "PARAMETER_PROVENANCE.json").read_text()),
 137             out / "APPLIED_PARAMETERS.html",
 138         )
 139     # Save the computed values or figure for inspection and reproducibility.
 140     portable_settings = dict(state["settings"])
 141     for key in ["parameter_bundle_dir", "initial_parameter_file", "excitation_table_file", "reference_table_dir"]:
 142         if portable_settings.get(key):
 143             portable_settings[key] = Path(portable_settings[key]).relative_to(ROOT).as_posix()
 144     (out / "settings.json").write_text(json.dumps(portable_settings, indent=2))
 145     print("Applied parameters saved: " + str(out / "APPLIED_PARAMETERS.json"), flush=True)
 146     result, state = simulate_forward_dynamics(state=state)
 147     result["degroote_force_velocity_enabled"] = (
 148         state["muscle_params"][:, 14].astype(bool)
 149         if state["muscle_params"].shape[1] > 14
 150         else np.zeros(len(state["names"]), bool)
 151     )
 152     # Record constitutive choices with the trajectory, not only in the config.
 153     result["passive_fiber_enabled"] = state["muscle_params"][:, 13] > 0
 154     # Save the computed values or figure for inspection and reproducibility.
 155     np.savez_compressed(out / "forward_simulation_timeseries.npz", **result)
 156     summary = dict(
 157         duration_s=float(result["time"][-1]),
 158         requested_s=state["settings"]["seconds"],
 159         failure=str(result["failure"]),
 160         mass_kg=float(state["model"].body_mass.sum()),
 161         wall_s=float(result["wall_seconds"]),
 162     )
 163     # Save the computed values or figure for inspection and reproducibility.
 164     (out / "simulation_summary.json").write_text(json.dumps(summary, indent=2))
 165     from neuromechanics.write_simulation_results import write_simulation_results
 166 
 167     write_simulation_results(result, state["settings"], out)
 168     from neuromechanics.walking_acceptance import walking_acceptance
 169 
 170     # Save the computed values or figure for inspection and reproducibility.
 171     (out / "walking_quality_metrics.json").write_text(
 172         json.dumps(walking_acceptance(result, state["settings"]), indent=2)
 173     )
 174     from neuromechanics.export_biomechanics import export_biomechanics
 175     export_biomechanics(result, state, out)
 176     print(json.dumps(summary, indent=2))
 177     print("Report: " + str(out / "simulation_report.html"))
 178     # An explicit --no-gui always overrides SHOW_GUI = True in run.py.
 179     if args.gui and not args.no_gui:
 180         import subprocess, sys
 181 
 182         subprocess.run(
 183             [
 184                 sys.executable,
 185                 str(ROOT / "Code/simulation_tools/view_walking_simulation.py"),
 186                 "--run",
 187                 str(out),
 188             ],
 189             check=True,
 190         )
 191 
 192 
 193 if __name__ == "__main__":
 194     main()
