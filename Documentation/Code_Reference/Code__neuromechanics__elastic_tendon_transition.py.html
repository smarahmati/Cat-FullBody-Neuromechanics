  12 def elastic_tendon_transition(strain_at_max_force):
  13     k = TENDON_TOE_CURVATURE
  14     if k == 3.0:
  15         # Preserve the previous operation order at the original coefficient.
  16         return 99 * strain_at_max_force * np.exp(3.0) / (166 * np.exp(3.0) - 67)
  17     numerator = 0.33 * k * np.exp(k)
  18     return strain_at_max_force * numerator / (numerator + 0.67 * np.expm1(k))
