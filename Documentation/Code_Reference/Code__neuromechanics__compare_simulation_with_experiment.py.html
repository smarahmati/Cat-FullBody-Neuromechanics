  24 def compare_simulation_with_experiment(result, settings):
  25     t = result["time"]
  26     target = experimental_comparison_targets(
  27         t,
  28         settings["period_s"],
  29         settings.get("reference_table_dir"),
  30     )
  31     u = result["excitation"]
  32     names = list(result["names"])
  33     full_target = experimental_comparison_targets(
  34         np.linspace(0, settings["period_s"], 801),
  35         settings["period_s"],
  36         settings.get("reference_table_dir"),
  37     )["excitation"]
  38     full_input = configured_excitation_samples(settings, u)
  39     joint = np.sqrt(np.mean(np.rad2deg(result["joint_q"] - target["joint_q"]) ** 2, axis=0))
  40     force = np.sqrt(np.mean((result["grf"] - target["grf"]) ** 2, axis=0))
  41     provenance = json.loads((ROOT / "Data/Experimental_Reference_Data/INPUT_PROVENANCE.json").read_text())
  42     emg = []
  43     for side, prefix in enumerate(["L", "R", "FL", "FR"]):
  44         for item in provenance["channels"]:
  45             if "EMG" not in item["kind"]:
  46                 # Skip the remaining work for this item.
  47                 continue
  48             fore = item["kind"].startswith("pooled_fore")
  49             if fore != (side >= 2):
  50                 # Skip the remaining work for this item.
  51                 continue
  52             name = prefix + "_" + item["channel"]
  53             if name not in names:
  54                 # Skip the remaining work for this item.
  55                 continue
  56             i = names.index(name)
  57             source_index = (
  58                 (22 if side == 2 else 31)
  59                 + [c["channel"] for c in provenance["channels"][11:]].index(item["channel"])
  60                 if fore
  61                 else (0 if side == 0 else 11)
  62                 + [c["channel"] for c in provenance["channels"][:11]].index(item["channel"])
  63             )
  64             ref = target["excitation"][:, source_index]
  65             shape = ref / max(1e-12, float(full_target[:, source_index].max()))
  66             pred = u[:, i] / max(0.02, float(full_input[:, i].max()))
  67             corr = (
  68                 float(np.corrcoef(pred, shape)[0, 1])
  69                 if np.std(pred) > 1e-9 and np.std(shape) > 1e-9
  70                 else 0.0
  71             )
  72             emg.append(
  73                 dict(
  74                     channel=name,
  75                     kind=item["kind"] if side != 0 else "assumed_LH_from_recorded_RH",
  76                     peak_normalized_rmse=float(np.sqrt(np.mean((pred - shape) ** 2))),
  77                     correlation=corr,
  78                 )
  79             )
  80     extra = load_reference_tables(settings.get("reference_table_dir"))[2]
  81     for side, prefix in enumerate(["L", "R"]):
  82         i = names.index(prefix + "_FHL")
  83         shape = np.array(
  84             [
  85                 periodic_interpolate(
  86                     extra,
  87                     time / settings["period_s"] + 0.5 + [0.5, 0][side],
  88                 )
  89                 for time in t
  90             ]
  91         )
  92         pred = u[:, i] / max(0.02, float(full_input[:, i].max()))
  93         emg.append(
  94             dict(
  95                 channel=prefix + "_FHL",
  96                 kind="recorded_RH_EMG" if side else "assumed_LH_from_recorded_RH",
  97                 peak_normalized_rmse=float(np.sqrt(np.mean((pred - shape) ** 2))),
  98                 correlation=(
  99                     float(np.corrcoef(pred, shape)[0, 1])
 100                     if np.std(pred) > 1e-9
 101                     else 0.0
 102                 ),
 103             )
 104         )
 105     scope = [
 106         "LH: assumed mirror",
 107         "RH: measured Ted_046",
 108         "LF: separate-study mean; pooled EMG",
 109         "RF: separate-study mean; pooled EMG",
 110     ]
 111     phase_scores = emg_input_timing_scores(names, settings)
 112     for item in emg:
 113         item["complete_input_cycle_timing"] = phase_scores[item["channel"]]
 114     return dict(
 115         duration_s=float(t[-1]),
 116         failure=str(result["failure"]),
 117         limbs=[
 118             dict(
 119                 limb=label,
 120                 joint_rmse_deg=joint[i].tolist(),
 121                 ap_grf_rmse_N=float(force[i, 0]),
 122                 vertical_grf_rmse_N=float(force[i, 2]),
 123             )
 124             for i, label in enumerate(scope)
 125         ],
 126         emg=emg,
 127         completed_60s_without_detected_fall=bool(
 128             t[-1] >= 59.99
 129             and not str(result["failure"])
 130         ),
 131         walking_acceptance_requires_separate_gates=True,
 132         calibration_to_assembled_source_references=True,
 133         EMG_peak_normalization=(
 134             "Complete configured input sequence, one fixed peak per muscle"
 135             if settings.get("excitation_sequence_file")
 136             else "Complete configured input cycle, not the displayed interval"
 137         ),
 138         EMG_timing_scope="Prescribed full-cycle intact input, normalized to intact input peaks; not a claim that the trial completed that cycle. Whole-trial shape RMSE uses the actual saved excitation.",
 139         independent_four_limb_validation=False,
 140     )
