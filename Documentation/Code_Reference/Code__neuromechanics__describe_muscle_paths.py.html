   1 """Read the real tendon transmission, including branches and wrap objects.
   2 
   3 This is an inspection function. It never substitutes a straight line for the
   4 engine path and never changes anatomy. Names come from the loaded model.
   5 """
   6 
   7 import numpy as np
   8 
   9 import mujoco as mj
  10 
  11 
  12 def describe_muscle_paths(state):
  13     """Return SI coordinates, connected bodies and ordered path elements.
  14 
  15     A MuJoCo spatial tendon can contain multiple branches. A branch endpoint
  16     is not necessarily a separate anatomical origin/insertion. Side sites
  17     choose the side of a wrapping object; the tendon does not pass through
  18     those sites. The distinction matters when interpreting an attachment edit.
  19     """
  20     s = state
  21     m = s["model"]
  22     d = s["data"]
  23     moving = {int(meta[5]): row for row, meta in enumerate(s["routing_meta"])}
  24     result = []
  25     for i, name in enumerate(s["names"]):
  26         tendon_id = int(m.actuator_trnid[i, 0])
  27         if int(m.actuator_trntype[i]) != int(mj.mjtTrn.mjTRN_TENDON):
  28             # Stop and explain the invalid model setting or failed calculation.
  29             raise ValueError(f"{name}: inspection expects a tendon transmission")
  30         start = int(m.tendon_adr[tendon_id])
  31         branches = [dict(divisor=1.0, elements=[])]
  32         for k in range(start, start + int(m.tendon_num[tendon_id])):
  33             kind = int(m.wrap_type[k])
  34             object_id = int(m.wrap_objid[k])
  35             if kind == int(mj.mjtWrap.mjWRAP_PULLEY):
  36                 branches.append(dict(divisor=float(m.wrap_prm[k]), elements=[]))
  37                 # Skip the remaining work for this item.
  38                 continue
  39             if kind == int(mj.mjtWrap.mjWRAP_SITE):
  40                 body = int(m.site_bodyid[object_id])
  41                 entry = dict(
  42                     kind="site",
  43                     name=m.site(object_id).name,
  44                     body=m.body(body).name,
  45                     local_position_m=m.site_pos[object_id].tolist(),
  46                     world_position_m=d.site_xpos[object_id].tolist(),
  47                     moving=object_id in moving,
  48                     editable_file="routing.py" if object_id in moving else "attachments.py",
  49                 )
  50             elif kind in (int(mj.mjtWrap.mjWRAP_CYLINDER), int(mj.mjtWrap.mjWRAP_SPHERE)):
  51                 body = int(m.geom_bodyid[object_id])
  52                 side = int(m.wrap_prm[k])
  53                 entry = dict(
  54                     kind="wrap",
  55                     name=m.geom(object_id).name,
  56                     shape=(
  57                         "cylinder"
  58                         if kind == int(mj.mjtWrap.mjWRAP_CYLINDER)
  59                         else "sphere"
  60                     ),
  61                     body=m.body(body).name,
  62                     local_position_m=m.geom_pos[object_id].tolist(),
  63                     radius_m=float(m.geom_size[object_id, 0]),
  64                     editable_file="wrapping.py",
  65                     side_site=(
  66                         None
  67                         if side < 0
  68                         else dict(
  69                             name=m.site(side).name,
  70                             body=m.body(int(m.site_bodyid[side])).name,
  71                             local_position_m=m.site_pos[side].tolist(),
  72                             moving=side in moving,
  73                             editable_file=(
  74                                 "routing.py"
  75                                 if side in moving
  76                                 else "attachments.py"
  77                             ),
  78                         )
  79                     ),
  80                 )
  81             else:
  82                 # Stop and explain the invalid model setting or failed calculation.
  83                 raise ValueError(f"{name}: unsupported wrap type {kind}")
  84             branches[-1]["elements"].append(entry)
  85         branches = [b for b in branches if b["elements"]]
  86         for branch_index, branch in enumerate(branches):
  87             sites = [e for e in branch["elements"] if e["kind"] == "site"]
  88             for index, point in enumerate(sites):
  89                 point["role"] = (
  90                     ("origin" if branch_index == 0 else "internal branch start")
  91                     if index == 0
  92                     else (
  93                         (
  94                             "insertion"
  95                             if branch_index == len(branches) - 1
  96                             else "internal branch end"
  97                         )
  98                         if index == len(sites) - 1
  99                         else "via point"
 100                     )
 101                 )
 102         result.append(
 103             dict(
 104                 muscle=str(name),
 105                 tendon=m.tendon(tendon_id).name,
 106                 branches=branches,
 107                 path_length_m=float(d.actuator_length[i]),
 108                 length_offset_m=float(s["mtu_length_offset"][i]),
 109                 MT_length_m=float(d.actuator_length[i] + s["mtu_length_offset"][i]),
 110                 tendon_gear=float(m.actuator_gear[i, 0]),
 111             )
 112         )
 113     return result
