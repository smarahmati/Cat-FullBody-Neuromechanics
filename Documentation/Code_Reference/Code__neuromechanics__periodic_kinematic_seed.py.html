   1 """Periodic kinematic seed.
   2 
   3 Inputs/outputs: see the named function arguments and returned values.
   4 Lengths use metres, time seconds, force newtons, and angles radians unless stated.
   5 See Documentation/model_methods.html for equations and parameter definitions.
   6 """
   7 
   8 import numpy as np
   9 
  10 
  11 def periodic_kinematic_seed(pose, period, stride, harmonics=8):
  12     """Smooth OFFLINE pose prior and differentiate it analytically.
  13 
  14     Raw measurements have a small cycle-end mismatch and linear-interpolation
  15     corners. Differentiating those corners creates false accelerations. This
  16     band-limited prior is used ONLY for inverse seeding, never for scoring or
  17     smoothing a forward result. The retained harmonics are documented.
  18     """
  19     n = len(pose)
  20     phase = np.arange(n) / n
  21     values = pose.copy()
  22     values[:, 0] -= stride * phase
  23     coefficients = np.fft.rfft(values, axis=0)
  24     coefficients[harmonics + 1 :] = 0.0
  25     omega = (2j * np.pi * np.arange(len(coefficients)) / period)[:, None]
  26     q = np.fft.irfft(coefficients, n=n, axis=0)
  27     v = np.fft.irfft(coefficients * omega, n=n, axis=0)
  28     a = np.fft.irfft(coefficients * omega**2, n=n, axis=0)
  29     q[:, 0] += stride * phase
  30     v[:, 0] += stride / period
  31     return q, v, a
