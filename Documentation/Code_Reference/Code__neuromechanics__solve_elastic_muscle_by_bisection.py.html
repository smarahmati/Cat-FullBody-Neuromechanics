   1 """Elastic fiber step in SI units; see Documentation/model_methods.html, section 5."""
   2 
   3 import numpy as np
   4 
   5 # Read the permitted number of fiber-force balancing steps from the config.
   6 from model_configuration.numerical_parameters import BISECTION_ITERATIONS
   7 
   8 from numba import njit
   9 
  10 from .muscle_force_length_velocity import muscle_force_length_velocity
  11 
  12 from .elastic_tendon_force import elastic_tendon_force
  13 
  14 
  15 # This compiled function calls muscle_force_length_velocity/elastic_tendon_force from other modules.
  16 # Numba disk caches do not reliably invalidate when imported callees change.
  17 # Compile once per process so edits to those physical laws always take effect.
  18 @njit(cache=False)
  19 def solve_elastic_muscle_by_bisection(
  20     mt,
  21     previous,
  22     activation,
  23     lopt,
  24     slack,
  25     params,
  26     e0,
  27     beta,
  28     dt,
  29 ):
  30     """Backward-Euler fiber step with native MTU geometry held at its new value."""
  31     n = len(mt)
  32     length = np.empty(n)
  33     velocity = np.empty(n)
  34     tension = np.empty(n)
  35     residual = np.empty(n)
  36     bound = np.zeros(n, np.int64)
  37     damping = np.asarray(beta).reshape(-1)
  38     for i in range(n):
  39         coefficient = damping[i] if len(damping) > 1 else damping[0]
  40         cosine = params[i, 9] if params.shape[1] > 9 else 1.0
  41         low = 0.2 * lopt[i]
  42         high = max(low + 1e-9, mt[i] / cosine)
  43         vscale = lopt[i] * params[i, 6]
  44         # Repeatedly narrow the fiber-length interval until the forces balance.
  45         for _ in range(BISECTION_ITERATIONS):
  46             lf = 0.5 * (low + high)
  47             v = (lf - previous[i]) / (dt * vscale)
  48             fl, fv, fp = muscle_force_length_velocity(lf / lopt[i], v, params[i])
  49             ft = elastic_tendon_force((mt[i] - lf * cosine) / slack[i] - 1, e0[i])
  50             error = ft - activation[i] * fl * fv - fp - coefficient * v
  51             if error > 0:
  52                 low = lf
  53             else:
  54                 high = lf
  55         lf = 0.5 * (low + high)
  56         vf = (lf - previous[i]) / dt
  57         fl, fv, fp = muscle_force_length_velocity(lf / lopt[i], vf / vscale, params[i])
  58         ft = elastic_tendon_force((mt[i] - lf * cosine) / slack[i] - 1, e0[i])
  59         length[i] = lf
  60         velocity[i] = vf
  61         tension[i] = params[i, 2] * ft
  62         residual[i] = params[i, 2] * (
  63             ft
  64             - activation[i] * fl * fv
  65             - fp
  66             - coefficient * vf / vscale
  67         )
  68         if (
  69             lf <= 0.200001 * lopt[i]
  70             or mt[i] - lf * cosine <= 1e-9
  71             or mt[i] <= 0.2 * lopt[i] * cosine
  72         ):
  73             bound[i] = 1
  74     return length, velocity, tension, residual, bound
