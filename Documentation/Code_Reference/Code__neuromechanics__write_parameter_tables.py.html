  13 def same_parameter_value(first, second):
  14     """Compare values without counting negligible round-off as a change."""
  15     if isinstance(first, (float, int)) and isinstance(second, (float, int)):
  16         return math.isclose(first, second, rel_tol=1e-10, abs_tol=1e-12)
  17     return first == second

  20 def explain_parameter_change(row, actual):
  21     """Explain documented choices; explicitly identify missing rationale."""
  22     original = row['original']
  23     if not same_parameter_value(actual, row['applied']):
  24         return 'Changed in this run relative to the supplied release. The reason for this user-selected value is not recorded here.'
  25     if original is not None and same_parameter_value(original, actual):
  26         return 'Retained the verified source value; no numerical change from that source.'
  27     # Lack of a source value must not be mistaken for a measured parameter.
  28     status = ('No verified source value is available for direct comparison. ' if original is None
  29               else 'Different from the verified source value. ')
  30     if isinstance(original, (float, int)) and isinstance(actual, (float, int)) and original != 0:
  31         status += f'Change relative to source: {100 * (actual - original) / abs(original):+.3g}%. '
  32     field = row['field']
  33     name = row['name']
  34     # These are explanations of calibration/model choices, not new measurements.
  35     if field == 'max_isometric_muscle_force_along_tendon_N':
  36         reason = 'Force capacity was calibrated with motor inputs to improve mechanical agreement; the fit does not uniquely identify it.'
  37     elif field in ['optimal_fiber_length_m', 'max_fiber_speed_optimal_lengths_per_s'] and name.startswith(('FL_', 'FR_')):
  38         reason = ('Fiber length and shortening speed were calibrated to set the force-length and force-velocity operating ranges. '
  39                   'Physical maximum fiber speed is optimal_fiber_length_m multiplied by max_fiber_speed_optimal_lengths_per_s. '
  40                   'The source and applied numbers are given in this row; these are not new tissue measurements.')
  41     elif field == 'degroote_force_velocity_enabled':
  42         reason = 'Model-law choice: 1 enables the smooth De Groote force-velocity relation; 0 selects the alternative implemented law.'
  43     elif field == 'passive_fiber_enabled':
  44         reason = ('The published passive force-length law is included in every muscle channel. '
  45                   'Its coefficients are from Prilutsky et al. (2016), Table 10.3, and Rahmati et al. (2025), Table 3. '
  46                   'Applying the same normalized law to added muscle channels is a modeling assumption.')
  47     elif field == 'normalized_fiber_damping':
  48         reason = 'Effective muscle force-balance coefficient; source-law conversion and calibration are described in the methods.'
  49     elif field == 'tendon_strain_at_max_force':
  50         reason = 'Assumed elastic-tendon compliance; not a measured tendon strain for this animal.'
  51     elif field in ['normal_stiffness_N_m32', 'normal_dissipation_s_m', 'shear_stiffness_N_m', 'shear_damping_Ns_m', 'friction_coefficient']:
  52         reason = 'Effective paw-contact parameter from model calibration; not an independently measured paw-material constant.'
  53     elif field == 'COM_position_in_body_m' and name == 'cat_root':
  54         reason = 'Trunk center of mass was calibrated to adjust body balance and limb loading.'
  55     elif field == 'body_position_in_parent_m' and name in ['FL_scapula', 'FR_scapula']:
  56         reason = 'Forelimb mounting position was calibrated to improve whole-body geometry and loading.'
  57     elif field == 'passive_damping' and name in ['FL_wrist', 'FR_wrist', 'FL_mcp', 'FR_mcp']:
  58         reason = 'Distal forelimb passive damping was calibrated to improve mechanical behavior.'
  59     else:
  60         reason = 'The available record does not establish an individual source-to-model adjustment rationale; no more specific reason is claimed.'
  61     return status + reason

  64 def parameter_source_description(row):
  65     """Return the scientific source statement supplied with a parameter."""
  66     return row.get('source', 'No verified source is established for this value.')

  69 def write_parameter_tables(parameters, provenance, output):
  70     """Compare the verified scientific source directly with actual simulator values."""
  71     escape = lambda x: html.escape(
  72         json.dumps(x, ensure_ascii=False)
  73         if isinstance(x, (list, dict))
  74         else (f"{x:.8g}" if isinstance(x, float) else str(x))
  75     )
  76     groups = {}
  77     for row in provenance["parameters"]:
  78         # Accept recorded configurations with the former force-field name.
  79         # This changes table terminology only; the archived input stays untouched.
  80         row = dict(row)
  81         old_force_name = "max_isometric_tendon_force_N"
  82         force_name = "max_isometric_muscle_force_along_tendon_N"
  83         if row["category"] == "muscle_tendon" and row["field"] == old_force_name:
  84             row["field"] = force_name
  85             row["meaning"] = (
  86                 "Maximum isometric muscle force (F0). "
  87                 "The pennation projection is already included; do not apply it twice."
  88             )
  89         category = row["category"]
  90         name = row["name"]
  91         field = row["field"]
  92         if category == "initial_state":
  93             # Skip the remaining work for this item.
  94             continue
  95         actual = (
  96             parameters[category][name]
  97             if category == "environment"
  98             else parameters[category][name][
  99                 old_force_name
 100                 if field == force_name and force_name not in parameters[category][name]
 101                 else field
 102             ]
 103         )
 104         cells = [
 105             name,
 106             field,
 107             row["unit"],
 108             row["meaning"],
 109             row["original"] if row["original"] is not None else "Not established",
 110             actual,
 111             explain_parameter_change(row, actual),
 112             parameter_source_description(row),
 113         ]
 114         groups.setdefault(category, []).append(
 115             "<tr>"
 116             + "".join("<td>" + escape(v) + "</td>" for v in cells)
 117             + "</tr>"
 118         )
 119     parts = [
 120         '<!doctype html><meta charset="utf-8"><title>Actual model parameter tables</title><style>body{margin:30px;font:16px/1.5 Arial;color:#19313e}table{border-collapse:collapse;font-size:13px;width:100%}th,td{border:1px solid #ccd5dc;padding:7px;text-align:left;overflow-wrap:anywhere}th{background:#e9f1f5;position:sticky;top:0}td{max-width:330px}th:nth-child(3),td:nth-child(3){min-width:52px}summary{font-size:20px;margin:25px 0;cursor:pointer}.note{background:#fff1d6;padding:16px}input{padding:9px;width:450px}</style>',
 121         '<h1>Source values and values used in this model</h1><p class="note"><strong>Source value:</strong> a verified value from the author’s original work or cited literature, in the listed units. '
 122         '<strong>Value used in this model:</strong> the actual value read back from the assembled simulation. '
 123         '<strong>What changed and why:</strong> the source-to-model difference and documented rationale. '
 124         '“Not established” means no verified numerical source value is available; the model value is an assumption, estimate or fitted value. '
 125         'Successful fitting does not prove that a fitted value is the animal’s true parameter.</p>',
 126         '<p>Positions use their specified local body frame. Initial conditions appear separately below. '
 127         'Displayed scalars use up to eight significant digits; saved JSON/CSV values retain full precision. '
 128         'Editing a parameter requires a new simulation and does not automatically retune excitation.</p>'
 129         '<input id="search" placeholder="Filter: e.g. R_VL, mass_kg, tendon, damping" oninput="filterRows(this.value)">',
 130     ]
 131     for category, rows in groups.items():
 132         parts.append(
 133             "<details "
 134             + ("open" if category == "muscle_tendon" else "")
 135             + "><summary>"
 136             + escape(category)
 137             + " ("
 138             + str(len(rows))
 139             + " entries)</summary><table><thead><tr>"
 140             + "".join(
 141                 "<th>" + x + "</th>"
 142                 for x in [
 143                     "Object",
 144                     "Parameter",
 145                     "Unit",
 146                     "Meaning",
 147                     "Source value",
 148                     "Value used in this model",
 149                     "What changed and why",
 150                     "Source / evidence",
 151                 ]
 152             )
 153             + "</tr></thead><tbody>"
 154             + "".join(rows)
 155             + "</tbody></table></details>"
 156         )
 157     parts.append(
 158         "<h2>Initial conditions (only at t = 0)</h2><p>These are fitted simulation states, not a claim of measured fiber lengths or contact shear. They are never restored each cycle.</p>"
 159     )
 160     for kind, rows in parameters["initial_state"].items():
 161         parts.append(
 162             "<details><summary>"
 163             + escape(kind)
 164             + "</summary><table><tr><th>Object</th><th>Applied initial value</th></tr>"
 165             + "".join(
 166                 "<tr><td>" + escape(k) + "</td><td>" + escape(v) + "</td></tr>"
 167                 for k, v in rows.items()
 168             )
 169             + "</table></details>"
 170         )
 171     parts.append('<script>function filterRows(q){q=q.toLowerCase();document.querySelectorAll("tbody tr").forEach(r=>r.hidden=!r.textContent.toLowerCase().includes(q));'
 172                  'if(q)document.querySelectorAll("details:not([data-history])").forEach(d=>d.open=true)}</script>')
 173     # Save the computed values or figure for inspection and reproducibility.
 174     Path(output).write_text("".join(parts), encoding="utf-8")
