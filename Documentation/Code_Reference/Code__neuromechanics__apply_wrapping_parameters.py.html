   3 import numpy as np

   5 import mujoco as mj
   6 
   7 from .load_parameter_file import load_parameter_file

   9 
  10 def apply_wrapping_parameters(state, path):
  11     """Only tendon wrap objects may be edited here; bones/contact are separate.
  12 
  13     Historical bundles without this file retain their original wrap geometry.
  14     Topology and body ownership are deliberately kept in the anatomical model.
  15     """
  16     if not path.exists():
  17         return
  18     m = state["model"]
  19     ids = sorted(
  20         {
  21             int(m.wrap_objid[k])
  22             for k in range(m.nwrap)
  23             if int(m.wrap_type[k])
  24             in (int(mj.mjtWrap.mjWRAP_CYLINDER), int(mj.mjtWrap.mjWRAP_SPHERE))
  25         }
  26     )
  27     entries = load_parameter_file(path)
  28     if set(entries) != {m.geom(i).name for i in ids}:
  29         # Stop and explain the invalid model setting or failed calculation.
  30         raise ValueError("wrapping.py: wrap object names differ from the model")
  31     for i in ids:
  32         name = m.geom(i).name
  33         row = entries[name]
  34         if set(row) != {"position_in_body_m", "orientation_in_body_wxyz", "radius_m"}:
  35             # Stop and explain the invalid model setting or failed calculation.
  36             raise ValueError(f"wrapping.py/{name}: unknown or missing field")
  37         position = np.asarray(row["position_in_body_m"], float)
  38         quaternion = np.asarray(row["orientation_in_body_wxyz"], float)
  39         radius = float(row["radius_m"])
  40         if (
  41             position.shape != (3,)
  42             or quaternion.shape != (4,)
  43             or not np.isfinite(position).all()
  44             or not np.isfinite(quaternion).all()
  45         ):
  46             # Stop and explain the invalid model setting or failed calculation.
  47             raise ValueError(f"wrapping.py/{name}: expected finite xyz and wxyz vectors")
  48         if (
  49             not np.isfinite(radius)
  50             or radius <= 0
  51             or not np.isclose(np.linalg.norm(quaternion), 1.0, atol=1e-7)
  52         ):
  53             # Stop and explain the invalid model setting or failed calculation.
  54             raise ValueError(
  55                 f"wrapping.py/{name}: radius must be positive and quaternion unit length"
  56             )
  57         if not np.array_equal(m.geom_pos[i], position) or not np.array_equal(
  58             m.geom_quat[i],
  59             quaternion,
  60         ):
  61             m.geom_sameframe[i] = 0  # Invalidate the compiled coincident-frame shortcut.
  62         m.geom_pos[i] = position
  63         m.geom_quat[i] = quaternion
  64         m.geom_size[i, 0] = radius
