   3 import numpy as np

   5 from numba import njit

   7 from .elastic_tendon_force import elastic_tendon_force
   8 
   9 from .muscle_force_length_velocity import muscle_force_length_velocity
  10 
  11 
  12 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  13 @njit(cache=False)
  14 def initial_elastic_force(mtu, fiber, activation, lopt, slack, parameters, e0, beta):
  15     """Evaluate tension, then invert the monotone fiber force/velocity law.
  16 
  17     This initializes a state once. It is never an added motor command. The
  18     wide velocity bracket permits diagnostics to expose out-of-domain seeds;
  19     independent validation still checks the published force-velocity domain.
  20     """
  21     n = len(mtu)
  22     force = np.empty(n)
  23     velocity = np.empty(n)
  24     residual = np.empty(n)
  25     for i in range(n):
  26         ft = elastic_tendon_force(
  27             (mtu[i] - fiber[i] * parameters[i, 9]) / slack[i] - 1,
  28             e0[i],
  29         )
  30         low = -20.0
  31         high = 20.0
  32         x = 0.0
  33         for _ in range(60):
  34             x = (low + high) / 2
  35             fl, fv, fp = muscle_force_length_velocity(fiber[i] / lopt[i], x, parameters[i])
  36             error = ft - activation[i] * fl * fv - fp - beta[i] * x
  37             if abs(error) < 1e-12:
  38                 # Leave this loop because the convergence or stopping condition was met.
  39                 break
  40             if error > 0:
  41                 low = x
  42             else:
  43                 high = x
  44         force[i] = ft * parameters[i, 2]
  45         velocity[i] = x * lopt[i] * parameters[i, 6]
  46         residual[i] = parameters[i, 2] * error
  47     return force, velocity, residual
