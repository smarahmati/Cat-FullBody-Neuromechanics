   5 See Documentation/model_methods.html for equations and parameter definitions.

   8 import numpy as np

  10 import mujoco as mj
  11 
  12 from .assemble_reference_kinematics import assemble_reference_kinematics
  13 
  14 from .update_muscle_paths import update_muscle_paths
  15 
  16 from .periodic_interpolate import periodic_interpolate
  17 
  18 from .activation_derivative import activation_derivative
  19 
  20 from .solve_elastic_muscle_by_bisection import solve_elastic_muscle_by_bisection
  21 
  22 from .muscle_excitation import muscle_excitation

  25 def precondition_fibers(s):
  26     """Before t=0 only: periodic activation/fiber history on reference lengths."""
  27     m, d = s["model"], s["data"]
  28     q0 = d.qpos.copy()
  29     v0 = d.qvel.copy()
  30     ref = assemble_reference_kinematics(s, 240)
  31     length = []
  32     for q in ref["pose"]:
  33         d.qpos[:] = q
  34         update_muscle_paths(s)
  35         mj.mj_forward(m, d)
  36         length.append(d.actuator_length + s["mtu_length_offset"])
  37     length = np.vstack([length, length[0]])
  38     a = np.zeros(m.nu)
  39     lf = s["fiber_length"].copy()
  40     dt = 0.001
  41     T = s["settings"]["period_s"]
  42     steps = round(3 * T / dt)
  43     dt = 3 * T / steps
  44     for t in np.arange(steps) * dt - 3 * T:
  45         a = np.clip(
  46             a + dt * activation_derivative(muscle_excitation(s, t), a, s["time_constants"]),
  47             0,
  48             1,
  49         )
  50         lf, v, f, r, b = solve_elastic_muscle_by_bisection(
  51             periodic_interpolate(length, (t + dt) / T),
  52             lf,
  53             a,
  54             s["lopt"],
  55             s["slack"],
  56             s["muscle_params"],
  57             s["e0"],
  58             s["fiber_damping_coefficients"],
  59             dt,
  60         )
  61     d.qpos[:] = q0
  62     d.qvel[:] = v0
  63     d.act[:] = a
  64     update_muscle_paths(s)
  65     mj.mj_forward(m, d)
  66     s.update(fiber_length=lf, fiber_velocity=v, force=f, residual=r, fiber_bound=b)
