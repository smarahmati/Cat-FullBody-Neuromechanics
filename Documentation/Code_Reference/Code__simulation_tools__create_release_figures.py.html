Code/simulation_tools/create_release_figures.py

   1 """Create readable late-trial comparison figures from saved forward results.
   2 
   3 This is postprocessing only: it never changes inputs, states, force values or
   4 time alignment. The last four seconds show the waveform detail. A separate
   5 last-ten-second table reports RMSE against the same fixed-clock references.
   6 Run from the root with: python Code/simulation_tools/create_release_figures.py
   7 """
   8 from pathlib import Path
   9 import json
  10 import sys
  11 import numpy as np
  12 import matplotlib
  13 matplotlib.use('Agg')
  14 import matplotlib.pyplot as plt
  15 
  16 # Resolve the repository from the script, so the working directory is irrelevant.
  17 ROOT=Path(__file__).resolve().parents[2]
  18 sys.path.insert(0,str(ROOT/'Code'))
  19 from neuromechanics.experimental_comparison_targets import experimental_comparison_targets
  20 from neuromechanics.emg_comparison_cycles import emg_comparison_cycles
  21 
  22 
  23 def create_release_figures():
  24     """Save all-joint, all-paw, all available EMG, and intervention comparisons."""
  25     folder=ROOT/'Simulation_Results/Intact_Walking'
  26     output=ROOT/'Documentation/Figures_and_Videos'
  27     output.mkdir(parents=True,exist_ok=True)
  28     with np.load(folder/'forward_simulation_timeseries.npz') as stored:
  29         result={key:stored[key] for key in stored.files}
  30     settings=json.loads((folder/'settings.json').read_text())
  31     times=result['time']
  32     late=times>=times[-1]-4
  33     score=times>=times[-1]-10
  34     target=experimental_comparison_targets(times,settings['period_s'],settings.get('reference_table_dir'))
  35     limb_labels=['LH · assumed mirror','RH · measured','LF · separate study','RF · separate study']
  36     joints=[['Hip','Knee','Ankle','MTP']]*2+[['Shoulder','Elbow','Wrist','MCP']]*2
  37     plt.rcParams.update({'font.size':10,'axes.spines.top':False,'axes.spines.right':False})
  38     fig,axes=plt.subplots(4,4,figsize=(16,10),sharex=True)
  39     for side in range(4):
  40         signs=np.array([-1,1,-1,-1]) if side<2 else np.array([1,-1,-1,-1])
  41         offset=np.array([0,np.pi,np.pi,np.pi]) if side<2 else np.full(4,np.pi)
  42         for j in range(4):
  43             a=axes[side,j]
  44             a.plot(times[late],np.rad2deg(offset[j]+signs[j]*target['joint_q'][late,side,j]),'k--',lw=1.2,label='Reference')
  45             a.plot(times[late],np.rad2deg(offset[j]+signs[j]*result['joint_q'][late,side,j]),color='#067f97',lw=1.4,label='Forward')
  46             a.set_title(limb_labels[side]+' · '+joints[side][j],fontsize=10)
  47             a.grid(alpha=.18)
  48         axes[side,0].set_ylabel('Anatomical angle (deg)')
  49     axes[0,0].legend(fontsize=8)
  50     for a in axes[-1]:a.set_xlabel('Simulation time (s)')
  51     fig.suptitle('All 16 joints · same anatomical conventions and fixed reference clocks\nLate four-second interval; no time warping',fontsize=15)
  52     fig.tight_layout(rect=(0,0,1,.94));fig.savefig(output/'joint_comparison.png',dpi=140);plt.close(fig)
  53     fig,axes=plt.subplots(4,2,figsize=(12,10),sharex=True)
  54     for side in range(4):
  55         for col,k in enumerate([0,2]):
  56             a=axes[side,col]
  57             a.plot(times[late],target['grf'][late,side,k],'k--',lw=1.2,label='Reference')
  58             a.plot(times[late],result['grf'][late,side,k],color='#067f97',lw=1.4,label='Raw contact force')
  59             a.set_title(limb_labels[side]+(' · anterior–posterior (AP) Fx' if k==0 else ' · vertical Fz'))
  60             a.set_ylabel('Force (N)');a.grid(alpha=.18)
  61             if k==0:a.axhline(0,color='gray',lw=.5)
  62     axes[0,0].legend(fontsize=8)
  63     for a in axes[-1]:a.set_xlabel('Simulation time (s)')
  64     fig.suptitle('Paw Ground_Reaction_Forces · unfiltered forward contact forces',fontsize=16)
  65     fig.tight_layout(rect=(0,0,1,.96));fig.savefig(output/'grf_comparison.png',dpi=150);plt.close(fig)
  66     pairs=emg_comparison_cycles(result['names'],settings)
  67     for prefix in ['L','R','FL','FR']:
  68         selected=[p for p in pairs if p['limb']==prefix]
  69         fig,axes=plt.subplots(2,4,figsize=(14,6),sharex=True,sharey=True)
  70         for a,pair in zip(axes.flat,selected):
  71             phase=np.arange(len(pair['reference']))/len(pair['reference'])
  72             a.plot(phase*100,pair['reference'],'k--',label='EMG reference')
  73             a.plot(phase*100,pair['excitation'],color='#067f97',label='Excitation / cycle peak')
  74             a.set_title(pair['channel']);a.grid(alpha=.18)
  75         for a in list(axes.flat)[len(selected):]:a.set_visible(False)
  76         axes[0,0].legend(fontsize=8)
  77         for a in axes[-1]:a.set_xlabel('Input cycle (%)')
  78         for a in axes[:,0]:a.set_ylabel('Peak-normalized shape')
  79         scope='measured RH EMG' if prefix=='R' else 'assumed from RH' if prefix=='L' else 'pooled forelimb EMG'
  80         fig.suptitle(prefix+' · '+scope+'\nShape normalization is for comparison only; raw excitation is unchanged.',fontsize=14)
  81         fig.tight_layout(rect=(0,0,1,.91));fig.savefig(output/('emg_'+prefix+'.png'),dpi=140);plt.close(fig)
  82     report={'window_start_s':float(times[score][0]),'window_end_s':float(times[-1]),'joint_RMSE_deg':np.sqrt(np.mean(np.rad2deg(result['joint_q'][score]-target['joint_q'][score])**2,axis=0)).tolist(),'GRF_RMSE_N':np.sqrt(np.mean((result['grf'][score]-target['grf'][score])**2,axis=0)).tolist()}
  83     (ROOT/'Simulation_Results/Intact_Walking/late_window_metrics.json').write_text(json.dumps(report,indent=2))
  84     print('Comparison figures and fixed-clock metrics saved.')
  85 
  86 
  87 if __name__=='__main__':
  88     create_release_figures()
