   1 """Compute a NEW fully forward trial, optionally opening the result viewer.
   2 
   3 Edit the defaults below, then press Run in your IDE. Command-line options win:
   4     python run.py --seconds 6 --gui
   5     python run.py --seconds 60 --no-gui
   6 The GUI displays the computed intact walking, muscle states and contact forces.
   7 """
   8 from pathlib import Path
   9 import os
  10 import sys
  11 import json
  12 
  13 # These limits prevent many competing CPU threads; they do not change equations.
  14 os.environ.setdefault("OPENBLAS_NUM_THREADS", "1")
  15 os.environ.setdefault("OMP_NUM_THREADS", "1")
  16 # Duration is simulation time, not the wall time needed to calculate it.
  17 SIMULATION_SECONDS = 6.0
  18 # True opens animation and plots AFTER calculation; False only saves results.
  19 SHOW_GUI = True
  20 # The current model has one clearly named parameter folder.
  21 PARAMETER_FOLDER = Path(__file__).resolve().parent / "Data" / "Model_Parameters"
  22 
  23 sys.path.insert(0, str(Path(__file__).resolve().parent / "Code"))
  24 # Select the constants before importing the numerical kernels.
  25 chosen = PARAMETER_FOLDER
  26 if "--parameters" in sys.argv:
  27     chosen = (Path(__file__).resolve().parent / sys.argv[sys.argv.index("--parameters") + 1]).resolve()
  28 elif "--config" in sys.argv:
  29     # Replaying a saved configuration must also reuse its formulation constants.
  30     saved = json.loads((Path(__file__).resolve().parent / sys.argv[sys.argv.index("--config") + 1]).read_text())
  31     chosen = Path(__file__).resolve().parent / saved.get("parameter_bundle_dir", "Data/Model_Parameters")
  32 os.environ["CAT_PARAMETER_FOLDER"] = str(chosen)
  33 from simulation_tools.run_forward_simulation import main
  34 
  35 if __name__ == "__main__":
  36     if "--parameters" not in sys.argv and "--config" not in sys.argv:
  37         sys.argv.extend(["--parameters", str(PARAMETER_FOLDER)])
  38     main(default_seconds=SIMULATION_SECONDS, default_gui=SHOW_GUI)
