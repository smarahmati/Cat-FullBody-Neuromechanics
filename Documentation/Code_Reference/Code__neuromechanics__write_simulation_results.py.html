  26 def write_simulation_results(result, settings, folder):
  27     """Compare raw predictions on the source clock, with readable time windows."""
  28     metrics = compare_simulation_with_experiment(result, settings)
  29     # Save the computed values or figure for inspection and reproducibility.
  30     (folder / "experimental_comparison_metrics.json").write_text(json.dumps(metrics, indent=2))
  31     target = experimental_comparison_targets(
  32         result["time"],
  33         settings["period_s"],
  34         settings.get("reference_table_dir"),
  35     )
  36     # Plot a short interval for waveform detail; metrics still use the full trial.
  37     window_end = float(result["time"][-1])
  38     window_start = max(float(result["time"][0]), window_end - 4.0)
  39     names = [["Hip", "Knee", "Ankle", "MTP"]] * 2 + [
  40         ["Shoulder", "Elbow", "Wrist", "MCP"]
  41     ] * 2
  42     limbs = ["LH (assumed)", "RH (measured)", "LF (separate study)", "RF (separate study)"]
  43     fig, ax = plt.subplots(4, 4, figsize=(15, 10), sharex=True)
  44     for side in range(4):
  45         signs = np.array([-1, 1, -1, -1]) if side < 2 else np.array([1, -1, -1, -1])
  46         offset = np.array([0, np.pi, np.pi, np.pi]) if side < 2 else np.full(4, np.pi)
  47         for j in range(4):
  48             a = ax[side, j]
  49             a.plot(
  50                 result["time"],
  51                 np.rad2deg(offset[j] + signs[j] * target["joint_q"][:, side, j]),
  52                 "k--",
  53                 label="Source reference",
  54             )
  55             a.plot(
  56                 result["time"],
  57                 np.rad2deg(offset[j] + signs[j] * result["joint_q"][:, side, j]),
  58                 color="#087f9c",
  59                 label="Forward result",
  60             )
  61             a.set_title(
  62                 f'{limbs[side]} {names[side][j]}\nRMSE {metrics["limbs"][side]["joint_rmse_deg"][j]:.1f}°',
  63                 fontsize=10,
  64             )
  65             a.grid(alpha=0.2)
  66             a.set_xlim(window_start, window_end)
  67         ax[side, 0].set_ylabel("Anatomical angle (deg)")
  68     ax[0, 0].legend(fontsize=8)
  69     [a.set_xlabel("Time (s)") for a in ax[-1]]
  70     fig.suptitle(f"Joint angles · {window_start:.2f}–{window_end:.2f} s\nRMSE labels use the complete trial; explore any interval in the interactive comparison", fontsize=12)
  71     fig.tight_layout(rect=(0, 0, 1, .94))
  72     # Save the computed values or figure for inspection and reproducibility.
  73     fig.savefig(folder / "all_joint_angles.png", dpi=140)
  74     plt.close(fig)
  75     fig, ax = plt.subplots(4, 2, figsize=(12, 10), sharex=True)
  76     for side in range(4):
  77         for col, index in enumerate([0, 2]):
  78             a = ax[side, col]
  79             a.plot(
  80                 result["time"],
  81                 target["grf"][:, side, index],
  82                 "k--",
  83                 label="Source reference",
  84             )
  85             a.plot(
  86                 result["time"],
  87                 result["grf"][:, side, index],
  88                 color="#087f9c",
  89                 label="Raw contact prediction",
  90             )
  91             a.set_title(limbs[side] + (" anterior–posterior (AP)" if col == 0 else " vertical"))
  92             a.set_ylabel("Force (N)")
  93             a.grid(alpha=0.2)
  94             a.set_xlim(window_start, window_end)
  95     ax[0, 0].legend(fontsize=8)
  96     [a.set_xlabel("Time (s)") for a in ax[-1]]
  97     fig.suptitle(f"Raw paw contact forces · {window_start:.2f}–{window_end:.2f} s", fontsize=12)
  98     fig.tight_layout(rect=(0, 0, 1, .95))
  99     # Save the computed values or figure for inspection and reproducibility.
 100     fig.savefig(folder / "all_grfs.png", dpi=140)
 101     plt.close(fig)
 102     rows = "".join(
 103         "<tr><td>"
 104         + html.escape(m["limb"])
 105         + "</td><td>"
 106         + ", ".join(f"{x:.2f}" for x in m["joint_rmse_deg"])
 107         + f'</td><td>{m["ap_grf_rmse_N"]:.2f}</td><td>{m["vertical_grf_rmse_N"]:.2f}</td></tr>'
 108         for m in metrics["limbs"]
 109     )
 110     content = f"""<!doctype html><meta charset="utf-8"><title>Four-limb forward fit</title><style>body{{max-width:1200px;margin:35px auto;font:17px/1.5 Arial;color:#19313e}}img{{width:100%}}table{{border-collapse:collapse;width:100%}}td,th{{border:1px solid #bbc;padding:8px}}.status{{background:#fff2d7;padding:15px}}</style><h1>Four-limb forward model: current trial</h1><p class="status">Duration: {metrics['duration_s']:.3f} s. Failure: {html.escape(metrics['failure'] or 'none detected within this run')}. This trial does not establish the requested 60-second fit unless the separate validation tests pass.</p><p>The four limbs are mechanically coupled. Excitation drives activation, elastic muscle–tendon forces and native articulated dynamics. Ground_Reaction_Forces come from physical paw contact. Raw reference curves and raw forward outputs use the same clock.</p><table><tr><th>Limb and data basis</th><th>Joint RMSE (degrees, proximal to distal)</th><th>AP GRF RMSE (N)</th><th>Vertical GRF RMSE (N)</th></tr>{rows}</table><p>LH is assumed from RH. Forelimb recordings are from a separate study; EMGs are pooled across sides. These are fitting targets, not simultaneous four-limb validation.</p><img src="all_joint_angles.png"><img src="all_grfs.png"><p><a href="experimental_comparison_metrics.json">All metrics, including each EMG channel</a> · <a href="settings.json">Applied parameters</a> · <a href="../../Documentation/model_methods.html">Equations, sources and assumptions</a></p>"""
 111     emg_files = plot_emg_comparison(result, settings, folder)
 112     # Browser navigation exposes the complete recording without crowding 60 s.
 113     from .write_interactive_comparison import write_interactive_comparison
 114     write_interactive_comparison(result, settings, folder)
 115     content = content.replace("Ground_Reaction_Forces", "GRFs")
 116     content = content.replace("Four-limb forward model: current trial", "Intact-cat forward simulation")
 117     content = content.replace("This trial does not establish the requested 60-second fit unless the separate validation tests pass.", "Duration alone does not establish complete experimental agreement or stability beyond the simulated interval.")
 118     content = content.replace("<table>", '<p style="padding:16px;background:#e6f5f3"><a href="interactive_experimental_comparison.html"><b>Explore all time points: movable four-second window, joint angles and GRFs</b></a></p><p>Static figures show the final four seconds. The table and plot-title RMSE values use the complete saved trial, without phase shifting.</p><table>', 1)
 119     if settings.get("excitation_sequence_file"):
 120         note = '<p class="status"><b>Finite, frozen muscle_excitation sequence.</b> Amplitudes were optimized offline across successive strides. The replay uses elapsed time only, with one initial state and no runtime corrections. This is not a demonstration of an autonomously stable periodic gait, and no repetition beyond the saved sequence is assumed.</p>'
 121         content = content.replace("<table>", note + "<table>", 1)
 122     gallery = "<h2>Excitation shapes compared with available EMG</h2>" + "".join(
 123         f'<img src="{name}" alt="Separate limb muscle_excitation and EMG comparison">'
 124         for name in emg_files
 125     )
 126     content = content.replace(
 127         '<p><a href="experimental_comparison_metrics.json">',
 128         gallery + '<p><a href="experimental_comparison_metrics.json">',
 129     )
 130     doc_link = os.path.relpath(ROOT / "Documentation/model_methods.html", folder).replace(os.sep, "/")
 131     content = content.replace("../../Documentation/model_methods.html", doc_link)
 132     if (folder / "APPLIED_PARAMETERS.html").exists():
 133         content += '<p><a href="APPLIED_PARAMETERS.html">Actual applied parameter table</a> · <a href="APPLIED_PARAMETERS.json">Machine-readable values</a> · <a href="INPUT_HASHES.json">Frozen config/reference hashes</a> · <a href="walking_quality_metrics.json">Walking acceptance checks</a></p>'
 134     if (folder / "ATTACHMENT_MAP.html").exists():
 135         content += '<p><a href="ATTACHMENT_MAP.html">Actual muscle origins, insertions, via points and wrapping</a> · <a href="ATTACHMENT_MAP.json">Connection and coordinate data</a></p>'
 136     # Save the computed values or figure for inspection and reproducibility.
 137     (folder / "simulation_report.html").write_text(content, encoding="utf-8")
 138     return metrics
