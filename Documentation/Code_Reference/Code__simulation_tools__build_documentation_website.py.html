   1 """Package the existing HTML documentation for GitHub Pages.
   2 
   3 This script does not run, fit or alter the neuromechanical model. It copies public
   4 documentation and supporting files into a NEW output directory, preserving the
   5 relative paths used by downloaded HTML. Large raw Results arrays stay in GitHub;
   6 links to those arrays are rewritten to their exact repository revision.
   7 
   8 Parameters:
   9   --output: new/empty folder for the website, usually _site in GitHub Actions.
  10   --repository: GitHub owner/name; defaults to the confirmed publication address.
  11   --revision: commit or branch used for links to downloadable raw result files.
  12   --include-untracked: local preview only; honor .gitignore for these files too.
  13 
  14 Example (from repository root):
  15   python Code/simulation_tools/build_documentation_website.py --output _site
  16 
  17 Only Python's standard library and Git are needed; no simulation dependencies.
  18 """

  42 def forbidden_path(relative):
  43     """Keep ignored environments, private reading copies and restricted meshes out."""
  44     parts = PurePosixPath(relative).parts
  45     return (any(part in {'.git', '.venv', '__pycache__', '.idea', '_site'} for part in parts)
  46             or relative.startswith('Documentation/References/Local_Papers/')
  47             or relative.startswith('Data/Skeletal_Geometry/Model/meshes/'))

  50 def public_repository_files(root, include_untracked=False):
  51     """Use Git's publication inventory, never a recursive copy of the workstation."""
  52     command = ['git', '-C', str(root), 'ls-files', '--cached', '-z']
  53     if include_untracked:
  54         command += ['--others', '--exclude-standard']
  55     names = subprocess.check_output(command).decode('utf-8').split('\0')
  56     return {name for name in names if name and not forbidden_path(name)
  57             and (root / name).is_file() and not (root / name).is_symlink()}

  60 def belongs_in_website(relative):
  61     """Include reports and assets; leave large raw simulation arrays in GitHub."""
  62     if forbidden_path(relative):
  63         return False
  64     if relative in ROOT_PAGES:
  65         return True
  66     if relative.startswith(('Documentation/', 'Code/', 'Data/', 'Verification/')):
  67         return True
  68     if relative.startswith('Simulation_Results/'):
  69         # These snapshots remain reproducibility downloads, not duplicated web pages.
  70         if any(part in {'Model_Parameters_Used', 'Experimental_References_Used', 'Simulation_Code_Used'}
  71                for part in PurePosixPath(relative).parts):
  72             return False
  73         return PurePosixPath(relative).suffix.lower() in RESULT_WEB_TYPES
  74     return False

  77 def local_target(page, link):
  78     """Resolve a relative HTML link using POSIX rules, also on Windows."""
  79     parsed = urlsplit(html.unescape(link))
  80     if parsed.scheme or parsed.netloc or not parsed.path:
  81         return None
  82     # Root-relative links break GitHub project-site prefixes: keep links relative.
  83     if parsed.path.startswith('/'):
  84         raise ValueError(f'Root-relative link in {page}: {link}')
  85     target = posixpath.normpath(posixpath.join(posixpath.dirname(page), unquote(parsed.path)))
  86     if target == '..' or target.startswith('../'):
  87         raise ValueError(f'Link escapes the project in {page}: {link}')
  88     return target

  91 def resolve_link(page, link, included, public, repository, revision):
  92     """Retain local pages and direct omitted public downloads to GitHub."""
  93     target = local_target(page, link)
  94     if target is None:
  95         return link
  96     parsed = urlsplit(html.unescape(link))
  97     suffix = ('?' + parsed.query if parsed.query else '') + ('#' + parsed.fragment if parsed.fragment else '')
  98     if target in included:
  99         return link
 100     # Directory URLs should resolve to explicit index pages on file:// and HTTP.
 101     index = target.rstrip('/') + '/index.html'
 102     if index in included:
 103         return posixpath.relpath(index, posixpath.dirname(page) or '.') + suffix
 104     if target in public:
 105         return f'https://github.com/{repository}/blob/{quote(revision, safe="")}/{quote(target, safe="/")}' + suffix
 106     if any(p.startswith(target.rstrip('/') + '/') for p in public):
 107         return f'https://github.com/{repository}/tree/{quote(revision, safe="")}/{quote(target, safe="/")}' + suffix
 108     raise ValueError(f'Missing or non-public linked file in {page}: {link}')

 111 def prepare_html(page, source, included, public, repository, revision):
 112     """Rewrite attributes inside real tags, leaving JavaScript and source listings intact."""
 113     class Rewrite(HTMLParser):
 114         def __init__(self):
 115             super().__init__(convert_charrefs=False)
 116             self.output = []
 117 
 118         def start_tag(self, tag):
 119             raw = self.get_starttag_text()
 120             def replace(match):
 121                 old = match['url']
 122                 new = resolve_link(page, old, included, public, repository, revision)
 123                 return match['key'] + '=' + match['quote'] + html.escape(html.unescape(new), quote=True) + match['quote']
 124             self.output.append(LINK_ATTRIBUTE.sub(replace, raw))
 125 
 126         handle_starttag = lambda self, tag, attrs: self.start_tag(tag)
 127         handle_startendtag = lambda self, tag, attrs: self.start_tag(tag)
 128         handle_endtag = lambda self, tag: self.output.append('</' + tag + '>')
 129         handle_data = lambda self, data: self.output.append(data)
 130         handle_entityref = lambda self, name: self.output.append('&' + name + ';')
 131         handle_charref = lambda self, name: self.output.append('&#' + name + ';')
 132         handle_comment = lambda self, data: self.output.append('<!--' + data + '-->')
 133         handle_decl = lambda self, decl: self.output.append('<!' + decl + '>')
 134         handle_pi = lambda self, data: self.output.append('<?' + data + '>')
 135 
 136     parser = Rewrite()
 137     parser.feed(source)
 138     parser.close()
 139     return ''.join(parser.output)

 142 def validate_website(output):
 143     """Fail the build if an HTML/SVG link, image, video or stylesheet is missing."""
 144     checked = 0
 145     for path in sorted(output.rglob('*')):
 146         if path.suffix.lower() not in {'.html', '.svg'}:
 147             continue
 148         page = path.relative_to(output).as_posix()
 149         class Collect(HTMLParser):
 150             def handle_starttag(self, tag, attrs):
 151                 nonlocal checked
 152                 for key, link in attrs:
 153                     if key not in {'href', 'src', 'data', 'poster', 'xlink:href'} or not link:
 154                         continue
 155                     target = local_target(page, link)
 156                     if target is not None:
 157                         checked += 1
 158                         if not (output / target).is_file():
 159                             raise ValueError(f'Broken website link: {page} -> {link}')
 160         parser = Collect()
 161         parser.feed(path.read_text(encoding='utf-8'))
 162     return checked

 165 def build_website(output, repository, revision, include_untracked=False):
 166     """Create a new static artifact; leave original files and simulation data intact."""
 167     output = Path(output).resolve()
 168     if output == ROOT or output in ROOT.parents:
 169         raise ValueError('The output must be a new folder, not the project or its parent.')
 170     if output.exists() and any(output.iterdir()):
 171         raise FileExistsError('Choose a new/empty output folder; existing output is not deleted.')
 172     if not re.fullmatch(r'[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+', repository):
 173         raise ValueError('Repository must be owner/name.')
 174     public = public_repository_files(ROOT, include_untracked)
 175     included = {name for name in public if belongs_in_website(name)}
 176     if not {'index.html', 'START_HERE.html'} <= included:
 177         raise ValueError('Missing entry pages. Commit the project first, or use --include-untracked for local preview.')
 178     output.mkdir(parents=True, exist_ok=True)
 179     for name in sorted(included):
 180         src, dst = ROOT / name, output / name
 181         dst.parent.mkdir(parents=True, exist_ok=True)
 182         if src.suffix.lower() in {'.html', '.svg'}:
 183             text = prepare_html(name, src.read_text(encoding='utf-8'), included, public, repository, revision)
 184             dst.write_text(text, encoding='utf-8')
 185         else:
 186             shutil.copyfile(src, dst)
 187     count = validate_website(output)
 188     size = sum((output / name).stat().st_size for name in included)
 189     if size > 900 * 1024 * 1024:
 190         raise ValueError('Website exceeds the 900 MiB packaging budget; keep large arrays in the repository.')
 191     audit = {'repository': repository, 'revision': revision, 'files': len(included),
 192              'bytes': size, 'checked_local_links': count, 'broken_links': [],
 193              'model_recomputed': False, 'raw_result_arrays_on_website': False,
 194              'private_papers_or_restricted_meshes_included': False}
 195     (output / 'WEBSITE_BUILD.json').write_text(json.dumps(audit, indent=2), encoding='utf-8')
 196     print(json.dumps(audit, indent=2))
 197     return audit
