  12 def emg_input_timing_scores(names, settings):
  13     pairs = emg_comparison_cycles(names, settings)
  14     T = settings["period_s"]
  15     threshold = settings.get("emg_phase_threshold", 0.10)
  16     if not settings.get("excitation_sequence_file"):
  17         return {
  18             p["channel"]: emg_phase_metrics(p["excitation"], p["reference"], T, threshold)
  19             for p in pairs
  20         }
  21     sequence = np.load(ROOT / settings["excitation_sequence_file"])["excitation_sequence"]
  22     names = list(names)
  23     peaks = np.maximum(0.02, sequence.max(axis=(0, 1)))
  24     scores = {}
  25     for pair in pairs:
  26         index = names.index(pair["channel"])
  27         phase = np.arange(len(pair["reference"])) / len(pair["reference"])
  28         rows = [
  29             emg_phase_metrics(
  30                 np.interp(phase, np.linspace(0, 1, len(table)), table[:, index])
  31                 # Use the intact peak so attenuation is still visible in the score.
  32                 / peaks[index]
  33 ,
  34                 pair["reference"],
  35                 T,
  36                 threshold,
  37             )
  38             for table in sequence
  39         ]
  40         summary = dict(rows[0])
  41         summary["active_intersection_over_union"] = min(
  42             r["active_intersection_over_union"]
  43             for r in rows
  44         )
  45         for key in [
  46             "phase_disagreement_fraction",
  47             "false_active_fraction_of_cycle",
  48             "missed_active_fraction_of_cycle",
  49         ]:
  50             summary[key] = max(r[key] for r in rows)
  51         for edge in ["onset", "offset"]:
  52             summary[edge] = {}
  53             for key in rows[0][edge]:
  54                 values = [r[edge][key] for r in rows if r[edge][key] is not None]
  55                 summary[edge][key] = max(values) if values else None
  56         for key in ["active_fraction_predicted", "predicted_burst_count"]:
  57             if key in rows[0]:
  58                 values = [r[key] for r in rows]
  59                 summary[key + "_range"] = [min(values), max(values)]
  60                 summary[key] = max(values)
  61         summary.update(
  62             scope="Worst timing agreement across every saved input stride; fixed whole-sequence peak normalization",
  63             per_stride=rows,
  64         )
  65         scores[pair["channel"]] = summary
  66     return scores
