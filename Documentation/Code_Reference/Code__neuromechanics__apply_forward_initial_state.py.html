  16 def apply_forward_initial_state(state, values):
  17     """Input keys: qpos, qvel, activation, fiber_length, bristle (SI units).
  18 
  19     Used only by initial setup or by restarting independent offline shooting
  20     trials. Calling it during a continuing simulation is not supported.
  21     """
  22     m, d = state["model"], state["data"]
  23     if d.time != 0:
  24         # Stop and explain the invalid model setting or failed calculation.
  25         raise ValueError("Initial state may only be applied at time zero")
  26     for key, size in [
  27         ("qpos", m.nq),
  28         ("qvel", m.nv),
  29         ("activation", m.na),
  30         ("fiber_length", m.nu),
  31     ]:
  32         if np.asarray(values[key]).shape != (size,) or not np.isfinite(values[key]).all():
  33             # Stop and explain the invalid model setting or failed calculation.
  34             raise ValueError("Invalid initial " + key)
  35     if (
  36         np.asarray(values["bristle"]).shape != state["bristle"].shape
  37         or not np.isfinite(values["bristle"]).all()
  38     ):
  39         # Stop and explain the invalid model setting or failed calculation.
  40         raise ValueError("Invalid initial contact memory")
  41     if np.any(np.asarray(values["activation"]) < 0) or np.any(
  42         np.asarray(values["activation"])
  43         > 1
  44     ):
  45         # Stop and explain the invalid model setting or failed calculation.
  46         raise ValueError("Initial activation must lie in [0,1]")
  47     if np.any(np.asarray(values["fiber_length"]) < 0.2 * state["lopt"]):
  48         # Stop and explain the invalid model setting or failed calculation.
  49         raise ValueError("Initial fiber length violates the muscle model lower bound")
  50     d.qpos[:] = values["qpos"]
  51     d.qvel[:] = values["qvel"]
  52     d.act[:] = values["activation"]
  53     state["fiber_length"] = np.array(values["fiber_length"], copy=True)
  54     state["bristle"] = np.array(values["bristle"], copy=True)
  55     update_muscle_paths(state)
  56     # Recalculate positions and routed muscle lengths at the new pose.
  57     mj.mj_fwdPosition(m, d)
  58     # Recalculate velocities at the new pose and generalized speed.
  59     mj.mj_fwdVelocity(m, d)
  60     mtu = d.actuator_length + state["mtu_length_offset"]
  61     force, velocity, residual = initial_elastic_force(
  62         mtu,
  63         state["fiber_length"],
  64         d.act,
  65         state["lopt"],
  66         state["slack"],
  67         state["muscle_params"],
  68         state["e0"],
  69         state["fiber_damping_coefficients"],
  70     )
  71     if not np.isfinite(force).all() or not np.isfinite(velocity).all():
  72         # Stop and explain the invalid model setting or failed calculation.
  73         raise ValueError("Initial muscle equilibrium is nonfinite")
  74     bounds = (
  75         (state["fiber_length"] <= 0.200001 * state["lopt"])
  76         | (mtu - state["fiber_length"] * state["cosine"] <= 1e-9)
  77     ).astype(np.int64)
  78     state.update(
  79         force=force,
  80         fiber_velocity=velocity,
  81         residual=residual,
  82         fiber_bound=bounds,
  83     )
  84     m.actuator_biasprm[:, 0] = -force
  85     d.ctrl[:] = muscle_excitation(state, 0.0)
  86     d.qfrc_applied[:] = 0.0
  87     d.xfrc_applied[:] = 0.0
  88     mj.mj_forward(m, d)
  89     compute_paw_contact_forces(state)
