   3 from pathlib import Path

   5 import numpy as np

   8 def read_time_series(path, expected_channels=None):
   9     """First column is phase (0..1); remaining headers are channel names.
  10 
  11     A complete cycle includes both endpoints. No silent clipping, column
  12     reordering, phase shift, filtering or extrapolation occurs here.
  13     """
  14     path = Path(path)
  15     headers = path.read_text(encoding="utf-8").splitlines()[0].split("\t")
  16     values = np.loadtxt(path, delimiter="\t", skiprows=1, ndmin=2)
  17     if headers[0] != "phase_0_to_1" or len(headers) != values.shape[1]:
  18         # Stop and explain the invalid model setting or failed calculation.
  19         raise ValueError(f"{path}: invalid column headers")
  20     if len(set(headers)) != len(headers) or not np.isfinite(values).all():
  21         # Stop and explain the invalid model setting or failed calculation.
  22         raise ValueError(f"{path}: duplicate names or nonfinite values")
  23     phase = values[:, 0]
  24     if len(phase) < 2 or phase[0] != 0 or phase[-1] != 1 or np.any(np.diff(phase) <= 0):
  25         # Stop and explain the invalid model setting or failed calculation.
  26         raise ValueError(f"{path}: phase must increase strictly from 0 to 1")
  27     if expected_channels is not None and headers[1:] != list(expected_channels):
  28         # Stop and explain the invalid model setting or failed calculation.
  29         raise ValueError(f"{path}: channel names/order differ from the model")
  30     return phase, headers[1:], values[:, 1:]
