   3 import numpy as np
   4 
   5 from numba import njit
   6 
   7 
   8 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
   9 @njit(cache=True)
  10 def pad_force_law(depth, velocity, bristle, dt, parameters):
  11     """Hunt--Crossley normal force and elastic Coulomb return mapping.
  12 
  13     depth: positive compression [m]; velocity: surface velocity [m/s].
  14     bristle: tangential elastic displacement [m]. dt advances that state.
  15     k: N/m^(3/2), dissipation: s/m, tangential stiffness: N/m,
  16     tangential damping: N*s/m. All parameters are explicitly calibrated,
  17     not claimed as measurements of this animal's pads.
  18     """
  19     k, c, kt, bt, mu = parameters
  20     if depth <= 0:
  21         return np.zeros(3), np.zeros(2), 0.0
  22     normal = k * depth**1.5 * max(0.0, 1.0 - c * velocity[2])
  23     if normal <= 0:
  24         return np.zeros(3), np.zeros(2), k * depth**2.5 / 2.5
  25     displacement = bristle + dt * velocity[:2]
  26     # The dashpot fades with compression; forces approach zero at contact onset.
  27     damping = bt * min(1.0, depth / 0.001)
  28     tangent = -kt * displacement - damping * velocity[:2]
  29     magnitude = np.linalg.norm(tangent)
  30     limit = mu * normal
  31     if magnitude > limit:
  32         tangent *= limit / max(magnitude, 1.0e-30)
  33         displacement = -(tangent + damping * velocity[:2]) / kt
  34     force = np.empty(3)
  35     force[:2] = tangent
  36     force[2] = normal
  37     return (
  38         force,
  39         displacement,
  40         (k * depth**2.5 / 2.5 + 0.5 * kt * np.dot(displacement, displacement)),
  41     )
