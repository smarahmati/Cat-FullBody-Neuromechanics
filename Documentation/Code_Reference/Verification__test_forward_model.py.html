  35     def test_muscle_force_name_preserves_recorded_parameter_values(self):
  36         # The explicit muscle-force name must preserve every recorded numeric value.
  37         current = load_parameter_file(ROOT/'Data/Model_Parameters/muscle_tendon.py')
  38         recorded = load_parameter_file(ROOT/'Simulation_Results/Intact_Walking/Model_Parameters_Used/muscle_tendon.py')
  39         self.assertEqual(current, recorded)
  40         field = 'max_isometric_muscle_force_along_tendon_N'
  41         # Check the force capacity actually used by the solver, without a second cosine.
  42         expected = [current[name][field] for name in self.state['names']]
  43         np.testing.assert_array_equal(self.state['muscle_params'][:,2], expected)

  45     def test_published_passive_curve_and_derivative(self):
  46         # Independent formula: Prilutsky 2016 Table 10.3, tension-only adaptation.
  47         p=self.state['muscle_params'][list(self.state['names']).index('R_VL')]
  48         for length in [0.6,1.0,1.1,1.2,1.5]:
  49             expected=max(0.,0.0075/11.6*np.expm1(11.6*(length-1)))
  50             self.assertAlmostEqual(muscle_force_length_velocity(length,0.,p)[2],expected,places=13)
  51         length=1.2
  52         h=1e-6
  53         numerical=(muscle_force_length_velocity(length+h,0,p)[2]-muscle_force_length_velocity(length-h,0,p)[2])/(2*h)
  54         self.assertAlmostEqual(muscle_force_curve_derivatives(length,0,p)[5],numerical,places=9)

  56     def test_unactivated_stretched_fiber_transmits_passive_tendon_force(self):
  57         # Construct a static series equilibrium: a=0, fiber velocity=0.
  58         # Tendon and parallel elastic tissue must carry the same positive force.
  59         s=self.state
  60         i=list(s['names']).index('R_VL')
  61         p=s['muscle_params'][i:i+1]
  62         length=1.2*s['lopt'][i]
  63         passive=0.0075/11.6*np.expm1(11.6*0.2)
  64         low,high=0.,float(s['e0'][i])
  65         for _ in range(70):
  66             middle=(low+high)/2
  67             if elastic_tendon_force(middle,s['e0'][i])<passive:low=middle
  68             else:high=middle
  69         total=length*p[0,9]+s['slack'][i]*(1+(low+high)/2)
  70         lf,v,force,residual,bound=solve_elastic_muscle_equilibrium(
  71             np.array([total]),np.array([length]),np.zeros(1),s['lopt'][i:i+1],
  72             s['slack'][i:i+1],p,s['e0'][i:i+1],np.array([0.05]),0.000125)
  73         self.assertGreater(force[0],0.)
  74         self.assertAlmostEqual(force[0],p[0,2]*passive,places=8)
  75         self.assertAlmostEqual(v[0],0.,places=8)
  76         self.assertLess(abs(residual[0]),1e-7)
  77         self.assertEqual(bound[0],0)

  79     def test_tendon_calibration_and_continuity(self):
  80         e0=0.04
  81         toe=elastic_tendon_transition(e0)
  82         self.assertEqual(elastic_tendon_force(-0.01,e0),0)
  83         self.assertAlmostEqual(elastic_tendon_force(e0,e0),1,places=12)
  84         self.assertAlmostEqual(elastic_tendon_force(toe,e0),0.33,places=12)
  85         h=1e-7
  86         left=(elastic_tendon_force(toe,e0)-elastic_tendon_force(toe-h,e0))/h
  87         right=(elastic_tendon_force(toe+h,e0)-elastic_tendon_force(toe,e0))/h
  88         self.assertAlmostEqual(left,right,delta=0.001)

  90     def test_input_repeats_without_mechanical_feedback(self):
  91         state = self.state
  92         period = state['settings']['period_s']
  93         initial_input = muscle_excitation(state, 0.17)
  94         np.testing.assert_allclose(initial_input, muscle_excitation(state, 0.17 + period), atol=1e-12)
  95         # Changing a recorded force must not change this prescribed motor input.
  96         old_force = state['grf'].copy()
  97         state['grf'][:] = 100.0
  98         np.testing.assert_array_equal(initial_input, muscle_excitation(state, 0.17))
  99         state['grf'][:] = old_force

 101     def test_contact_cannot_pull_and_obeys_friction_limit(self):
 102         p=self.state['contact_parameters'][0]
 103         f,_,_=viscoelastic_paw_pad(0,np.array([1.,0.,0.]),np.zeros(2),0.000125,p)
 104         np.testing.assert_array_equal(f,np.zeros(3))
 105         f,_,_=viscoelastic_paw_pad(0.002,np.array([1.,0.,-0.1]),np.zeros(2),0.000125,p)
 106         self.assertGreaterEqual(f[2],0)
 107         self.assertLessEqual(np.linalg.norm(f[:2]),p[4]*f[2]+1e-12)
 108         self.assertLess(f[0],0)

 110     def test_edited_mass_is_actually_applied(self):
 111         with tempfile.TemporaryDirectory(prefix='cat_parameter_test_') as temp:
 112             folder=Path(temp)/'parameters'
 113             shutil.copytree(ROOT/'Data/Model_Parameters',folder,ignore=shutil.ignore_patterns('*.html'))
 114             data=load_parameter_file(folder/'bodies.py')
 115             key=next(iter(data))
 116             old=float(data[key]['mass_kg'])
 117             data[key]['mass_kg']=old*1.01
 118             (folder/'bodies.py').write_text('PARAMETERS = '+repr(data),encoding='utf-8')
 119             s=initialize_neuromechanical_model(load_model_configuration(folder))
 120             self.assertAlmostEqual(s['model'].body(key).mass[0],old*1.01,places=12)

 122     def test_short_forward_reproduces_release(self):
 123         s=initialize_neuromechanical_model(self.cfg)
 124         result,_=simulate_forward_dynamics({'seconds':0.05},state=s)
 125         with np.load(ROOT/'Simulation_Results/Intact_Walking/forward_simulation_timeseries.npz') as baseline:
