  14 def precondition_periodic_muscles(s):
  15     """Reference kinematics apply only to the negative-time muscle prehistory.
  16 
  17     No joint pose, velocity or contact state is reset after t=0. The cached
  18     reference lengths depend on fixed geometry/reference, never on muscle_excitation.
  19     In contrast, activation and fiber states are recomputed for every input.
  20     """
  21     m, d = s["model"], s["data"]
  22     q = d.qpos.copy()
  23     v0 = d.qvel.copy()
  24     cfg = s["settings"]
  25     n = cfg["prehistory_samples"]
  26     offsets = cfg.get("initial_joint_angle_offsets_rad", {})
  27     cache_key = tuple(sorted(offsets.items()))
  28     if "prehistory_mtu" not in s or s.get("prehistory_geometry_key", ()) != cache_key:
  29         reference = assemble_reference_kinematics(s, n)
  30         length = []
  31         for row in reference["pose"]:
  32             d.qpos[:] = row
  33             for name, value in offsets.items():
  34                 d.qpos[m.joint(name).qposadr[0]] += value
  35             update_muscle_paths(s)
  36             # Recalculate positions and routed muscle lengths at the new pose.
  37             mj.mj_fwdPosition(m, d)
  38             length.append(d.actuator_length + s["mtu_length_offset"])
  39         s["prehistory_mtu"] = np.asarray(length)
  40         s["prehistory_geometry_key"] = cache_key
  41     a, lf, v, f, r, b, da, dl = periodic_muscle_history_kernel(
  42         s["prehistory_mtu"],
  43         s["excitation_table"],
  44         s["lopt"],
  45         s["slack"],
  46         s["muscle_params"],
  47         s["e0"],
  48         s["fiber_damping_coefficients"],
  49         s["time_constants"],
  50         cfg["period_s"],
  51         cfg["prehistory_cycles"],
  52     )
  53     d.qpos[:] = q
  54     d.qvel[:] = v0
  55     d.act[:] = a
  56     s.update(
  57         fiber_length=lf,
  58         fiber_velocity=v,
  59         force=f,
  60         residual=r,
  61         fiber_bound=b,
  62         prehistory_activation_repeat_error=da,
  63         prehistory_fiber_repeat_error=dl,
  64     )
  65     m.actuator_biasprm[:, 0] = -f
  66     update_muscle_paths(s)
  67     mj.mj_forward(m, d)
