   1 """Advance legacy timestep.
   2 
   3 Inputs/outputs: see the named function arguments and returned values.
   4 Lengths use metres, time seconds, force newtons, and angles radians unless stated.
   5 See Documentation/model_methods.html for equations and parameter definitions.
   6 """
   7 
   8 import numpy as np
   9 
  10 from model_configuration.numerical_parameters import MAXIMUM_GENERALIZED_SPEED
  11 
  12 import mujoco as mj
  13 
  14 from .muscle_excitation import muscle_excitation
  15 
  16 from .update_muscle_paths import update_muscle_paths
  17 
  18 from .compute_paw_contact_forces import compute_paw_contact_forces
  19 
  20 from .solve_elastic_muscle_by_bisection import solve_elastic_muscle_by_bisection
  21 
  22 
  23 def advance_partitioned_timestep(s, offline_motor_command=None):
  24     """Excitation -> activation -> elastic force -> motion -> physical contact.
  25 
  26     No reference angles, force templates, support controller, inverse dynamics,
  27     body-position reset, or externally commanded trunk wrench is used here.
  28     """
  29     m, d = s["model"], s["data"]
  30     d.ctrl[:] = (
  31         muscle_excitation(s, d.time)
  32         if offline_motor_command is None
  33         else offline_motor_command
  34     )
  35     d.qfrc_applied[:] = 0.0
  36     d.xfrc_applied[:] = s["contact_wrench"]
  37     # Integrate the common skeleton and native activation by one mechanical timestep.
  38     mj.mj_step(m, d)
  39     update_muscle_paths(s)
  40     mj.mj_forward(m, d)
  41     lf, v, f, r, b = solve_elastic_muscle_by_bisection(
  42         d.actuator_length + s["mtu_length_offset"],
  43         s["fiber_length"],
  44         d.act,
  45         s["lopt"],
  46         s["slack"],
  47         s["muscle_params"],
  48         s["e0"],
  49         s["fiber_damping_coefficients"],
  50         m.opt.timestep,
  51     )
  52     s.update(fiber_length=lf, fiber_velocity=v, force=f, residual=r, fiber_bound=b)
  53     m.actuator_biasprm[:, 0] = -f
  54     compute_paw_contact_forces(s, m.opt.timestep)
  55     mj.mj_forward(m, d)
  56     return bool(
  57         np.isfinite(d.qpos).all()
  58         and np.isfinite(f).all()
  59         and np.max(abs(d.qvel)) < MAXIMUM_GENERALIZED_SPEED
  60     )
