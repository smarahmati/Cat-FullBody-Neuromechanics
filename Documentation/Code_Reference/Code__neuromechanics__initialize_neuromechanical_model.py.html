  25 def initialize_neuromechanical_model(overrides=None, graphics=False):
  26     """Set initial state once. Subsequent stepping never prescribes motion."""
  27     settings = dict(SETTINGS)
  28     settings.update(overrides or {})
  29     directory = (
  30         ROOT / "Data/Skeletal_Geometry/Model"
  31         if settings["model_variant"] == "base"
  32         else ROOT / "Data/Skeletal_Geometry/Model" / settings["model_variant"]
  33     )
  34     state = dict(np.load(directory / "muscle_parameters.npz"))
  35     model = load_skeletal_anatomy(graphics, settings["model_variant"])
  36     model.opt.timestep = settings["dt_s"]
  37     data = mj.MjData(model)
  38     state.update(model=model, data=data, settings=settings)
  39     if settings["activation_parameter_mode"] == "source":
  40         model.actuator_dynprm[:, :2] = state["time_constants"]
  41     elif settings["activation_parameter_mode"] != "legacy":
  42         # Stop and explain the invalid model setting or failed calculation.
  43         raise ValueError("Unknown activation parameter mode")
  44     if settings["excitation_file"]:
  45         state["excitation_table"] = np.load(ROOT / settings["excitation_file"])[
  46             "excitation_table"
  47         ]
  48     if settings["excitation_sequence_file"]:
  49         state["excitation_sequence"] = np.load(ROOT / settings["excitation_sequence_file"])[
  50             "excitation_sequence"
  51         ]
  52         seq = state["excitation_sequence"]
  53         if (
  54             seq.ndim != 3
  55             or seq.shape[2] != model.nu
  56             or not np.isfinite(seq).all()
  57             or np.any(seq < 0)
  58             or np.any(seq > 1)
  59         ):
  60             # Stop and explain the invalid model setting or failed calculation.
  61             raise ValueError("Invalid frozen muscle_excitation sequence")
  62         if len(seq) > 1 and not np.allclose(seq[:-1, -1], seq[1:, 0], atol=1e-12):
  63             # Stop and explain the invalid model setting or failed calculation.
  64             raise ValueError("Frozen sequence has discontinuous stride boundaries")
  65     if "excitation_control_points" in settings:
  66         state["excitation_table"] = np.asarray(settings["excitation_control_points"], float)
  67     if settings.get("excitation_table_file"):
  68         from .read_time_series import read_time_series
  69 
  70         phase, channels, table = read_time_series(
  71             ROOT / settings["excitation_table_file"],
  72             state["names"],
  73         )
  74         if np.any(table < 0) or np.any(table > 1):
  75             # Stop and explain the invalid model setting or failed calculation.
  76             raise ValueError("Excitation table must stay within [0,1]")
  77         if not np.allclose(phase, np.linspace(0, 1, len(phase)), atol=1e-12, rtol=0):
  78             # Stop and explain the invalid model setting or failed calculation.
  79             raise ValueError("Excitation phase grid must be uniform")
  80         state["excitation_table"] = table
  81     if state["excitation_table"].shape[1] != model.nu:
  82         # Stop and explain the invalid model setting or failed calculation.
  83         raise ValueError("Excitation channel count mismatch")
  84     if not np.allclose(state["excitation_table"][0], state["excitation_table"][-1]):
  85         # Stop and explain the invalid model setting or failed calculation.
  86         raise ValueError("Nonperiodic muscle_excitation endpoints")
  87     state["muscle_params"][28:, 2] *= settings["fore_strength_scale"]
  88     state["slack"][28:] *= settings["fore_slack_scale"]
  89     state["muscle_params"][46:88, 2] *= settings["girdle_strength_scale"]
  90     for name, scale in settings["muscle_force_scales"].items():
  91         if scale <= 0:
  92             # Stop and explain the invalid model setting or failed calculation.
  93             raise ValueError("Muscle force scale must be positive")
  94         state["muscle_params"][list(state["names"]).index(name), 2] *= scale
  95     state["fiber_damping_coefficients"][28:] = settings["fore_fiber_damping"]
  96     for name, values in settings["girdle_length_overrides"].items():
  97         i = list(state["names"]).index(name)
  98         state["lopt"][i], state["slack"][i] = values
  99     if settings["fore_passive_fiber"]:
 100         state["muscle_params"][28:, 13] = 1.0
 101     modes = [settings["hind_force_velocity_mode"], settings["fore_force_velocity_mode"]]
 102     if any(mode not in ("source", "degroote_2016") for mode in modes):
 103         # Stop and explain the invalid model setting or failed calculation.
 104         raise ValueError("Unknown muscle force-velocity model")
 105     if "degroote_2016" in modes:
 106         state["muscle_params"] = np.pad(state["muscle_params"], ((0, 0), (0, 1)))
 107         if modes[0] == "degroote_2016":
 108             state["muscle_params"][:28, 14] = 1.0
 109         if modes[1] == "degroote_2016":
 110             state["muscle_params"][28:, 14] = 1.0
 111     if settings["fore_damping_mode"] == "source_velocity":
 112         state["fiber_damping_coefficients"][28:] = (
 113             0.02
 114             * state["lopt"][28:]
 115             * state["muscle_params"][28:, 6]
 116         )
 117     elif settings["fore_damping_mode"] != "normalized":
 118         # Stop and explain the invalid model setting or failed calculation.
 119         raise ValueError("Unknown fore fiber damping mode")
 120     for name, limits in settings["joint_limit_overrides"].items():
 121         model.jnt_range[model.joint(name).id] = limits
 122     for side in ["FL", "FR"]:
 123         model.body_pos[model.body(side + "_scapula").id] += settings["fore_mount_offset_m"]
 124     model.body_ipos[model.body("cat_root").id] += settings["trunk_com_offset_m"]
 125     j = mj.mj_name2id(model, mj.mjtObj.mjOBJ_JOINT, "lumbosacral_pitch")
 126     if j >= 0:
 127         k, rest, damping = settings["spine_passive"]
 128         model.jnt_stiffness[j] = k
 129         model.qpos_spring[model.jnt_qposadr[j]] = rest
 130         model.dof_damping[model.jnt_dofadr[j]] = damping
 131     for side in ["FL", "FR"]:
 132         for axis in ["x", "z"]:
 133             j = mj.mj_name2id(model, mj.mjtObj.mjOBJ_JOINT, side + "_scapula_" + axis)
 134             if j >= 0:
 135                 model.jnt_stiffness[j] = settings["scapula_translation_passive"][0]
 136                 model.dof_damping[model.jnt_dofadr[j]] = settings[
 137                     "scapula_translation_passive"
 138                 ][1]
 139     for name, values in settings["passive_joint_overrides"].items():
 140         j = model.joint(name).id
 141         if any(key not in ("stiffness", "damping", "springref") for key in values):
 142             # Stop and explain the invalid model setting or failed calculation.
 143             raise ValueError("Unknown passive joint parameter")
 144         if any(not np.isfinite(value) for value in values.values()):
 145             # Stop and explain the invalid model setting or failed calculation.
 146             raise ValueError("Nonfinite passive joint parameter")
 147         if any(values.get(key, 0) < 0 for key in ["stiffness", "damping"]):
 148             # Stop and explain the invalid model setting or failed calculation.
 149             raise ValueError("Passive stiffness and damping must be nonnegative")
 150         if "stiffness" in values:
 151             model.jnt_stiffness[j] = values["stiffness"]
 152         if "damping" in values:
 153             model.dof_damping[model.jnt_dofadr[j]] = values["damping"]
 154         if "springref" in values:
 155             model.qpos_spring[model.jnt_qposadr[j]] = values["springref"]
 156     state["joint_indices"] = np.array(
 157         [
 158             [
 159                 model.joint(side + "_" + j).qposadr[0]
 160                 for j in (
 161                     ["hip", "knee", "ankle", "mtp"]
 162                     if side in ["L", "R"]
 163                     else ["shoulder", "elbow", "wrist", "mcp"]
 164                 )
 165             ]
 166             for side in ["L", "R", "FL", "FR"]
 167         ]
 168     )
 169     state["paw_ids"] = np.array(
 170         [model.site(side + "_paw").id for side in ["L", "R", "FL", "FR"]]
 171     )
 172     state["pad_ids"] = np.array(
 173         [
 174             i
 175             for i in range(model.ngeom)
 176             if "_paw_pad" in model.geom(i).name or "_contactpad" in model.geom(i).name
 177         ]
 178     )
 179     state["pad_side"] = np.array(
 180         [
 181             ["L", "R", "FL", "FR"].index(model.geom(i).name.split("_")[0])
 182             for i in state["pad_ids"]
 183         ]
 184     )
 185     model.geom_contype[state["pad_ids"]] = 0
 186     model.geom_conaffinity[state["pad_ids"]] = 0
 187     model.geom_size[state["pad_ids"][state["pad_side"] >= 2], 0] += settings[
 188         "fore_pad_radius_offset_m"
 189     ]
 190     if np.any(model.geom_size[state["pad_ids"], 0] <= 0):
 191         # Stop and explain the invalid model setting or failed calculation.
 192         raise ValueError("Nonpositive paw pad radius")
 193     state["bristle"] = np.zeros((len(state["pad_ids"]), 2))
 194     state["bristle"][: len(state["hind_bristle"])] = state["hind_bristle"]
 195     if settings["initialize_contact_memory"] and settings["reference_seed"]:
 196         saved_seed = np.load(ROOT / settings["reference_seed"])
 197         if "initial_bristle" not in saved_seed:
 198             # Stop and explain the invalid model setting or failed calculation.
 199             raise ValueError("The selected reference has no saved contact memory")
 200         state["bristle"][:] = saved_seed["initial_bristle"]
 201     # Fore contact initially inherits the resolved hind pad material law.
 202     # This is an explicit cross-paw assumption; four fitted multipliers above.
 203     state["contact_parameters"] = np.tile(
 204         np.mean(state["hind_contact_parameters"], axis=0),
 205         (len(state["pad_ids"]), 1),
 206     )
 207     n = len(state["hind_contact_parameters"])
 208     state["contact_parameters"][:n] = state["hind_contact_parameters"]
 209     state["contact_parameters"][n:, :4] *= settings["fore_contact_scale"]
 210     material_scale = np.asarray(settings["contact_material_scale"], float)
 211     if (
 212         material_scale.shape != (5,)
 213         or not np.isfinite(material_scale).all()
 214         or np.any(material_scale <= 0)
 215     ):
 216         # Stop and explain the invalid model setting or failed calculation.
 217         raise ValueError("Contact material multipliers must be five finite positive values")
 218     state["contact_parameters"][:, :5] *= material_scale
 219     if settings.get("parameter_bundle_dir"):
 220         from .apply_neuromechanical_parameters import apply_neuromechanical_parameters
 221 
 222         apply_neuromechanical_parameters(state, settings["parameter_bundle_dir"])
 223     mj.mj_setConst(model, data)
 224     data.qpos[:] = state["qpos"]
 225     data.qvel[:] = state["qvel"]
 226     data.act[:] = state["activation"]
 227     data.ctrl[:] = muscle_excitation(state, 0.0)
 228     if settings["initialize_from_assembled_reference"]:
 229         from .assemble_reference_kinematics import assemble_reference_kinematics
 230 
 231         ref = assemble_reference_kinematics(state)
 232         data.qpos[:] = ref["pose"][0]
 233         data.qvel[:] = ref["velocity"][0]
 234     data.qpos[1] += settings["initial_height_offset_m"]
 235     data.qpos[2] += settings["initial_pitch_offset_rad"]
 236     data.qvel[:3] += settings["initial_root_velocity_offset"]
 237     for name, value in settings["initial_joint_angle_offsets_rad"].items():
 238         data.qpos[model.joint(name).qposadr[0]] += value
 239     for name, value in settings["initial_joint_velocity_offsets_rad_s"].items():
 240         data.qvel[model.joint(name).dofadr[0]] += value
 241     update_muscle_paths(state)
 242     mj.mj_forward(model, data)
 243     if settings.get("initial_parameter_file"):
 244         from .apply_initial_parameter_file import apply_initial_parameter_file
 245 
 246         apply_initial_parameter_file(state, ROOT / settings["initial_parameter_file"])
 247     elif settings.get("forward_initial_state_file"):
 248         # An explicit fitted state already supplies activation, fibers and
 249         # contact memory. Computing a negative-time history and immediately
 250         # discarding it would not change this state, only initialization cost.
 251         from .apply_forward_initial_state import apply_forward_initial_state
 252 
 253         apply_forward_initial_state(
 254             state,
 255             dict(np.load(ROOT / settings["forward_initial_state_file"])),
 256         )
 257     elif settings["periodic_muscle_initialization"]:
 258         from .precondition_periodic_muscles import precondition_periodic_muscles
 259 
 260         precondition_periodic_muscles(state)
 261     elif settings["precondition_fibers"]:
 262         from .precondition_fibers import precondition_fibers
 263 
 264         runtime_table = state["excitation_table"]
 265         if settings["initialization_excitation_file"]:
 266             state["excitation_table"] = np.load(
 267                 ROOT
 268                 / settings["initialization_excitation_file"]
 269             )["excitation_table"]
 270         precondition_fibers(state)
 271         state["excitation_table"] = runtime_table
 272         data.ctrl[:] = muscle_excitation(state, 0.0)
 273     else:
 274         # Use preserved hind initial fibers; solve fore static equilibrium only.
 275         mt = data.actuator_length + state["mtu_length_offset"]
 276         lf, v, f, r, b = solve_elastic_muscle_by_bisection(
 277             mt,
 278             state["fiber_length"],
 279             data.act,
 280             state["lopt"],
 281             state["slack"],
 282             state["muscle_params"],
 283             state["e0"],
 284             state["fiber_damping_coefficients"],
 285             1e6,
 286         )
 287         state["fiber_length"][28:] = lf[28:]
 288         state["force"][28:] = f[28:]
 289     model.actuator_biasprm[:, 0] = -state["force"]
 290     mj.mj_forward(model, data)
 291     compute_paw_contact_forces(state)
 292     # Check that the model still satisfies the required numerical assumption.
 293     assert model.neq == 0 and model.nv in [21, 25, 26] and model.nu == len(state["names"])
 294     return state
