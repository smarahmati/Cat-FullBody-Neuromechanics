   1 """Experimental comparison targets.
   2 
   3 Inputs/outputs: see the named function arguments and returned values.
   4 Lengths use metres, time seconds, force newtons, and angles radians unless stated.
   5 See Documentation/model_methods.html for equations and parameter definitions.
   6 """
   7 
   8 import numpy as np
   9 
  10 from model_configuration.paths import ROOT
  11 
  12 from .periodic_interpolate import periodic_interpolate
  13 
  14 
  15 def experimental_comparison_targets(times, period_s=0.6833333333333333, reference_dir=None):
  16     """Side-specific reference clock; only RH is a simultaneous measured trial.
  17 
  18     LH mirrors RH with the saved geometric consistency adjustment. Forelimb
  19     angles/Ground_Reaction_Forces have separate Left/Right source curves, but EMGs are pooled.
  20     These arrays score a run; they never enter the forward mechanical step.
  21     """
  22     from .load_reference_tables import load_reference_tables
  23 
  24     z, correction, _ = load_reference_tables(reference_dir)
  25     q = []
  26     f = []
  27     u = []
  28     scap = []
  29     for t in times:
  30         phase = t / period_s + 0.5 + np.array([0.5, 0.0, 0.25, 0.75])
  31         angles = np.array(
  32             [periodic_interpolate(z["joint_q"][:, s], phase[s]) for s in range(4)]
  33         )
  34         angles[0] += periodic_interpolate(correction, t / period_s)
  35         q.append(angles)
  36         f.append([periodic_interpolate(z["grf"][:, s], phase[s]) for s in range(4)])
  37         u.append(
  38             np.r_[
  39                 periodic_interpolate(z["excitation"][:, :11], phase[0]),
  40                 periodic_interpolate(z["excitation"][:, 11:22], phase[1]),
  41                 periodic_interpolate(z["excitation"][:, 22:31], phase[2]),
  42                 periodic_interpolate(z["excitation"][:, 31:40], phase[3]),
  43             ]
  44         )
  45         scap.append(
  46             [
  47                 periodic_interpolate(z["scapula_angle_by_side"][:, s], phase[s + 2])
  48                 - np.mean(z["scapula_angle"])
  49                 for s in range(2)
  50             ]
  51         )
  52     return dict(
  53         joint_q=np.array(q),
  54         grf=np.array(f),
  55         excitation=np.array(u),
  56         scapula=np.array(scap),
  57     )
