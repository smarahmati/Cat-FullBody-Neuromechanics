   6 def activation_dynamics(
   7     muscle_excitation,
   8     activation,
   9     activation_time_s,
  10     deactivation_time_s,
  11 ):
  12     """Return da/dt in 1/s; this diagnostic does not integrate a second state.
  13 
  14     The production integrator advances MuJoCo's native activation state.
  15     The same time constants are supplied by activation.py. This expression
  16     documents and tests the zero-switch-smoothing branch used by this model.
  17     """
  18     # Keep the incoming neural command between no excitation (0) and full excitation (1).
  19     u = np.clip(muscle_excitation, 0.0, 1.0)
  20     # Keep the current active state within its physical range, from 0 to 1.
  21     a = np.clip(activation, 0.0, 1.0)
  22     # Let the response time depend on the present activation, as in the native muscle law.
  23     factor = 0.5 + 1.5 * a
  24     # Use activation timing when the command rises and deactivation timing when it falls.
  25     tau = np.where(u > a, activation_time_s * factor, deactivation_time_s / factor)
  26     # The difference between command and activation determines how quickly activation changes.
  27     return (u - a) / tau
