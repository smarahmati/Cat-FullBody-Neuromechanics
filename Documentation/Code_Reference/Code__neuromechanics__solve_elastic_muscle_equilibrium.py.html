   1 """Safeguarded Newton solve of the SAME backward-Euler elastic equation.
   2 
   3 This is a numerical acceleration, not a new muscle or tendon law. A bracket
   4 and bisection fallback protect the solve at piecewise-curve transitions.
   5 """
   6 
   7 import numpy as np
   8 
   9 from numba import njit
  10 
  11 from .muscle_force_curve_derivatives import muscle_force_curve_derivatives
  12 
  13 from .elastic_tendon_force import elastic_tendon_force
  14 
  15 from model_configuration.formulation_constants import TENDON_TOE_CURVATURE
  16 
  17 from model_configuration.numerical_parameters import (
  18     MINIMUM_FIBER_LENGTH_RATIO,
  19     MINIMUM_LENGTH_GAP_M,
  20     MAXIMUM_FIBER_ITERATIONS,
  21     FIBER_FORCE_TOLERANCE,
  22     FIBER_LENGTH_TOLERANCE_M,
  23     NEWTON_FALLBACK_ITERATION,
  24 )
  25 
  26 from .elastic_tendon_transition import elastic_tendon_transition

  29 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  30 @njit(cache=False)
  31 def solve_elastic_muscle_equilibrium(
  32     muscle_tendon_length_m,
  33     previous_fiber_length_m,
  34     activation,
  35     optimal_fiber_length_m,
  36     tendon_slack_length_m,
  37     muscle_parameters,
  38     strain_at_maximum_force,
  39     fiber_damping,
  40     step_s,
  41 ):
  42     muscle_count = len(muscle_tendon_length_m)
  43     length = np.empty(muscle_count)
  44     velocity = np.empty(muscle_count)
  45     tension = np.empty(muscle_count)
  46     residual = np.empty(muscle_count)
  47     bound = np.zeros(muscle_count, np.int64)
  48     damping = np.asarray(fiber_damping).reshape(-1)
  49     for i in range(muscle_count):
  50         coefficient = damping[i] if len(damping) > 1 else damping[0]
  51         cosine = muscle_parameters[i, 9] if muscle_parameters.shape[1] > 9 else 1.0
  52         lower = MINIMUM_FIBER_LENGTH_RATIO * optimal_fiber_length_m[i]
  53         upper = max(lower + MINIMUM_LENGTH_GAP_M, muscle_tendon_length_m[i] / cosine)
  54         low, high = lower, upper
  55         maximum_fiber_speed_m_s = optimal_fiber_length_m[i] * muscle_parameters[i, 6]
  56         fiber_length_m = min(high, max(low, previous_fiber_length_m[i]))
  57         curvature = TENDON_TOE_CURVATURE
  58         toe = elastic_tendon_transition(strain_at_maximum_force[i])
  59         for iteration in range(MAXIMUM_FIBER_ITERATIONS):
  60             normalized_velocity = (fiber_length_m - previous_fiber_length_m[i]) / (
  61                 step_s
  62                 * maximum_fiber_speed_m_s
  63             )
  64             active_length_factor, velocity_factor, passive_length_factor, dfl, dfv, dfp = (
  65                 muscle_force_curve_derivatives(
  66                     fiber_length_m / optimal_fiber_length_m[i],
  67                     normalized_velocity,
  68                     muscle_parameters[i],
  69                 )
  70             )
  71             strain = (
  72                 muscle_tendon_length_m[i]
  73                 - fiber_length_m * cosine
  74             ) / tendon_slack_length_m[i] - 1
  75             normalized_tendon_force = elastic_tendon_force(
  76                 strain,
  77                 strain_at_maximum_force[i],
  78             )
  79             error = (
  80                 normalized_tendon_force
  81                 - activation[i] * active_length_factor * velocity_factor
  82                 - passive_length_factor
  83                 - coefficient * normalized_velocity
  84             )
  85             if abs(error) < FIBER_FORCE_TOLERANCE:
  86                 # Leave this loop because the convergence or stopping condition was met.
  87                 break
  88             if error > 0:
  89                 low = fiber_length_m
  90             else:
  91                 high = fiber_length_m
  92             if high - low < FIBER_LENGTH_TOLERANCE_M:
  93                 fiber_length_m = 0.5 * (low + high)
  94                 # Leave this loop because the convergence or stopping condition was met.
  95                 break
  96             dft = (
  97                 0.0
  98                 if strain <= 0
  99                 else (
 100                     0.33
 101                     * curvature
 102                     / toe
 103                     * np.exp(curvature * strain / toe)
 104                     / np.expm1(curvature)
 105                     if strain <= toe
 106                     else 0.67 / (strain_at_maximum_force[i] - toe)
 107                 )
 108             )
 109             derivative = (
 110                 -dft * cosine / tendon_slack_length_m[i]
 111                 - (activation[i] * dfl * velocity_factor + dfp) / optimal_fiber_length_m[i]
 112                 - (activation[i] * active_length_factor * dfv + coefficient)
 113                 / (step_s * maximum_fiber_speed_m_s)
 114             )
 115             candidate = (
 116                 fiber_length_m - error / derivative
 117                 if derivative < 0
 118                 else 0.5 * (low + high)
 119             )
 120             if (
 121                 candidate <= low
 122                 or candidate >= high
 123                 or iteration > NEWTON_FALLBACK_ITERATION
 124             ):
 125                 candidate = 0.5 * (low + high)
 126             fiber_length_m = candidate
 127         fiber_velocity_m_s = (fiber_length_m - previous_fiber_length_m[i]) / step_s
 128         active_length_factor, velocity_factor, passive_length_factor, dfl, dfv, dfp = (
 129             muscle_force_curve_derivatives(
 130                 fiber_length_m / optimal_fiber_length_m[i],
 131                 fiber_velocity_m_s / maximum_fiber_speed_m_s,
 132                 muscle_parameters[i],
 133             )
 134         )
 135         normalized_tendon_force = elastic_tendon_force(
 136             (muscle_tendon_length_m[i] - fiber_length_m * cosine) / tendon_slack_length_m[i]
 137             - 1,
 138             strain_at_maximum_force[i],
 139         )
 140         length[i] = fiber_length_m
 141         velocity[i] = fiber_velocity_m_s
 142         tension[i] = muscle_parameters[i, 2] * normalized_tendon_force
 143         residual[i] = muscle_parameters[i, 2] * (
 144             normalized_tendon_force
 145             - activation[i] * active_length_factor * velocity_factor
 146             - passive_length_factor
 147             - coefficient * fiber_velocity_m_s / maximum_fiber_speed_m_s
 148         )
 149         if (
 150             fiber_length_m <= 0.200001 * optimal_fiber_length_m[i]
 151             or muscle_tendon_length_m[i] - fiber_length_m * cosine <= 1e-9
 152             or muscle_tendon_length_m[i] <= 0.2 * optimal_fiber_length_m[i] * cosine
 153         ):
 154             bound[i] = 1
 155     return length, velocity, tension, residual, bound
