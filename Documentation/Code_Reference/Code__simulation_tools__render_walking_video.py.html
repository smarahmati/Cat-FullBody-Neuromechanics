Code/simulation_tools/render_walking_video.py

   1 """Render a video of the cat from saved forward-dynamics states.
   2 
   3 This script does not drive or solve the model. The visible movement and paw
   4 forces come from forward_simulation_timeseries.npz. Arguments select the saved time interval,
   5 output frame rate, and playback speed. Time is in seconds; force is in newtons.
   6 """
   7 
   8 import argparse
   9 import json
  10 import os
  11 from pathlib import Path
  12 import subprocess
  13 import sys
  14 
  15 ROOT = Path(__file__).resolve().parents[2]
  16 sys.path.insert(0, str(ROOT / "Code"))
  17 import numpy as np
  18 import mujoco as mj
  19 import imageio_ffmpeg
  20 from PIL import Image, ImageDraw, ImageFont
  21 from neuromechanics.initialize_neuromechanical_model import initialize_neuromechanical_model
  22 from neuromechanics.interpolate_simulation_for_display import (
  23     interpolate_simulation_for_display,
  24 )
  25 from neuromechanics.update_saved_visual_state import update_saved_visual_state
  26 
  27 
  28 def render_walking_video(
  29     run_folder, output, start_s=56.0, end_s=59.9, frames_per_second=40, playback_speed=0.5
  30 ):
  31     """Turn saved states into frames; interpolate only for visual smoothness."""
  32     folder = ROOT / run_folder
  33     result = dict(np.load(folder / "forward_simulation_timeseries.npz"))
  34     settings = json.loads((folder / "settings.json").read_text())
  35     state = initialize_neuromechanical_model(settings, graphics=True)
  36     model, data = state["model"], state["data"]
  37     width, height = 1100, 720
  38     model.vis.global_.offwidth = width
  39     model.vis.global_.offheight = height
  40     renderer = mj.Renderer(model, height=height, width=width)
  41     camera = mj.MjvCamera()
  42     camera.distance = 0.95
  43     camera.azimuth = 90
  44     camera.elevation = -17
  45     camera.lookat[:] = [0.15, 0.0, 0.22]
  46     output = Path(output)
  47     output.parent.mkdir(parents=True, exist_ok=True)
  48     command = [
  49         imageio_ffmpeg.get_ffmpeg_exe(),
  50         "-y",
  51         "-f",
  52         "rawvideo",
  53         "-vcodec",
  54         "rawvideo",
  55         "-s",
  56         f"{width}x{height}",
  57         "-pix_fmt",
  58         "rgb24",
  59         "-r",
  60         str(frames_per_second),
  61         "-i",
  62         "-",
  63         "-an",
  64         "-c:v",
  65         "libx264",
  66         "-crf",
  67         "20",
  68         "-pix_fmt",
  69         "yuv420p",
  70         "-movflags",
  71         "+faststart",
  72         str(output),
  73     ]
  74     encoder = subprocess.Popen(command, stdin=subprocess.PIPE, stderr=subprocess.PIPE)
  75     start_s = max(float(result["time"][0]), start_s)
  76     end_s = min(float(result["time"][-1]), end_s)
  77     frame_count = int((end_s - start_s) / playback_speed * frames_per_second)
  78     try:
  79         font = ImageFont.truetype("DejaVuSans.ttf", 22)
  80     except OSError:
  81         font = ImageFont.load_default(size=22)
  82     colors = [(30, 215, 235), (255, 45, 165), (250, 190, 40), (180, 110, 255)]
  83     for frame in range(frame_count):
  84         current_s = start_s + frame * playback_speed / frames_per_second
  85         sample = interpolate_simulation_for_display(result, current_s)
  86         update_saved_visual_state(state, sample)
  87         camera.lookat[0] = data.xpos[model.body("pelvis").id, 0] + 0.12
  88         renderer.update_scene(data, camera=camera)
  89         renderer.scene.flags[mj.mjtRndFlag.mjRND_SHADOW] = False
  90         renderer.scene.flags[mj.mjtRndFlag.mjRND_REFLECTION] = False
  91         for paw, force in enumerate(sample["grf"]):
  92             magnitude = float(np.linalg.norm(force))
  93             if magnitude < 0.05:
  94                 # Skip the remaining work for this item.
  95                 continue
  96             arrow = renderer.scene.geoms[renderer.scene.ngeom]
  97             origin = data.site_xpos[state["paw_ids"][paw]].copy()
  98             origin[2] = 0.002
  99             rgba = np.r_[np.asarray(colors[paw]) / 255.0, 1.0]
 100             mj.mjv_initGeom(
 101                 arrow,
 102                 mj.mjtGeom.mjGEOM_ARROW,
 103                 np.zeros(3),
 104                 np.zeros(3),
 105                 np.eye(3).ravel(),
 106                 rgba,
 107             )
 108             mj.mjv_connector(
 109                 arrow, mj.mjtGeom.mjGEOM_ARROW, 0.0035, origin, origin + 0.006 * force
 110             )
 111             renderer.scene.ngeom += 1
 112         pixels = Image.fromarray(renderer.render())
 113         text = ImageDraw.Draw(pixels)
 114         text.rectangle((0, 0, width, 90), fill=(16, 30, 42))
 115         text.text(
 116             (20, 12),
 117             f"CAT | forward dynamics | t = {current_s:.2f} s | playback {playback_speed:g}x",
 118             font=font,
 119             fill="white",
 120         )
 121         text.text(
 122             (20, 48),
 123             "Elastic tendons; prescribed excitation; calculated paw-contact forces",
 124             font=font,
 125             fill="white",
 126         )
 127         for paw, label in enumerate(["LH", "RH", "LF", "RF"]):
 128             text.text(
 129                 (20 + paw * 270, height - 38),
 130                 f'{label}: Fx {sample["grf"][paw,0]:+.1f}, Fz {sample["grf"][paw,2]:.1f} N',
 131                 font=font,
 132                 fill=colors[paw],
 133             )
 134         encoder.stdin.write(np.asarray(pixels).tobytes())
 135         if frame == 0:
 136             pixels.save(output.with_suffix(".png"))
 137     encoder.stdin.close()
 138     error_text = encoder.stderr.read().decode(errors="replace")
 139     if encoder.wait() != 0:
 140         # Stop and explain the invalid model setting or failed calculation.
 141         raise RuntimeError(error_text)
 142     renderer.close()
 143     metadata = dict(
 144         source=str(folder / "forward_simulation_timeseries.npz"),
 145         start_s=start_s,
 146         end_s=end_s,
 147         frame_count=frame_count,
 148         frames_per_second=frames_per_second,
 149         playback_speed=playback_speed,
 150         species="cat",
 151         arrows="saved Ground_Reaction_Forces at projected paw sites; archived center of pressure unavailable",
 152     )
 153     # Save the computed values or figure for inspection and reproducibility.
 154     output.with_suffix(".json").write_text(json.dumps(metadata, indent=2))
 155     print("VIDEO_SAVED " + str(output))
 156 
 157 
 158 if __name__ == "__main__":
 159     parser = argparse.ArgumentParser(description=__doc__)
 160     parser.add_argument("--run", default="Simulation_Results/Intact_Walking")
 161     parser.add_argument(
 162         "--output", default=str(ROOT / "Documentation/Figures_and_Videos/walking_simulation.mp4")
 163     )
 164     arguments = parser.parse_args()
 165     render_walking_video(arguments.run, arguments.output)
