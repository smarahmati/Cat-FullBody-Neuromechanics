  10 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  11 @njit(cache=True)
  12 def calculate_muscle_route_points(q, meta, geom):
  13     out = np.empty((len(meta), 2))
  14     for k in range(len(meta)):
  15         adr, j1, j2, fore, mono = meta[k, :5]
  16         ls = geom[k, 10]
  17         g = geom[k, :10]
  18         a = (
  19             (math.pi + q[adr + j1] if j1 == 0 else math.pi - q[adr + j1])
  20             if fore
  21             else (
  22                 -q[adr + j1]
  23                 if j1 == 0
  24                 else (math.pi + q[adr + j1] if j1 == 1 else math.pi - q[adr + j1])
  25             )
  26         )
  27         if mono:
  28             a1, a2, p1, p2, r1, f1, n1 = g[:7]
  29             beta = f1 * math.acos(max(-1.0, min(1.0, r1 / a1))) + p1
  30             beta2 = -f1 * math.acos(max(-1.0, min(1.0, r1 / a2))) + p2
  31             theta = abs(f1 - n1) * math.pi + f1 * n1 * (a - n1 * (beta - beta2))
  32             ang = math.pi + beta + f1 * max(theta, 0.0) / 2
  33             out[k, 0] = r1 * 1.05 * math.cos(ang) + ls
  34             out[k, 1] = r1 * 1.05 * math.sin(ang)
  35         else:
  36             b = (
  37                 (math.pi + q[adr + j2] if j2 == 0 else math.pi - q[adr + j2])
  38                 if fore
  39                 else (
  40                     -q[adr + j2]
  41                     if j2 == 0
  42                     else (math.pi + q[adr + j2] if j2 == 1 else math.pi - q[adr + j2])
  43                 )
  44             )
  45             out[k] = route_two_joint_muscle(g, a, b, ls)[1]
  46     return out
