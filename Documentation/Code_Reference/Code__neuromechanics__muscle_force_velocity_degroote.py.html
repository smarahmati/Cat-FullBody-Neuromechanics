   1 """De Groote et al. (2016) smooth normalized fiber force-velocity curve.
   2 
   3 Primary implementation: OpenSim DeGrooteFregly2016Muscle.h, with its corrected
   4 constants that pass through (-1,0) and (0,1). Original rounded paper values
   5 were d1=-0.318 and d4=0.886; applied values are recorded below. This optional
   6 limb-selectable adaptation is NOT the force-velocity law in Rahmati's MATLAB model.
   7 Normalized velocity is positive for lengthening. Published domain [-1,1].
   8 Outside that domain we retain the algebraic curve but prevent negative active
   9 tension; excursions must be reported, not interpreted as validated behavior.
  10 """
  11 
  12 import numpy as np
  13 
  14 from numba import njit
  15 
  16 from model_configuration.formulation_constants import FV_D1, FV_D2, FV_D3, FV_D4
  17 
  18 
  19 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  20 @njit(cache=False)
  21 def muscle_force_velocity_degroote(velocity):
  22     x = FV_D2 * velocity + FV_D3
  23     raw = FV_D1 * np.arcsinh(x) + FV_D4
  24     value = max(0.0, raw)
  25     derivative = FV_D1 * FV_D2 / np.sqrt(1 + x * x) if raw > 0 else 0.0
  26     return value, derivative
