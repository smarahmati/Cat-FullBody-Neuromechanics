   1 """Timing agreement between input muscle_excitation and EMG on the original clock."""
   2 
   3 import numpy as np
   4 
   5 from scipy.optimize import linear_sum_assignment
   6 
   7 
   8 def emg_phase_metrics(muscle_excitation, reference, period, threshold=0.10):
   9     """Score one complete cycle; both arrays are already peak-normalized.
  10 
  11     The detection threshold is an analysis choice. No signal is phase-shifted,
  12     filtered, baseline-subtracted or modified. Edge matching is circular, so
  13     a burst crossing the end of the cycle is handled as one burst.
  14     """
  15     u = np.asarray(muscle_excitation, float)
  16     r = np.asarray(reference, float)
  17     if u.shape != r.shape or u.ndim != 1:
  18         # Stop and explain the invalid model setting or failed calculation.
  19         raise ValueError("Use equally sampled one-dimensional cycles")
  20     predicted = u > threshold
  21     expected = r > threshold
  22 
  23     def edges(mask):
  24         return np.flatnonzero(mask & ~np.roll(mask, 1)), np.flatnonzero(
  25             ~mask
  26             & np.roll(mask, 1)
  27         )
  28 
  29     on, off = edges(predicted)
  30     ron, roff = edges(expected)
  31     n = len(u)
  32 
  33     def match(got, wanted):
  34         if not len(got) or not len(wanted):
  35             return dict(
  36                 mean_absolute_error_s=None,
  37                 maximum_absolute_error_s=None,
  38                 unmatched_predicted=len(got),
  39                 unmatched_reference=len(wanted),
  40             )
  41         difference = ((got[:, None] - wanted[None, :] + n / 2) % n - n / 2) * period / n
  42         rows, cols = linear_sum_assignment(abs(difference))
  43         # Calculate the error in this comparison.
  44         error = abs(difference[rows, cols])
  45         return dict(
  46             mean_absolute_error_s=float(error.mean()),
  47             maximum_absolute_error_s=float(error.max()),
  48             unmatched_predicted=int(len(got) - len(rows)),
  49             unmatched_reference=int(len(wanted) - len(cols)),
  50         )
  51 
  52     union = np.count_nonzero(predicted | expected)
  53     return dict(
  54         threshold_fraction_of_peak=threshold,
  55         active_intersection_over_union=(
  56             float(np.count_nonzero(predicted & expected) / union)
  57             if union
  58             else 1.0
  59         ),
  60         phase_disagreement_fraction=float(np.mean(predicted != expected)),
  61         active_fraction_predicted=float(predicted.mean()),
  62         active_fraction_reference=float(expected.mean()),
  63         false_active_fraction_of_cycle=float(np.mean(predicted & ~expected)),
  64         missed_active_fraction_of_cycle=float(np.mean(~predicted & expected)),
  65         onset=match(on, ron),
  66         offset=match(off, roff),
  67         predicted_burst_count=int(len(on)),
  68         reference_burst_count=int(len(ron)),
  69     )
