  15 def write_interactive_comparison(result, settings, folder):
  16     """Write selectable 1/2/4/8-second or full-trial views with a time slider."""
  17     folder = Path(folder)
  18     target = experimental_comparison_targets(result['time'], settings['period_s'], settings.get('reference_table_dir'))
  19     labels = ['LH — assumed mirror', 'RH — recorded', 'LF — separate study', 'RF — separate study']
  20     payload = {'time': np.round(result['time'], 6).tolist(), 'kinematics': [], 'forces': []}
  21     for side in range(4):
  22         joints = ['Hip', 'Knee', 'Ankle', 'MTP'] if side < 2 else ['Shoulder', 'Elbow', 'Wrist', 'MCP']
  23         signs = [-1, 1, -1, -1] if side < 2 else [1, -1, -1, -1]
  24         offsets = [0, np.pi, np.pi, np.pi] if side < 2 else [np.pi] * 4
  25         for joint, name in enumerate(joints):
  26             payload['kinematics'].append({'title': labels[side] + ' · ' + name, 'limb': side, 'unit': 'deg',
  27                 'prediction': np.round(np.rad2deg(offsets[joint] + signs[joint] * result['joint_q'][:, side, joint]), 6).tolist(),
  28                 'reference': np.round(np.rad2deg(offsets[joint] + signs[joint] * target['joint_q'][:, side, joint]), 6).tolist()})
  29         for component, name in [(0, 'Anterior–posterior GRF'), (2, 'Vertical GRF')]:
  30             payload['forces'].append({'title': labels[side] + ' · ' + name, 'limb': side, 'unit': 'N',
  31                 'prediction': np.round(result['grf'][:, side, component], 6).tolist(),
  32                 'reference': np.round(target['grf'][:, side, component], 6).tolist()})
  33     relative = lambda path: os.path.relpath(ROOT / path, folder).replace(os.sep, '/')

  35     page = page.replace('STYLE', relative('Documentation/style.css')).replace('HOME', relative('START_HERE.html'))
