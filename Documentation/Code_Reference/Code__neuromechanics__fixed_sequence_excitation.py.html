   6 def fixed_sequence_excitation(state, time):
   7     dt = state["model"].opt.timestep
   8     period = state["settings"]["period_s"]
   9     steps = round(period / dt)
  10     tick = round(time / dt)
  11     if abs(steps * dt - period) > 1e-10:
  12         # Stop and explain the invalid model setting or failed calculation.
  13         raise ValueError(
  14             "Frozen sequence requires an integer number of integration steps per stride"
  15         )
  16     if "offline_cycle_input" in state:
  17         start, table = state["offline_cycle_input"]
  18         phase = np.clip((tick - start) / steps, 0.0, 1.0)
  19     else:
  20         sequence = state["excitation_sequence"]
  21         cycle, within = divmod(tick, steps)
  22         if cycle == len(sequence) and within == 0:
  23             return sequence[-1, -1].copy()
  24         if cycle >= len(sequence):
  25             # Stop and explain the invalid model setting or failed calculation.
  26             raise ValueError(
  27                 "Frozen muscle_excitation sequence exhausted; no untested repetition is allowed"
  28             )
  29         table = sequence[cycle]
  30         phase = within / steps
  31     position = phase * (len(table) - 1)
  32     lo = min(int(position), len(table) - 2)
  33     weight = position - lo
  34     return np.clip((1 - weight) * table[lo] + weight * table[lo + 1], 0.0, 1.0)
