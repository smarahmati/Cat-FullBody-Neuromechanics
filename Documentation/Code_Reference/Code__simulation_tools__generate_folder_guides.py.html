 158 def describe(path):
 159     """Explain each item from its scientific role or Python module documentation."""
 160     if path.name in PURPOSE:
 161         return PURPOSE[path.name]
 162     if path.suffix == '.py':
 163         try:
 164             return (ast.get_docstring(ast.parse(path.read_text(encoding='utf-8'))) or path.stem.replace('_',' ')).split('\n')[0]
 165         except (SyntaxError, UnicodeError):
 166             pass
 167     if 'Original_Study_Data' in path.parts:
 168         if path.name.startswith('Activations_'):
 169             return 'Original inverse-analysis activation estimates for the trial number in the filename; not a measured EMG recording.'
 170         if path.suffix == '.m':
 171             return 'Original MATLAB source for the cited study. Open it to inspect its equations, variable names and source parameter conventions.'
 172         if path.suffix == '.mat':
 173             return 'Original MATLAB numerical data. Open with MATLAB or scipy.io.loadmat to inspect named variables; source units and conventions follow the associated study code.'
 174         return 'Original study source record documenting its data or anatomical calculations; retained for scientific traceability.'
 175     return path.stem.replace('_',' ') + ({'.csv': ' — table with named columns.', '.tsv': ' — tab-separated signal table.', '.npz': ' — named numerical arrays.', '.json': ' — machine-readable data/metadata.', '.png': ' — figure.', '.svg': ' — scalable diagram.', '.mp4': ' — walking animation.'}.get(path.suffix,'.'))

 178 def link(page, target):
 179     """Return a portable relative link for downloaded HTML and GitHub Pages."""
 180     return os.path.relpath(target,page.parent).replace('\\','/')
 181

 183 def explain_navigation_lists(page):
 184     """Add meaning beside bare local-file bullets without changing detailed prose."""
 185     original = page.read_text(encoding='utf-8')
 186 
 187     def explain(match):
 188         address = html.unescape(match.group(1))
 189         parsed = urlsplit(address)
 190         if parsed.scheme or parsed.netloc or not parsed.path:
 191             return match.group(0)
 192         target = (page.parent / unquote(parsed.path)).resolve()
 193         if not target.is_relative_to(ROOT) or not target.exists():
 194             return match.group(0)
 195         item = target.parent if target.name == 'index.html' else target
 196         # Folder labels follow their current descriptive names; source file names stay exact.
 197         label = item.name.replace('_', ' ') if item.is_dir() else item.name
 198         return f'<li><a href="{html.escape(address, quote=True)}">{html.escape(label)}</a> — {html.escape(describe(item))}</li>'
 199 
 200     updated = re.sub(r'<li>\s*<a href="([^"]+)">[^<]+</a>(?:\s*—[^<]*)?\s*</li>', explain, original)
 201     # Scientific result guides describe their contents, not the packaging workflow.
 202     updated = updated.replace('Fresh forward result with passive fiber force in all 104 channels; no fine-tuning.', 'Computed intact-cat forward results. The columns and named arrays below describe the simulated quantities.')
 203     if updated != original:
 204         page.write_text(updated, encoding='utf-8')

 207 def write_page(page,title,body):
 208     """Use the common documentation style without an external web dependency."""
 209     page.parent.mkdir(parents=True,exist_ok=True)
 210     page.write_text(f'<!doctype html><html lang="en"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>{html.escape(title)}</title><link rel="stylesheet" href="{link(page,ROOT/"Documentation/style.css")}"></head><body>{MARKER}<header><div class="tag">Cat · Full-body neuromechanics</div><h1>{html.escape(title)}</h1><nav><a href="{link(page,ROOT/"START_HERE.html")}">Start here</a> · <a href="{link(page,ROOT/"Documentation/project_structure.html")}">Find a file</a></nav></header><main>{body}</main></body></html>',encoding='utf-8')

 213 def main():
 214     """Generate only from Git's public inventory; ignored local data stays private."""
 215     output=subprocess.check_output(['git','-C',str(ROOT),'ls-files','--cached','--others','--exclude-standard','-z']).decode('utf-8')
 216     files=sorted({ROOT/p for p in output.split('\0') if p and (ROOT/p).is_file()})
 217     folders={q for p in files for q in p.parents if q.is_relative_to(ROOT)}
 218     for page in files:
 219         if page.suffix == '.html' and 'Code_Reference' not in page.parts:
 220             explain_navigation_lists(page)
 221     for folder in sorted(folders):
 222         if folder==ROOT or any(x.startswith('.') for x in folder.relative_to(ROOT).parts):continue
 223         page=folder/'index.html'
 224         if page.exists() and MARKER not in page.read_text(encoding='utf-8'):continue
 225         children=sorted({p if p.parent==folder else folder/p.relative_to(folder).parts[0] for p in files if p.is_relative_to(folder) and p!=page},key=lambda p:(not p.is_dir(),p.name.lower()))
 226         rows=''.join(f'<tr><td><a href="{html.escape(link(page,p/"index.html" if p.is_dir() else p))}">{html.escape(p.name)}</a></td><td>{html.escape(describe(p))}</td></tr>' for p in children)
 227         write_page(page,folder.name.replace('_',' '),f'<p>{html.escape(describe(folder))}</p><p><code>{folder.relative_to(ROOT).as_posix()}</code></p><table><tr><th>File or folder</th><th>Purpose and units</th></tr>{rows}</table>')
 228     guide=ROOT/'Documentation/project_structure.html'
 229     paths=['run.py','view.py','reproduce.py','Code/neuromechanics','Code/model_configuration','Code/simulation_tools','Data/Model_Parameters','Data/Muscle_Excitation_Input','Data/Experimental_Reference_Data','Data/Original_Study_Data','Data/Skeletal_Geometry','Simulation_Results/Intact_Walking','Verification','Documentation/References','Documentation/Figures_and_Videos']
 230     rows=''.join(f'<tr><td><a href="{link(guide,ROOT/p/"index.html" if (ROOT/p).is_dir() else ROOT/p)}"><code>{p}</code></a></td><td>{html.escape(describe(ROOT/p))}</td></tr>' for p in paths)
 231     write_page(guide,'Project folders and files','<p>Start with run.py. Model parameters, prescribed excitation, experimental references and computed results have separate, descriptive folders. Each folder has a navigation guide.</p><table><tr><th>Path from project root</th><th>Purpose</th></tr>'+rows+'<h2>Names and scientific identifiers</h2><p>Scientific muscle channel identifiers are defined in the <a href="muscle_dictionary.html">muscle dictionary</a>. Original research filenames remain unchanged for traceability. README.md, index.html and __init__.py follow standard GitHub, website and Python conventions.</p>')
 232     count=0
 233     for source in files:
 234         if source.suffix!='.py' or 'Simulation_Results' in source.parts:continue
 235         relative=source.relative_to(ROOT).as_posix()
 236         target=ROOT/'Documentation/Code_Reference'/(relative.replace('/','__')+'.html')
 237         lines=''.join(f'<span id="L{i}">{i:4}</span> {html.escape(line)}\n' for i,line in enumerate(source.read_text(encoding='utf-8').splitlines(),1))
 238         write_page(target,describe(source),f'<p><a href="{link(target,source)}">{relative}</a></p><pre style="overflow:auto;white-space:pre">{lines}</pre>')
 239         count+=1
 240     print(f'Generated folder guides and {count} source-code pages.')
 241

 243 if __name__=='__main__':
 244     main()
