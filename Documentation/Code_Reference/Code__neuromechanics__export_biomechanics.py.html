   1 """Export readable time series and mechanics derived from SAVED forward states.
   2 
   3 This postprocessing does not change the simulated trajectory. CSV files use
   4 seconds and SI units; anatomical joint angles are also exported in degrees.
   5 Dense muscle-by-coordinate arrays are compressed in NPZ to keep files small.
   6 """

  10 import mujoco as mj

  12 from .muscle_moment_arms import muscle_moment_arms
  13 from .muscle_joint_moments import muscle_joint_moments
  14 from .muscle_force_length_velocity import muscle_force_length_velocity

  17 def save_table(path, times, values, columns):
  18     """Write a CSV with explicit column names and a time_s first column."""
  19     path = Path(path)
  20     path.parent.mkdir(parents=True, exist_ok=True)
  21     matrix = np.asarray(values).reshape(len(times), -1)
  22     np.savetxt(path, np.column_stack((times, matrix)), delimiter=',',
  23                header=','.join(['time_s'] + list(columns)), comments='', fmt='%.10g')

  26 def export_biomechanics(result, state, folder):
  27     """Export input, activation, fiber/tendon, motion, force and moment folders."""
  28     folder = Path(folder)
  29     times = result['time']
  30     muscles = [str(n) for n in result['names']]
  31     coordinates = [str(n) for n in result['coordinate_names']]
  32     paths = {
  33         'excitation': ('Muscle_Excitation/excitation_0_to_1.csv', ''),
  34         'activation': ('Muscle_Activation/activation_0_to_1.csv', ''),
  35         'force': ('Muscle_Forces/tendon_tension_N.csv', '_N'),
  36         'fiber_length': ('Muscle_Tendon_States/fiber_length_m.csv', '_m'),
  37         'fiber_velocity': ('Muscle_Tendon_States/fiber_velocity_m_s.csv', '_m_s'),
  38         'mtu_length': ('Muscle_Tendon_States/whole_MTU_length_m.csv', '_m'),
  39         'tendon_length': ('Muscle_Tendon_States/elastic_tendon_length_m.csv', '_m'),
  40     }
  41     for key, (name, unit) in paths.items():
  42         save_table(folder/name, times, result[key], [m+unit for m in muscles])
  43     # Evaluate the parallel elastic contribution at each saved fiber length.
  44     # F0 is already tendon-projected in this model: no extra cos(alpha) factor.
  45     # This component is included in the fiber/tendon equilibrium during solving;
  46     # it is NOT an additional external force applied a second time afterwards.
  47     parameters = state['muscle_params']
  48     normalized_length = result['fiber_length'] / result['optimal_fiber_length_m']
  49     passive_force = np.zeros_like(normalized_length)
  50     for channel in range(len(muscles)):
  51         for sample in range(len(times)):
  52             passive_factor = muscle_force_length_velocity(
  53                 normalized_length[sample, channel], 0.0, parameters[channel]
  54             )[2]
  55             passive_force[sample, channel] = parameters[channel, 2] * passive_factor
  56     save_table(folder/'Muscle_Forces/passive_fiber_tendon_component_N.csv',
  57                times, passive_force, [m+'_N' for m in muscles])
  58     save_table(folder/'Muscle_Tendon_States/normalized_fiber_length.csv',
  59                times, normalized_length, muscles)
  60     passive_audit = {
  61         'law': 'Prilutsky 2016 Table 10.3 / Rahmati 2025 Table 3, tension-only adaptation',
  62         'enabled_channels': int(np.count_nonzero(parameters[:, 13] > 0)),
  63         'total_channels': len(muscles),
  64         'channels': {name: {
  65             'enabled': bool(parameters[i, 13] > 0),
  66             'maximum_normalized_fiber_length': float(normalized_length[:, i].max()),
  67             'peak_passive_tendon_component_N': float(passive_force[:, i].max()),
  68         } for i, name in enumerate(muscles)},
  69     }
  70     (folder/'passive_fiber_force_summary.json').write_text(json.dumps(passive_audit, indent=2))
  71     strain = result['tendon_length']/state['slack'] - 1
  72     save_table(folder/'Muscle_Tendon_States/tendon_strain.csv', times, strain, muscles)
  73     save_table(folder/'Joint_Kinematics/generalized_positions.csv', times, result['qpos'], coordinates)
  74     save_table(folder/'Joint_Kinematics/generalized_velocities.csv', times, result['qvel'], coordinates)
  75     # Finite differences of saved velocities are clearly labelled; they are not
  76     # presented as exact integration-step accelerations.
  77     acceleration = np.gradient(result['qvel'], times, axis=0) if len(times)>1 else np.zeros_like(result['qvel'])
  78     save_table(folder/'Joint_Kinematics/acceleration_from_saved_velocities.csv', times, acceleration, coordinates)
  79     joint_names = [side+'_'+j for side, joints in [('LH',['hip','knee','ankle','MTP']),('RH',['hip','knee','ankle','MTP']),('LF',['shoulder','elbow','wrist','MCP']),('RF',['shoulder','elbow','wrist','MCP'])] for j in joints]
  80     anatomical = np.empty_like(result['joint_q'])
  81     for side in range(4):
  82         signs = np.array([-1,1,-1,-1]) if side<2 else np.array([1,-1,-1,-1])
  83         offset = np.array([0,np.pi,np.pi,np.pi]) if side<2 else np.full(4,np.pi)
  84         anatomical[:,side] = np.rad2deg(offset+signs*result['joint_q'][:,side])
  85     save_table(folder/'Joint_Kinematics/anatomical_joint_angles_deg.csv',times,anatomical,joint_names)
  86     save_table(folder/'Ground_Reaction_Forces/raw_paw_forces_N.csv',times,result['grf'],[s+'_'+axis+'_N' for s in ['LH','RH','LF','RF'] for axis in ['Fx','Fy','Fz']])
  87 
  88     # Reconstruct geometry on the completed state workspace. No stepping,
  89     # contact force recalculation or motor adaptation occurs here.
  90     model, data = state['model'], state['data']
  91     arms = np.empty((len(times),model.nu,model.nv))
  92     moments = np.empty_like(arms)
  93     paw_positions = np.empty((len(times),4,3))
  94     body_positions = np.empty((len(times),model.nbody,3))
  95     for row in range(len(times)):
  96         data.qpos[:] = result['qpos'][row]
  97         data.qvel[:] = result['qvel'][row]
  98         update_muscle_paths(state)
  99         mj.mj_fwdPosition(model,data)
 100         arms[row] = muscle_moment_arms(model,data)
 101         moments[row], _ = muscle_joint_moments(arms[row],result['force'][row])
 102         paw_positions[row] = data.site_xpos[state['paw_ids']]
 103         body_positions[row] = data.xipos
 104     for directory in ['Moment_Arms','Muscle_Joint_Moments','Body_Motion']:
 105         (folder/directory).mkdir(exist_ok=True)
 106     np.savez_compressed(folder/'Moment_Arms/signed_transmission_moment_arms.npz',time=times,values=arms,muscle_names=muscles,coordinate_names=coordinates)
 107     np.savez_compressed(folder/'Muscle_Joint_Moments/individual_contributions.npz',time=times,values=moments,muscle_names=muscles,coordinate_names=coordinates)
 108     save_table(folder/'Muscle_Joint_Moments/summed_muscle_generalized_forces.csv',times,moments.sum(axis=1),coordinates)
 109     np.savez_compressed(folder/'Body_Motion/body_and_paw_positions.npz',time=times,paw_positions=paw_positions,body_com_positions=body_positions,body_names=[model.body(i).name for i in range(model.nbody)])
 110     metadata = {
 111         'time':'s; absolute simulation clock',
 112         'muscle_input':'excitation; prescribed in advance; no sensory feedback',
 113         'force':'positive tensile force, N',
 114         'passive_fiber_tendon_component':'F0 * fP(l_f/l_opt), N; evaluated at saved fiber state; included inside muscle equilibrium, not added again to tendon tension; zero below stretch threshold even when enabled',
 115         'generalized_units':'hinges: rad, rad/s, rad/s^2 and N m; sliders: m, m/s, m/s^2 and N',
 116         'moments':'muscle-only generalized contributions; no claim of inverse-dynamics or reaction moments',
 117         'axes':'world +x progression, +z upward; anterior–posterior (AP) Fx can be braking (<0) or propulsive (>0)',
 118         'moment_arm':'-dL_MT/dq from the actual engine transmission; same moving route geometry as simulation',
 119         'accelerations':'finite differences of saved velocities, not exact step acceleration',
 120         'reference_scope':'RH measured; LH assumed mirror; forelimbs separate study, EMG pooled',
 121     }
 122     (folder/'OUTPUT_DICTIONARY.json').write_text(json.dumps(metadata,indent=2),encoding='utf-8')
