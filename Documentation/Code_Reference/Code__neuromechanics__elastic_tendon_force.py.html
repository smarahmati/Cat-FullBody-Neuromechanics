   5 from numba import njit

   7 from model_configuration.formulation_constants import TENDON_TOE_CURVATURE

   9 from .elastic_tendon_transition import elastic_tendon_transition

  12 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  13 @njit(cache=False)
  14 def elastic_tendon_force(strain, strain_at_maximum_force):
  15     curvature = TENDON_TOE_CURVATURE
  16     toe = elastic_tendon_transition(strain_at_maximum_force)
  17     slope = 0.67 / (strain_at_maximum_force - toe)
  18     if strain <= 0:
  19         return 0.0
  20     if strain <= toe:
  21         return 0.33 * np.expm1(curvature * strain / toe) / np.expm1(curvature)
  22     return 0.33 + slope * (strain - toe)
