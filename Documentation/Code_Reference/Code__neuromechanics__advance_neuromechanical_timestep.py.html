  20 def advance_neuromechanical_timestep(state):
  21     """Fixed muscle_excitation -> activation/motion -> elastic force/contact update.
  22 
  23     This is the same partitioned numerical scheme as the optimized research
  24     step. References are absent from this loop. There is no feedback command,
  25     target-angle enforcement, external balance support or stride reset.
  26     """
  27     # Get the common skeleton and its current positions, velocities and activation.
  28     model, data = state["model"], state["data"]
  29     # Read this instant's fixed motor inputs, from the prescribed intact input table.
  30     motor_input = muscle_excitation(state, data.time)
  31     # Advance activation and motion using the muscle/contact forces already computed.
  32     multibody_system_dynamics(state, motor_input)
  33     # Move the analytical via points with the updated bones and joint angles.
  34     update_muscle_paths(state)
  35     # Recalculate positions and routed muscle lengths at the new pose.
  36     mj.mj_fwdPosition(model, data)
  37     # Recalculate velocities at the new pose and generalized speed.
  38     mj.mj_fwdVelocity(model, data)
  39     # Solve the new elastic tendon-fiber force balance at the updated MT lengths.
  40     muscle_tendon_dynamics(state)
  41     # Update pad compression, shear memory, Ground_Reaction_Forces and their moments on the skeleton.
  42     foot_contact_model(state, model.opt.timestep)
  43     # Tell the outer simulation loop whether this step remains numerically usable.
  44     return bool(
  45         # Every coordinate must be a finite number.
  46         np.isfinite(data.qpos).all()
  47         # Every tendon force must be a finite number.
  48         and np.isfinite(state["force"]).all()
  49         # Reject an extreme speed indicating numerical divergence.
  50         and np.max(abs(data.qvel)) < MAXIMUM_GENERALIZED_SPEED
  51     )
