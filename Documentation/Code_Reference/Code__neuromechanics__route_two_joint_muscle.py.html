   9

  11 
  12 from .clipped_acos import clipped_acos as acos

  14 from .polar_point import polar_point as polar

  17 # Compile this numerical function for repeated evaluations; its equations stay unchanged.
  18 @njit(cache=True)
  19 def route_two_joint_muscle(g, a, b, ls):
  20     """Length and intermediate tangent point, translated from the MATLAB cases."""
  21     a1, a2, p1, p2, r1, f1, n1 = g[:7]
  22     beta1 = f1 * acos(r1 / a1) + p1
  23     r2, f2, n2 = g[7:10]
  24     beta2 = -f2 * acos(r2 / a2) + p2
  25     h1 = math.sqrt(ls * ls - (f1 * r1 - f2 * r2) ** 2)
  26     rho1 = -f1 * acos((r2 * r2 + h1 * h1 - r1 * r1 - ls * ls) / (-2 * r1 * ls))
  27     rho2 = f2 * acos((r1 * r1 + h1 * h1 - r2 * r2 - ls * ls) / (-2 * r2 * ls))
  28     t1 = abs(f1 - n1) * math.pi + f1 * n1 * (a - n1 * (beta1 - rho1))
  29     t2 = abs(f2 - n2) * math.pi + f2 * n2 * (b + n2 * (beta2 - rho2))
  30     O = polar(a1, math.pi + n1 * (math.pi - a) + p1)
  31     I = np.array([ls, 0.0]) + polar(a2, n2 * (b - math.pi) + p2)
  32     if t1 >= 0 and t2 >= 0:
  33         P = polar(r1, rho1)
  34         Q = np.array([ls, 0.0]) + polar(r2, math.pi + rho2)
  35         L = (
  36             math.sqrt(max(0, a1 * a1 - r1 * r1))
  37             + r1 * t1
  38             + h1
  39             + r2 * t2
  40             + math.sqrt(max(0, a2 * a2 - r2 * r2))
  41         )
  42     elif t1 < 0 and t2 >= 0:
  43         h4 = math.sqrt(ls * ls + a1 * a1 - 2 * a1 * ls * math.cos(a - n1 * p1))
  44         rm = f2 * abs(
  45             f1 * acos((a1 * a1 - ls * ls - h4 * h4) / (-2 * ls * h4))
  46             + f2 * acos(r2 / h4)
  47         )
  48         tm = abs(f2 - n2) * math.pi + f2 * n2 * (b + n2 * (beta2 - rm))
  49         P = O
  50         Q = np.array([ls, 0.0]) + polar(r2, math.pi + rm)
  51         if tm < 0:
  52             Q = I
  53         L = math.sqrt(h4 * h4 - r2 * r2) + r2 * tm + math.sqrt(max(0, a2 * a2 - r2 * r2))
  54     elif t1 >= 0 and t2 < 0:
  55         h5 = math.sqrt(ls * ls + a2 * a2 - 2 * a2 * ls * math.cos(b + n2 * p2))
  56         rm = -f1 * abs(
  57             f1 * acos((a2 * a2 - ls * ls - h5 * h5) / (-2 * ls * h5))
  58             + f2 * acos(r1 / h5)
  59         )
  60         tm = abs(f1 - n1) * math.pi + f1 * n1 * (a - n1 * (beta1 - rm))
  61         P = polar(r1, rm)
  62         Q = I
  63         if tm < 0:
  64             P = O
  65         L = math.sqrt(max(0, a1 * a1 - r1 * r1)) + r1 * tm + math.sqrt(h5 * h5 - r1 * r1)
  66     else:
  67         P = O
  68         Q = I
  69         h6 = a1 * math.cos(math.pi - a + n1 * p1)
  70         h7 = a2 * math.cos(math.pi - b - n2 * p2)
  71         L = math.sqrt(
  72             (
  73                 f1 * math.sqrt(max(0, a1 * a1 - h6 * h6))
  74                 - f2 * math.sqrt(max(0, a2 * a2 - h7 * h7))
  75             )
  76             ** 2
  77             + (ls + h6 + h7) ** 2
  78         )
  79     return L, (P + Q) / 2, (t1, t2)
