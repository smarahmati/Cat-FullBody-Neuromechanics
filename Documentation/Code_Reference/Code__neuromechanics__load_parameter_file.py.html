   1 """Read a commented, editable Python dictionary without executing code."""
   2
   3 import ast
   4
   5 from pathlib import Path
   6
   7
   8 def load_parameter_file(path):
   9     """The file contains comments/docstrings and one PARAMETERS assignment.
  10
  11     Using literal dictionaries keeps configuration approachable while avoiding
  12     hidden computations. Every applied number is visible in the file.
  13     """
  14     tree = ast.parse(Path(path).read_text(encoding="utf-8"))
  15     assignments = [
  16         node
  17         for node in tree.body
  18         if isinstance(node, ast.Assign)
  19         and len(node.targets) == 1
  20         and isinstance(node.targets[0], ast.Name)
  21         and node.targets[0].id == "PARAMETERS"
  22     ]
  23     if len(assignments) != 1:
  24         # Stop and explain the invalid model setting or failed calculation.
  25         raise ValueError(f"{path}: expected one PARAMETERS dictionary")
  26     values = ast.literal_eval(assignments[0].value)
  27     if not isinstance(values, dict):
  28         # Stop and explain the invalid model setting or failed calculation.
  29         raise ValueError("PARAMETERS must be a dictionary")
  30     # Saved configurations may use the former name for this same force scale.
  31     # Normalize the name only: retain the value and its existing pennation projection.
  32     if Path(path).name == "muscle_tendon.py":
  33         old_name = "max_isometric_tendon_force_N"
  34         force_name = "max_isometric_muscle_force_along_tendon_N"
  35         for muscle, parameters in values.items():
  36             if old_name in parameters:
  37                 # Two names would make it unclear which user-edited value to apply.
  38                 if force_name in parameters:
  39                     raise ValueError(f"{path}/{muscle}: duplicate isometric force fields")
  40                 parameters[force_name] = parameters.pop(old_name)
  41     return values
