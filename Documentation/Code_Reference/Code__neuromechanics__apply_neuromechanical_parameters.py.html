  14 def apply_neuromechanical_parameters(state, folder):
  15     """All assignments happen before forward integration, never during walking.
  16 
  17     Values are absolute applied values, not multipliers applied repeatedly.
  18     Existing settings without a bundle preserve the historical model exactly.
  19     Names and topology must match the selected anatomical model.
  20     """
  21     folder = ROOT / Path(folder)
  22     s = state
  23     m = s["model"]
  24     names = list(s["names"])
  25 
  26     # Reject misspelled/unknown entries rather than silently ignore an edit.
  27     def checked(filename, fields=None, record_names=None):
  28         values = load_parameter_file(folder / filename)
  29         if record_names is not None and set(values) != set(record_names):
  30             # Stop and explain the invalid model setting or failed calculation.
  31             raise ValueError(f"{filename}: model record names differ")
  32         for name, record in values.items():
  33             if fields is not None and set(record) != set(fields):
  34                 # Stop and explain the invalid model setting or failed calculation.
  35                 raise ValueError(f"{filename}/{name}: unknown or missing field")
  36             numbers = list(record.values()) if isinstance(record, dict) else [record]
  37             if any(not np.isfinite(np.asarray(v, float)).all() for v in numbers):
  38                 # Stop and explain the invalid model setting or failed calculation.
  39                 raise ValueError(f"{filename}/{name}: nonfinite value")
  40         return values
  41 
  42     muscles = checked("muscle_tendon.py", MUSCLE_FIELDS, names)
  43     if set(muscles) != set(names):
  44         # Stop and explain the invalid model setting or failed calculation.
  45         raise ValueError("Muscle configuration does not match model channels")
  46     if s["muscle_params"].shape[1] < 15:
  47         s["muscle_params"] = np.pad(
  48             s["muscle_params"],
  49             ((0, 0), (0, 15 - s["muscle_params"].shape[1])),
  50         )
  51     for name, values in muscles.items():
  52         i = names.index(name)
  53         for field, (array, column, unit, meaning) in MUSCLE_FIELDS.items():
  54             value = float(values[field])
  55             if not np.isfinite(value):
  56                 # Stop and explain the invalid model setting or failed calculation.
  57                 raise ValueError(f"Nonfinite {name}.{field}")
  58             if column is None:
  59                 s[array][i] = value
  60             else:
  61                 s[array][i, column] = value
  62         if (
  63             min(
  64                 s["lopt"][i],
  65                 s["slack"][i],
  66                 s["e0"][i],
  67                 s["muscle_params"][i, 2],
  68                 s["muscle_params"][i, 6],
  69             )
  70             <= 0
  71         ):
  72             # Stop and explain the invalid model setting or failed calculation.
  73             raise ValueError(f"Nonpositive muscle scale: {name}")
  74         if not 0 < s["cosine"][i] <= 1 or s["fiber_damping_coefficients"][i] < 0:
  75             # Stop and explain the invalid model setting or failed calculation.
  76             raise ValueError(f"Invalid pennation or damping: {name}")
  77         if (
  78             values["active_force_length_exponent"] < 1
  79             or min(
  80                 values["source_shortening_curvature"],
  81                 values["source_lengthening_curvature"],
  82             )
  83             <= 0
  84         ):
  85             # Stop and explain the invalid model setting or failed calculation.
  86             raise ValueError(f"Invalid fiber curve coefficients: {name}")
  87         if any(
  88             values[key] not in (0, 1)
  89             for key in ["passive_fiber_enabled", "degroote_force_velocity_enabled"]
  90         ):
  91             # Stop and explain the invalid model setting or failed calculation.
  92             raise ValueError(f"Muscle switches must be 0 or 1: {name}")
  93         s["muscle_params"][i, 9] = s["cosine"][i]
  94     for name, values in checked(
  95         "activation.py",
  96         [
  97             "activation_time_constant_s",
  98             "deactivation_time_constant_s",
  99             "switch_smoothing_width",
 100         ],
 101         names,
 102     ).items():
 103         i = names.index(name)
 104         tau = [values["activation_time_constant_s"], values["deactivation_time_constant_s"]]
 105         if min(tau) <= 0 or values["switch_smoothing_width"] < 0:
 106             # Stop and explain the invalid model setting or failed calculation.
 107             raise ValueError("Invalid activation time constants/smoothing")
 108         s["time_constants"][i] = tau
 109         m.actuator_dynprm[i, :3] = tau + [values["switch_smoothing_width"]]
 110     for name, values in checked(
 111         "bodies.py",
 112         BODY_FIELDS,
 113         [m.body(i).name for i in range(1, m.nbody)],
 114     ).items():
 115         i = m.body(name).id
 116         for field, (array, unit, meaning) in BODY_FIELDS.items():
 117             getattr(m, array)[i] = values[field]
 118         if m.body_mass[i] < 0 or np.any(m.body_inertia[i] < 0):
 119             # Stop and explain the invalid model setting or failed calculation.
 120             raise ValueError(f"Negative mass or inertia: {name}")
 121         if np.max(m.body_inertia[i]) > np.sum(m.body_inertia[i]) / 2 + 1e-12:
 122             # Stop and explain the invalid model setting or failed calculation.
 123             raise ValueError(f"Inertia triangle inequality: {name}")
 124         for quat in [m.body_iquat[i], m.body_quat[i]]:
 125             if not np.isclose(np.linalg.norm(quat), 1.0, atol=1e-7):
 126                 # Stop and explain the invalid model setting or failed calculation.
 127                 raise ValueError(f"Quaternion must be unit length: {name}")
 128     for name, values in checked(
 129         "joints.py",
 130         JOINT_FIELDS,
 131         [m.joint(i).name for i in range(m.njnt)],
 132     ).items():
 133         i = m.joint(name).id
 134         for field, (array, unit, meaning) in JOINT_FIELDS.items():
 135             index = (
 136                 int(m.jnt_dofadr[i])
 137                 if array.startswith("dof_")
 138                 else int(m.jnt_qposadr[i]) if array == "qpos_spring" else i
 139             )
 140             getattr(m, array)[index] = values[field]
 141         if values["passive_stiffness"] < 0 or values["passive_damping"] < 0:
 142             # Stop and explain the invalid model setting or failed calculation.
 143             raise ValueError("Passive coefficients must be nonnegative")
 144         if not np.isclose(np.linalg.norm(m.jnt_axis[i]), 1.0, atol=1e-7):
 145             # Stop and explain the invalid model setting or failed calculation.
 146             raise ValueError(f"Joint axis must be unit length: {name}")
 147         if values["armature"] < 0 or values["range"][0] > values["range"][1]:
 148             # Stop and explain the invalid model setting or failed calculation.
 149             raise ValueError(f"Invalid armature/limits: {name}")
 150     pads = checked(
 151         "contact.py",
 152         list(CONTACT_FIELDS) + ["radius_m", "position_in_body_m"],
 153         [m.geom(int(i)).name for i in s["pad_ids"]],
 154     )
 155     for row, geom in enumerate(s["pad_ids"]):
 156         values = pads[m.geom(int(geom)).name]
 157         m.geom_size[geom, 0] = values["radius_m"]
 158         m.geom_pos[geom] = values["position_in_body_m"]
 159         for field, (column, unit, meaning) in CONTACT_FIELDS.items():
 160             s["contact_parameters"][row, column] = values[field]
 161         if (
 162             m.geom_size[geom, 0] <= 0
 163             or np.any(s["contact_parameters"][row] < 0)
 164             or min(s["contact_parameters"][row, [0, 2]]) <= 0
 165         ):
 166             # Stop and explain the invalid model setting or failed calculation.
 167             raise ValueError("Invalid pad geometry/material value")
 168     dynamic = {int(meta[5]) for meta in s["routing_meta"]}
 169     for name, values in checked(
 170         "attachments.py",
 171         ["position_in_body_m", "orientation_in_body_wxyz"],
 172         [m.site(i).name for i in range(m.nsite) if i not in dynamic],
 173     ).items():
 174         if np.shape(values["position_in_body_m"]) != (3,) or np.shape(
 175             values["orientation_in_body_wxyz"]
 176         ) != (4,):
 177             # Stop and explain the invalid model setting or failed calculation.
 178             raise ValueError(f"attachments.py/{name}: expected xyz and wxyz vectors")
 179         i = m.site(name).id
 180         if not np.array_equal(
 181             m.site_pos[i],
 182             values["position_in_body_m"],
 183         ) or not np.array_equal(m.site_quat[i], values["orientation_in_body_wxyz"]):
 184             # The compiled XML can mark a site as coincident with its body.
 185             # After an edit, that shortcut would ignore the new coordinates.
 186             m.site_sameframe[i] = 0
 187         m.site_pos[i] = values["position_in_body_m"]
 188         m.site_quat[i] = values["orientation_in_body_wxyz"]
 189         if not np.isclose(np.linalg.norm(m.site_quat[i]), 1.0, atol=1e-7):
 190             # Stop and explain the invalid model setting or failed calculation.
 191             raise ValueError(f"Invalid site quaternion: {name}")
 192     from .apply_wrapping_parameters import apply_wrapping_parameters
 193 
 194     apply_wrapping_parameters(s, folder / "wrapping.py")
 195     route_fields = [
 196         "origin_radius_m",
 197         "insertion_radius_m",
 198         "origin_angle_rad",
 199         "insertion_angle_rad",
 200         "proximal_wrap_radius_m",
 201         "proximal_wrap_sign",
 202         "proximal_joint_sign",
 203         "distal_wrap_radius_m",
 204         "distal_wrap_sign",
 205         "distal_joint_sign",
 206         "joint_spacing_m",
 207     ]
 208     routes = checked(
 209         "routing.py",
 210         route_fields,
 211         [m.site(int(meta[5])).name for meta in s["routing_meta"]],
 212     )
 213     for row, meta in enumerate(s["routing_meta"]):
 214         name = m.site(int(meta[5])).name
 215         values = routes[name]
 216         s["routing_geom"][row] = [
 217             values[key]
 218             for key in [
 219                 "origin_radius_m",
 220                 "insertion_radius_m",
 221                 "origin_angle_rad",
 222                 "insertion_angle_rad",
 223                 "proximal_wrap_radius_m",
 224                 "proximal_wrap_sign",
 225                 "proximal_joint_sign",
 226                 "distal_wrap_radius_m",
 227                 "distal_wrap_sign",
 228                 "distal_joint_sign",
 229                 "joint_spacing_m",
 230             ]
 231         ]
 232     environment = checked("environment.py")
 233     if set(environment) - {
 234         "gravity_m_s2",
 235         "solver_iterations",
 236         "solver_tolerance",
 237         "integrator",
 238     }:
 239         # Stop and explain the invalid model setting or failed calculation.
 240         raise ValueError("Unknown environment parameter")
 241     m.opt.gravity[:] = environment["gravity_m_s2"]
 242     m.opt.iterations = int(environment["solver_iterations"])
 243     m.opt.tolerance = float(environment["solver_tolerance"])
 244     # Read the configured native integration scheme; old parameter snapshots may omit it.
 245     integrator = environment.get("integrator", int(m.opt.integrator))
 246     # Accept only the four named engine schemes listed in environment.py.
 247     if integrator not in (0, 1, 2, 3):
 248         # Stop and explain the invalid model setting or failed calculation.
 249         raise ValueError("Invalid MuJoCo integration scheme")
 250     # Apply the choice before any dynamics steps are calculated.
 251     m.opt.integrator = int(integrator)
 252     if m.opt.iterations < 1 or m.opt.tolerance < 0:
 253         # Stop and explain the invalid model setting or failed calculation.
 254         raise ValueError("Invalid dynamics solver parameters")
 255     # MuJoCo recomputes all derived inertial constants immediately after this
 256     # function returns. Initial states and negative-time preparation follow.
 257     s["parameter_bundle_applied"] = str(folder)
