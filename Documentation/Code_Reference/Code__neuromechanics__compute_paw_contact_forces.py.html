  15 def compute_paw_contact_forces(state, advance_s=0.0):
  16     """Actual paw deformation/slip -> forces and body wrenches, all four paws."""
  17     model, data = state["model"], state["data"]
  18     pad_indices = state["pad_ids"]
  19     velocity = np.zeros((4, 6))
  20     position = np.zeros((4, 3))
  21     for i, site in enumerate(state["paw_ids"]):
  22         mj.mj_objectVelocity(model, data, mj.mjtObj.mjOBJ_SITE, int(site), velocity[i], 0)
  23         position[i] = data.site_xpos[site]
  24     paw_forces, center_of_pressure, wrench, depth, energy, power = (
  25         sum_paw_contact_forces_and_moments(
  26             data.geom_xpos[pad_indices],
  27             model.geom_size[pad_indices, 0],
  28             state["pad_side"],
  29             model.geom_bodyid[pad_indices],
  30             data.xipos,
  31             position,
  32             velocity,
  33             state["bristle"],
  34             advance_s,
  35             state["contact_parameters"],
  36             True,
  37         )
  38     )
  39     state.update(
  40         grf=paw_forces,
  41         center_of_pressure=center_of_pressure,
  42         contact_wrench=wrench,
  43         pad_depth=depth,
  44         pad_energy=energy,
  45         contact_power=power,
  46     )
