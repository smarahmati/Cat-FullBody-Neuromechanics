   1 """Assemble reference kinematics.
   2 
   3 Inputs/outputs: see the named function arguments and returned values.
   4 Lengths use metres, time seconds, force newtons, and angles radians unless stated.
   5 See Documentation/model_methods.html for equations and parameter definitions.
   6 """
   7 
   8 import numpy as np
   9 
  10 import mujoco as mj
  11 
  12 from model_configuration.paths import ROOT
  13 
  14 from .experimental_comparison_targets import experimental_comparison_targets
  15 
  16 from .periodic_interpolate import periodic_interpolate
  17 
  18 
  19 def assemble_reference_kinematics(state, count=240):
  20     """Offline scoring/initialization trajectory; never called by advance_partitioned_timestep.
  21 
  22     Each limb retains its source angle convention and a fixed assembly phase.
  23     The hind fitted initial offset is applied to the root, not to measurements.
  24     This assembles separate studies; it is not a recorded four-limb gait.
  25     """
  26     m = state["model"]
  27     T = state["settings"]["period_s"]
  28     time = np.arange(count) * T / count
  29     if state["settings"].get("reference_seed"):
  30         seed = np.load(ROOT / state["settings"]["reference_seed"])
  31         out = {}
  32         for key in [
  33             "pose",
  34             "velocity",
  35             "acceleration",
  36             "joint_q",
  37             "grf",
  38             "excitation",
  39             "scapula",
  40         ] + (["cop"] if "cop" in seed else []):
  41             value = seed[key]
  42             grid = np.arange(len(value)) / len(value)
  43             flat = value.reshape(len(value), -1).copy()
  44             if key == "pose":
  45                 stride = (
  46                     float(seed["stride_m"])
  47                     if "stride_m" in seed
  48                     else float(
  49                         np.load(ROOT / "Data/Experimental_Reference_Data/right_hindlimb_walking_reference.npz")["stride_m"]
  50                     )
  51                 )
  52                 flat[:, 0] -= stride * grid
  53             new = np.column_stack(
  54                 [np.interp(time / T, np.r_[grid, 1.0], np.r_[v, v[0]]) for v in flat.T]
  55             )
  56             if key == "pose":
  57                 new[:, 0] += stride * time / T
  58             out[key] = new.reshape((count,) + value.shape[1:])
  59         return dict(time=time, **out)
  60     ref = dict(np.load(ROOT / "Data/Experimental_Reference_Data/right_hindlimb_walking_reference.npz"))
  61     raw = ref["pose"]
  62     phase_grid = np.arange(len(raw)) / len(raw)
  63     pose = np.tile(state["qpos"], (count, 1))
  64     target = experimental_comparison_targets(time, T)
  65     interp = lambda x: np.interp(
  66         np.arange(count) / count,
  67         np.r_[phase_grid, 1.0],
  68         np.r_[x, x[0]],
  69     )
  70     stride = float(ref["stride_m"])
  71     detrend = raw[:, 0] - stride * phase_grid
  72     pose[:, 0] = interp(detrend) + stride * time / T
  73     pose[:, 1] = interp(raw[:, 2]) + (state["qpos"][1] - raw[0, 2])
  74     base = raw[0, 3:7]
  75     inverse = base * np.array([1, -1, -1, -1])
  76     angles = []
  77     for q in raw[:, 3:7]:
  78         rel = np.zeros(4)
  79         mj.mju_mulQuat(rel, q, inverse)
  80         angles.append(2 * np.arctan2(rel[2], rel[0]))
  81     pose[:, 2] = interp(np.unwrap(angles)) + state["qpos"][2]
  82     for k in range(4):
  83         pose[:, state["joint_indices"][k]] = target["joint_q"][:, k]
  84     # The separate fore study supplies shoulder heights and scapular angles.
  85     # Infer one common trunk pitch offline. This is an assembly assumption;
  86     # runtime lumbosacral motion is free and only a passive spring acts there.
  87     z = np.load(ROOT / "Data/Experimental_Reference_Data/hindlimb_forelimb_reference_cycles.npz")
  88     scratch = mj.MjData(m)
  89     spine = m.joint("lumbosacral_pitch").qposadr[0]
  90     for k, t in enumerate(time):
  91         wanted = np.mean(
  92             [
  93                 periodic_interpolate(
  94                     z["shoulder_z_by_side"][:, i],
  95                     t / T + 0.5 + [0.25, 0.75][i],
  96                 )
  97                 for i in range(2)
  98             ]
  99         )
 100         theta = 0.0
 101         for iteration in range(4):
 102             pose[k, spine] = theta - pose[k, 2]
 103             for i, side in enumerate(["FL", "FR"]):
 104                 pose[k, m.joint(side + "_scapulothoracic").qposadr[0]] = (
 105                     target["scapula"][k, i]
 106                     + theta
 107                 )
 108             scratch.qpos[:] = pose[k]
 109             # Recalculate positions and routed muscle lengths at the new pose.
 110             mj.mj_fwdPosition(m, scratch)
 111             actual = np.mean(
 112                 [scratch.xpos[m.body(side + "_upperarm").id, 2] for side in ["FL", "FR"]]
 113             )
 114             theta += np.clip((actual - wanted) / 0.2294, -0.10, 0.10)
 115         pose[k, spine] = theta - pose[k, 2]
 116         for i, side in enumerate(["FL", "FR"]):
 117             pose[k, m.joint(side + "_scapulothoracic").qposadr[0]] = (
 118                 target["scapula"][k, i]
 119                 + theta
 120             )
 121         # Match each source shoulder height with the passive sling translation.
 122         # This is OFFLINE reference assembly; during simulation the shoulder
 123         # translation is determined by inertia, muscle forces and contacts.
 124         scratch.qpos[:] = pose[k]
 125         # Recalculate positions and routed muscle lengths at the new pose.
 126         mj.mj_fwdPosition(m, scratch)
 127         for i, side in enumerate(["FL", "FR"]):
 128             wanted_i = periodic_interpolate(
 129                 z["shoulder_z_by_side"][:, i],
 130                 t / T + 0.5 + [0.25, 0.75][i],
 131             )
 132             actual_i = scratch.xpos[m.body(side + "_upperarm").id, 2]
 133             pose[k, m.joint(side + "_scapula_z").qposadr[0]] = (
 134                 wanted_i
 135                 - actual_i
 136             ) / np.cos(theta)
 137     from .periodic_kinematic_seed import periodic_kinematic_seed
 138 
 139     pose, velocity, acceleration = periodic_kinematic_seed(pose, T, stride, 8)
 140     return dict(
 141         time=time,
 142         pose=pose,
 143         velocity=velocity,
 144         acceleration=acceleration,
 145         **target
 146     )
