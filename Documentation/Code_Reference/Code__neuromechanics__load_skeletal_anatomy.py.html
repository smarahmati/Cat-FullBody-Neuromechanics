  12 def load_skeletal_anatomy(graphics=False, variant="base"):
  13     """Select an explicitly saved anatomy variant; base files are preserved."""
  14     directory = ROOT / "Data/Skeletal_Geometry/Model" if variant == "base" else ROOT / "Data/Skeletal_Geometry/Model" / variant
  15     root = ET.parse(directory / "cat.xml").getroot()
  16     meshdir = ROOT / "Data/Skeletal_Geometry/Model/meshes"
  17     root.find("compiler").set("meshdir", str(meshdir))
  18     use_meshes = graphics and all((meshdir / mesh.get("file", "")).is_file() for mesh in root.find("asset").findall("mesh"))
  19     if graphics and not use_meshes:
  20         print("Optional anatomical meshes absent: using schematic bones; mechanics are unchanged.")
  21     if not use_meshes:
  22         for geom in root.iter("geom"):
  23             if geom.get("type") == "mesh":
  24                 geom.set("type", "sphere")
  25                 geom.set("size", "1e-6")
  26                 if graphics:
  27                     geom.set("size", "0.008")
  28                     geom.set("rgba", "0.9 0.86 0.72 1")
  29                 geom.attrib.pop("mesh", None)
  30         for mesh in list(root.find("asset").findall("mesh")):
  31             root.find("asset").remove(mesh)
  32     if graphics and not use_meshes:
  33         for body in root.iter("body"):
  34             child = body.find("body")
  35             end = np.fromstring(child.get("pos", "0.03 0 0"), sep=" ") if child is not None else np.array([0.025, 0, 0])
  36             if np.linalg.norm(end) < 0.01:
  37                 end = np.array([0.03, 0, 0])
  38             for geom in body.findall("geom"):
  39                 if geom.get("mesh") is None and ("_bone" in geom.get("name", "") or "axial_surface" in geom.get("name", "")):
  40                     geom.set("type", "capsule")
  41                     geom.set("fromto", "0 0 0 " + " ".join(map(str,end)))
  42                     geom.attrib.pop("pos", None)
  43                     geom.attrib.pop("quat", None)
  44                     geom.set("size", "0.008")
  45     m = mj.MjModel.from_xml_string(ET.tostring(root, encoding="unicode"))
  46     for key, value in np.load(directory / "mechanics_exact.npz").items():
  47         if key.endswith("_names"):
  48             # Skip the remaining work for this item.
  49             continue
  50         if not use_meshes and key == "geom_size":
  51             ids = [
  52                 i
  53                 for i in range(m.ngeom)
  54                 if m.geom(i).name.find("_bone") < 0
  55                 and m.geom(i).name.find("axial_surface") < 0
  56             ]
  57             getattr(m, key)[ids] = value[ids]
  58         else:
  59             getattr(m, key)[:] = value
  60     # Mesh geoms never enter collisions and all mass/inertia is explicit.
  61     mj.mj_setConst(m, mj.MjData(m))
  62     return m
