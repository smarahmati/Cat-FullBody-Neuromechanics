  10 def walking_acceptance(result, settings):
  11     """Compare unfiltered signals and sustained contact cycles to fixed targets.
  12 
  13     These tolerances are engineering acceptance targets, not uncertainty
  14     intervals for the biological recordings. LH and fore EMG data limitations
  15     remain in compare_simulation_with_experiment(). Contact dwell is only event detection, never force
  16     smoothing or an alteration of plotted/scored Ground_Reaction_Forces.
  17     """
  18     t = result["time"]
  19     metrics = compare_simulation_with_experiment(result, settings)
  20     T = settings["period_s"]
  21     target = experimental_comparison_targets(t, T, settings.get("reference_table_dir"))
  22     late = t >= max(0.0, float(t[-1]) - 10.0)
  23     q = np.sqrt(
  24         np.mean(np.rad2deg(result["joint_q"][late] - target["joint_q"][late]) ** 2, axis=0)
  25     )
  26     grf = np.sqrt(np.mean((result["grf"][late] - target["grf"][late]) ** 2, axis=0))
  27     angle_limit = np.array([8.0, 6.0, 8.0, 8.0])[:, None]
  28     cycles = []
  29     dt = float(np.median(np.diff(t)))
  30     dwell = max(1, round(0.025 / dt))
  31     for limb in range(4):
  32         raw = result["grf"][:, limb, 2] > 1.0
  33         on = bool(raw[0])
  34         counter = 0
  35         touch = []
  36         off = []
  37         for k, value in enumerate(raw):
  38             counter = counter + 1 if bool(value) != on else 0
  39             if counter >= dwell:
  40                 on = bool(value)
  41                 (touch if on else off).append(float(t[k - dwell + 1]))
  42                 counter = 0
  43         recent = np.array(off)
  44         recent = recent[recent >= max(0.0, float(t[-1]) - 10.0)]
  45         periods = np.diff(recent)
  46         cycles.append(
  47             dict(
  48                 completed_cycles=max(0, len(off) - 1),
  49                 last_window_median_period_s=(
  50                     float(np.median(periods))
  51                     if len(periods)
  52                     else None
  53                 ),
  54                 period_relative_sd=(
  55                     float(np.std(periods) / np.mean(periods))
  56                     if len(periods) > 1
  57                     else None
  58                 ),
  59             )
  60         )
  61     sufficient = bool(float(t[-1]) >= 59.99 and not str(result["failure"]))
  62     rhythm = bool(
  63         sufficient
  64         and all(
  65             c["completed_cycles"] >= int(0.8 * float(t[-1]) / T)
  66             and c["last_window_median_period_s"] is not None
  67             and abs(c["last_window_median_period_s"] / T - 1) < 0.10
  68             and c["period_relative_sd"] is not None
  69             and c["period_relative_sd"] < 0.10
  70             for c in cycles
  71         )
  72     )
  73     force_signs = []
  74     for limb in range(4):
  75         fx = result["grf"][late, limb, 0]
  76         force_signs.append(bool(fx.min() < -0.5 and fx.max() > 0.5))
  77     emg_pass = all(
  78         e["peak_normalized_rmse"] < 0.25 and e["correlation"] > 0.7
  79         for e in metrics["emg"]
  80     )
  81     timing = []
  82     for item in metrics["emg"]:
  83         score = item["complete_input_cycle_timing"]
  84         edges = all(
  85             score[key]["unmatched_predicted"] == 0
  86             and score[key]["unmatched_reference"] == 0
  87             and (
  88                 score[key]["maximum_absolute_error_s"] is None
  89                 or score[key]["maximum_absolute_error_s"]
  90                 <= settings.get("emg_phase_edge_tolerance_s", 0.035)
  91             )
  92             for key in ["onset", "offset"]
  93         )
  94         timing.append(
  95             bool(
  96                 score["active_intersection_over_union"]
  97                 >= settings.get("emg_phase_iou_min", 0.8)
  98                 and edges
  99             )
 100         )
 101     tests = dict(
 102         uninterrupted_60s=sufficient,
 103         rhythmic_four_paw_contacts=rhythm,
 104         all_joint_RMSE=bool(np.all(q < angle_limit)),
 105         all_AP_RMSE=bool(np.all(grf[:, 0] < 0.6)),
 106         all_vertical_RMSE=bool(np.all(grf[:, 2] < 1.5)),
 107         braking_and_propulsion_all_paws=all(force_signs),
 108         available_EMG_shapes=bool(emg_pass),
 109     )
 110     tests["available_EMG_active_quiet_timing"] = bool(all(timing))
 111     if "max_interior_fiber_residual_N" in result:
 112         tests["interior_elastic_equilibrium"] = bool(
 113             float(result["max_interior_fiber_residual_N"])
 114             < 1e-4
 115         )
 116         tests["no_geometric_fiber_bounds"] = bool(int(result["fiber_bound_steps"]) == 0)
 117     else:
 118         tests["interior_elastic_equilibrium"] = False
 119         tests["no_geometric_fiber_bounds"] = False
 120     if settings.get("fore_force_velocity_mode") == "degroote_2016":
 121         tests["published_fore_force_velocity_domain"] = bool(
 122             "peak_abs_normalized_fiber_velocity" in result
 123             and np.max(result["peak_abs_normalized_fiber_velocity"][28:]) <= 1.0 + 1e-6
 124         )
 125     if settings.get("hind_force_velocity_mode") == "degroote_2016":
 126         tests["published_hind_force_velocity_domain"] = bool(
 127             "peak_abs_normalized_fiber_velocity" in result
 128             and np.max(result["peak_abs_normalized_fiber_velocity"][:28]) <= 1.0 + 1e-6
 129         )
 130     if "degroote_force_velocity_enabled" in result:
 131         enabled = np.asarray(result["degroote_force_velocity_enabled"], bool)
 132         if np.any(enabled):
 133             tests["configured_force_velocity_domain"] = bool(
 134                 np.max(result["peak_abs_normalized_fiber_velocity"][enabled])
 135                 <= 1.0 + 1e-6
 136             )
 137     return dict(
 138         tests=tests,
 139         primary_calibration_pass=bool(all(tests.values())),
 140         half_step_and_perturbation_still_required=True,
 141         late_window_start_s=float(t[late][0]),
 142         late_joint_RMSE_deg=q.tolist(),
 143         late_AP_vertical_RMSE_N=grf[:, [0, 2]].tolist(),
 144         contact_cycles=cycles,
 145         threshold_provenance="Declared engineering targets comparable to preserved hindlimb fit; not published biological tolerances.",
 146         independent_four_limb_validation=False,
 147     )
